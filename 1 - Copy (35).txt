<script>

/************************************************
 * STATE
 ************************************************/

let purchaseLines = [];

let posting = false;


/************************************************
 * INITIALIZE
 ************************************************/

document.addEventListener(
  "DOMContentLoaded",
  function () {

    setToday();

    addPurchaseLine();

  }
);


/************************************************
 * DATE
 ************************************************/

function setToday() {

  const input =
    document.getElementById(
      "purchaseDate"
    );

  const now =
    new Date();

  const year =
    now.getFullYear();

  const month =
    String(
      now.getMonth() + 1
    ).padStart(2, "0");

  const day =
    String(
      now.getDate()
    ).padStart(2, "0");

  input.value =
    year +
    "-" +
    month +
    "-" +
    day;

}


/************************************************
 * ADD LINE
 ************************************************/

function addPurchaseLine() {

  purchaseLines.push({

    itemCode: "",

    quantity: 1,

    unitCost: 0,

    discount: 0

  });

  renderLines();

}


/************************************************
 * REMOVE LINE
 ************************************************/

function removePurchaseLine(index) {

  if (
    purchaseLines.length === 1
  ) {

    showError(
      "At least one purchase item is required."
    );

    return;

  }

  purchaseLines.splice(
    index,
    1
  );

  renderLines();

}


/************************************************
 * UPDATE LINE
 ************************************************/

function updateLine(
  index,
  field,
  value
) {

  if (
    !purchaseLines[index]
  ) {
    return;
  }

  if (
    field === "itemCode"
  ) {

    purchaseLines[index][field] =
      String(value || "").trim();

  } else {

    const number =
      Number(value);

    purchaseLines[index][field] =
      Number.isFinite(number)
        ? number
        : 0;

  }

  recalculate();

}


/************************************************
 * RENDER LINES
 ************************************************/

function renderLines() {

  const body =
    document.getElementById(
      "itemsBody"
    );

  body.innerHTML = "";


  if (
    purchaseLines.length === 0
  ) {

    body.innerHTML =
      '<tr><td colspan="9" class="empty">No items</td></tr>';

    recalculate();

    return;

  }


  purchaseLines.forEach(
    function(line, index) {

      const gross =
        line.quantity *
        line.unitCost;

      const discount =
        line.discount;

      const net =
        Math.max(
          0,
          gross - discount
        );


      const row =
        document.createElement(
          "tr"
        );


      row.innerHTML = `

        <td>
          ${index + 1}
        </td>

        <td>
          <input
            class="item-code"
            type="text"
            value="${escapeHtml(line.itemCode)}"
            oninput="updateLine(
              ${index},
              'itemCode',
              this.value
            )"
          >
        </td>

        <td>
          <input
            class="item-name"
            type="text"
            value=""
            placeholder="Server validates item"
            readonly
          >
        </td>

        <td>
          <input
            type="number"
            min="0.001"
            step="0.001"
            value="${line.quantity}"
            oninput="updateLine(
              ${index},
              'quantity',
              this.value
            )"
          >
        </td>

        <td>
          <input
            type="text"
            value=""
            placeholder="Server"
            readonly
          >
        </td>

        <td>
          <input
            type="number"
            min="0"
            step="0.001"
            value="${line.unitCost}"
            oninput="updateLine(
              ${index},
              'unitCost',
              this.value
            )"
          >
        </td>

        <td>
          <input
            type="number"
            min="0"
            step="0.001"
            value="${line.discount}"
            oninput="updateLine(
              ${index},
              'discount',
              this.value
            )"
          >
        </td>

        <td class="line-total">
          ${formatNumber(net)}
        </td>

        <td>
          <button
            type="button"
            class="btn-danger"
            onclick="removePurchaseLine(${index})"
          >
            Remove
          </button>
        </td>

      `;

      body.appendChild(row);

    }
  );


  recalculate();

}


/************************************************
 * RECALCULATE
 *
 * UI preview only.
 *
 * Final calculation belongs to
 * createPurchase().
 ************************************************/

function recalculate() {

  let subtotal = 0;

  purchaseLines.forEach(
    function(line) {

      const quantity =
        Number(line.quantity) || 0;

      const unitCost =
        Number(line.unitCost) || 0;

      const discount =
        Number(line.discount) || 0;

      subtotal +=
        Math.max(
          0,
          quantity * unitCost - discount
        );

    }
  );


  const documentDiscount =
    Number(
      document.getElementById(
        "documentDiscount"
      ).value
    ) || 0;


  const taxable =
    Math.max(
      0,
      subtotal - documentDiscount
    );


  document.getElementById(
    "summarySubtotal"
  ).textContent =
    formatNumber(subtotal);


  document.getElementById(
    "summaryDiscount"
  ).textContent =
    formatNumber(documentDiscount);


  document.getElementById(
    "summaryTaxable"
  ).textContent =
    formatNumber(taxable);


  /*
   * Tax is intentionally NOT calculated
   * inside the browser.
   *
   * Server-side Tax Engine owns it.
   */

  document.getElementById(
    "summaryTotal"
  ).textContent =
    "Calculated on Post";

}


/************************************************
 * COLLECT DATA
 ************************************************/

function collectPurchaseData() {

  const referenceNo =
    getValue("referenceNo");

  const date =
    getValue("purchaseDate");

  const supplierCode =
    getValue("supplierCode");

  const warehouseCode =
    getValue("warehouseCode");

  const paymentType =
    getValue("paymentType");

  const dueDate =
    getValue("dueDate");

  const currency =
    getValue("currency") ||
    "EGP";

  const discount =
    Number(
      getValue(
        "documentDiscount"
      )
    ) || 0;


  const lines =
    purchaseLines.map(
      function(line) {

        return {

          itemCode:
            String(
              line.itemCode || ""
            ).trim(),

          quantity:
            Number(
              line.quantity
            ),

          unitCost:
            Number(
              line.unitCost
            ),

          discount:
            Number(
              line.discount
            ) || 0

        };

      }
    );


  return {

    referenceNo:
      referenceNo,

    date:
      date,

    supplierCode:
      supplierCode,

    warehouseCode:
      warehouseCode,

    paymentType:
      paymentType,

    dueDate:
      dueDate,

    currency:
      currency,

    discount:
      discount,

    lines:
      lines

  };

}


/************************************************
 * VALIDATE CLIENT INPUT
 *
 * Server validation remains authoritative.
 ************************************************/

function validateClientData(
  data
) {

  if (
    !data.supplierCode
  ) {

    throw new Error(
      "Supplier Code is required."
    );

  }


  if (
    !data.warehouseCode
  ) {

    throw new Error(
      "Warehouse Code is required."
    );

  }


  if (
    !Array.isArray(data.lines) ||
    data.lines.length === 0
  ) {

    throw new Error(
      "At least one purchase item is required."
    );

  }


  data.lines.forEach(
    function(line, index) {

      if (
        !line.itemCode
      ) {

        throw new Error(
          "Item Code is required on line " +
          (index + 1)
        );

      }


      if (
        !Number.isFinite(
          line.quantity
        ) ||
        line.quantity <= 0
      ) {

        throw new Error(
          "Invalid quantity on line " +
          (index + 1)
        );

      }


      if (
        !Number.isFinite(
          line.unitCost
        ) ||
        line.unitCost < 0
      ) {

        throw new Error(
          "Invalid unit cost on line " +
          (index + 1)
        );

      }


      if (
        !Number.isFinite(
          line.discount
        ) ||
        line.discount < 0
      ) {

        throw new Error(
          "Invalid line discount on line " +
          (index + 1)
        );

      }

    }
  );


  if (
    !["CASH", "CREDIT"]
      .includes(
        data.paymentType
      )
  ) {

    throw new Error(
      "Payment Type must be CASH or CREDIT."
    );

  }


  if (
    !Number.isFinite(
      data.discount
    ) ||
    data.discount < 0
  ) {

    throw new Error(
      "Invalid purchase discount."
    );

  }

}


/************************************************
 * POST PURCHASE
 *
 * ONLY SERVER FUNCTION CALLED:
 *
 *   createPurchase(data)
 ************************************************/

function postPurchase() {

  if (posting) {
    return;
  }


  clearMessage();


  try {

    const data =
      collectPurchaseData();


    validateClientData(
      data
    );


    posting = true;

    setPostingState(
      true
    );


    setStatus(
      "Posting purchase..."
    );


    google.script.run

      .withSuccessHandler(
        function(result) {

          posting = false;

          setPostingState(
            false
          );


          if (
            !result ||
            result.success !== true
          ) {

            showError(
              "Purchase was not posted."
            );

            setStatus(
              ""
            );

            return;

          }


          showSuccess(
            "Purchase posted successfully.\n\n" +
            "Purchase ID: " +
            (result.purchaseId || "") +
            "\n" +
            "Reference: " +
            (result.referenceNo || "") +
            "\n" +
            "Net Total: " +
            formatNumber(
              result.netTotal
            ) +
            "\n" +
            "Status: " +
            (result.status || "POSTED")
          );


          setStatus(
            "Purchase posted successfully."
          );


          /*
           * Do not automatically reset the form.
           *
           * The posted document must remain visible
           * to the user.
           */

        }
      )

      .withFailureHandler(
        function(error) {

          posting = false;

          setPostingState(
            false
          );


          const message =
            error &&
            error.message
              ? error.message
              : String(error);


          showError(
            message
          );


          setStatus(
            "Posting failed."
          );

        }
      )

      .createPurchase(
        data
      );

  } catch (error) {

    posting = false;

    setPostingState(
      false
    );


    showError(
      error.message ||
      String(error)
    );


    setStatus(
      ""
    );

  }

}


/************************************************
 * RESET
 ************************************************/

function resetPurchaseForm() {

  if (posting) {
    return;
  }


  clearMessage();


  document.getElementById(
    "referenceNo"
  ).value = "";


  document.getElementById(
    "supplierCode"
  ).value = "";


  document.getElementById(
    "warehouseCode"
  ).value = "";


  document.getElementById(
    "paymentType"
  ).value = "CASH";


  document.getElementById(
    "dueDate"
  ).value = "";


  document.getElementById(
    "currency"
  ).value = "EGP";


  document.getElementById(
    "documentDiscount"
  ).value = "0";


  purchaseLines = [];


  addPurchaseLine();


  setToday();


  setStatus(
    ""
  );

}


/************************************************
 * POSTING STATE
 ************************************************/

function setPostingState(
  active
) {

  const button =
    document.getElementById(
      "postButton"
    );


  button.disabled =
    active;


  button.textContent =
    active
      ? "Posting..."
      : "Post Purchase";

}

 * STATUS

) {

}

}

) {

  const element =

    );

  element.textContent =

}


function clearMessage() {

  const element =
    document.getElementById(
      "message"
    );


  element.className =
    "message";


  element.textContent =
    "";

}


/************************************************
 * HELPERS
 ************************************************/

function getValue(
  id
) {

  const element =
    document.getElementById(
      id
    );


  return element
    ? String(
        element.value || ""
      ).trim()
    : "";

}


function formatNumber(
  value
) {

  const number =
    Number(value);


  if (
    !Number.isFinite(number)
  ) {

    return "0.000";

  }


  return number.toFixed(3);

}


function escapeHtml(
  value
) {

  return String(
    value || ""
  )
    .replace(
      /&/g,
      "&amp;"
    )
    .replace(
      /</g,
      "&lt;"
    )
    .replace(
      />/g,
      "&gt;"
    )
    .replace(
      /"/g,
      "&quot;"
    )
    .replace(
      /'/g,
      "&#039;"
    );

}
