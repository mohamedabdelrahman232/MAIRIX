    /************************************************************
     * MAIRIX ERP
     * UI ENTRY POINT
     *
     * FILE:
     * MAIRIX_UI_Index.html
     *
     * PURPOSE:
     *
     * 1. Start the MAIRIX UI.
     * 2. Ask the server for the authenticated user context.
     * 3. Never trust a UI-supplied User ID.
     * 4. Hand the server result to the UI application shell.
     *
     * IMPORTANT:
     *
     * This file does NOT perform authorization.
     *
     * Server-side authorization remains authoritative.
     *
     * The UI only uses the returned context to decide what
     * should be rendered or hidden.
     ************************************************************/


    /************************************************************
     * DOM HELPERS
     ************************************************************/

    function mairixUIElement(
      id
    ) {

      return document.getElementById(
        id
      );

    }


    /************************************************************
     * SHOW LOADING
     ************************************************************/

    function mairixUIShowLoading(
      message
    ) {

      var loading =
        mairixUIElement(
          "mairixLoading"
        );

      var error =
        mairixUIElement(
          "mairixError"
        );

      var root =
        mairixUIElement(
          "mairixApplicationRoot"
        );


      if (loading) {

        loading.style.display =
          "flex";

      }


      if (error) {

        error.style.display =
          "none";

      }


      if (root) {

        root.style.display =
          "none";

      }


      var messageElement =
        mairixUIElement(
          "mairixLoadingMessage"
        );


      if (
        messageElement &&
        message
      ) {

        messageElement.textContent =
          message;

      }

    }


    /************************************************************
     * SHOW ERROR
     ************************************************************/

    function mairixUIShowError(
      message
    ) {

      var loading =
        mairixUIElement(
          "mairixLoading"
        );

      var error =
        mairixUIElement(
          "mairixError"
        );

      var root =
        mairixUIElement(
          "mairixApplicationRoot"
        );

      var messageElement =
        mairixUIElement(
          "mairixErrorMessage"
        );


      if (loading) {

        loading.style.display =
          "none";

      }


      if (root) {

        root.style.display =
          "none";

      }


      if (messageElement) {

        messageElement.textContent =
          message ||
          "حدث خطأ أثناء تحميل النظام.";

      }


      if (error) {

        error.style.display =
          "flex";

      }

    }


    /************************************************************
     * SHOW APPLICATION ROOT
     ************************************************************/

    function mairixUIShowApplication()
    {

      var loading =
        mairixUIElement(
          "mairixLoading"
        );

      var error =
        mairixUIElement(
          "mairixError"
        );

      var root =
        mairixUIElement(
          "mairixApplicationRoot"
        );


      if (loading) {

        loading.style.display =
          "none";

      }


      if (error) {

        error.style.display =
          "none";

      }


      if (root) {

        root.style.display =
          "block";

      }

    }


    /************************************************************
     * SERVER ERROR NORMALIZER
     ************************************************************/

    function mairixUINormalizeServerError(
      error
    ) {

      if (!error) {

        return "تعذر الاتصال بالخادم.";

      }


      if (
        typeof error ===
        "string"
      ) {

        return error;

      }


      if (
        error.message
      ) {

        return String(
          error.message
        );

      }


      try {

        return JSON.stringify(
          error
        );

      } catch (
        stringifyError
      ) {

        return "حدث خطأ غير معروف.";

      }

    }


    /************************************************************
     * BOOTSTRAP APPLICATION
     ************************************************************/

    function mairixUIBootstrap(
      context
    ) {

      /*
       * IMPORTANT:
       *
       * We do NOT decide whether this user is authorized
       * to perform transactions here.
       *
       * The server/API remains authoritative.
       *
       * The context is only passed to the UI shell so the
       * navigation and visible components can be rendered.
       */

      if (
        !context ||
        context.success !== true
      ) {

        mairixUIShowError(
          "تعذر الحصول على بيانات المستخدم الحالي."
        );

        return;

      }


      if (
        typeof mairixUIInitializeApp ===
        "function"
      ) {

        mairixUIInitializeApp(
          context
        );

        mairixUIShowApplication();

        return;

      }


      /*
       * Temporary foundation fallback.
       *
       * This exists only until the Application Shell file
       * is loaded.
       */

      var root =
        mairixUIElement(
          "mairixApplicationRoot"
        );


      if (root) {

        root.innerHTML =

          "<div " +
          "style=\"" +
          "padding:24px;" +
          "background:#fff;" +
          "margin:20px;" +
          "border-radius:10px;" +
          "\">" +

          "<h2>MAIRIX ERP</h2>" +

          "<p>" +
          "تم التحقق من جلسة المستخدم بنجاح." +
          "</p>" +

          "</div>";

      }


      mairixUIShowApplication();

    }


    /************************************************************
     * LOAD CURRENT AUTHENTICATED USER
     ************************************************************/

    function mairixUILoadCurrentUser()
    {

      mairixUIShowLoading(
        "جاري التحقق من المستخدم..."
      );


      /*
       * IMPORTANT:
       *
       * The UI does NOT send a userId.
       *
       * The server determines the authenticated user.
       *
       * This follows the existing MAIRIX security contract.
       */

      if (
        typeof google ===
        "undefined" ||
        !google.script ||
        !google.script.run
      ) {

        mairixUIShowError(
          "بيئة MAIRIX غير متاحة."
        );

        return;

      }


      google
        .script
        .run
        .withSuccessHandler(
          function(result) {

            if (
              !result ||
              result.success !== true
            ) {

              mairixUIShowError(
                (
                  result &&
                  result.error &&
                  result.error.message
                ) ||
                "تعذر تحميل المستخدم الحالي."
              );

              return;

            }


            mairixUIBootstrap(
              result
            );

          }
        )
        .withFailureHandler(
          function(error) {

            mairixUIShowError(
              mairixUINormalizeServerError(
                error
              )
            );

          }
        )
        .apiGetCurrentUser();

    }


    /************************************************************
     * INITIAL START
     ************************************************************/

    document.addEventListener(

      function() {

        mairixUILoadCurrentUser();

      }
