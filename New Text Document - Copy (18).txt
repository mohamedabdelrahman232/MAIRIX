 * MAIRIX ERP
 * RUNTIME IDENTITY & PERMISSION MERGE
 *
 * FILE:
 * 35-C-MAIRIX_RuntimeMerge.gs
 *
 * PURPOSE:
 *
 * CENTRALIZE THE RUNTIME PATH:
 *
 * USERS
 *   ↓
 * CURRENT USER
 *   ↓
 * ROLE CODE
 *   ↓
 * ROLE_PERMISSIONS
 *   ↓
 * PERMISSION DECISION
 *
 * WAREHOUSE ACCESS:
 *
 * USERS
 *   ↓
 * USER_WAREHOUSES
 *   ↓
 * ACCESS DECISION
 *
 * IMPORTANT ARCHITECTURE:
 *
 * EMPLOYEES DOES NOT OWN USER ID.
 *
 * EMPLOYEES
 *     Employee Code
 *
 * USERS
 *     User ID
 *     Employee Code

 ************************************************/


/************************************************
 * CONFIG
 ************************************************/

const MAIRIX_RUNTIME_MERGE = Object.freeze({

  SHEETS: {

    USERS: "USERS",

    ROLES: "ROLES",

    ROLE_PERMISSIONS: "ROLE_PERMISSIONS",

    USER_PERMISSIONS: "USER_PERMISSIONS",

    EMPLOYEES: "EMPLOYEES",

    USER_WAREHOUSES: "USER_WAREHOUSES"

  },

  SESSION_PROPERTY:

    "MAIRIX_CURRENT_USER_ID"

});


/************************************************
 * TEXT HELPER
 ************************************************/

function mairixRuntimeCleanText(value) {

  if (
    value === null ||
    value === undefined
  ) {
    return "";
  }

  return String(value).trim();

}


/************************************************
 * NORMALIZE KEY
 ************************************************/

function mairixRuntimeNormalizeKey(value) {

  return mairixRuntimeCleanText(value)
    .toUpperCase();

}


/************************************************
 * BOOLEAN HELPER
 ************************************************/

function mairixRuntimeIsTrue(value) {

  if (value === true) {
    return true;
  }

  if (value === 1) {
    return true;
  }

  const normalized =
    mairixRuntimeNormalizeKey(value);

  return (
    normalized === "TRUE" ||
    normalized === "YES" ||
    normalized === "Y" ||
    normalized === "1"
  );

}


/************************************************
 * GET SHEET
 ************************************************/

function mairixRuntimeGetSheet(sheetName) {

  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(sheetName);

  if (!sheet) {

    throw new Error(
      "Required sheet not found: " +
      sheetName
    );

  }

  return sheet;

}


/************************************************
 * GET HEADERS
 ************************************************/

function mairixRuntimeGetHeaders(sheet) {

  if (!sheet) {

    throw new Error(
      "Sheet is required."
    );

  }

  if (sheet.getLastColumn() < 1) {

    throw new Error(
      "Sheet has no columns: " +
      sheet.getName()
    );

  }

  return sheet
    .getRange(
      1,
      1,
      1,
      sheet.getLastColumn()
    )
    .getDisplayValues()[0]
    .map(function(header) {

      return mairixRuntimeCleanText(header);

    });

}


/************************************************
 * FIND HEADER INDEX
 ************************************************/

function mairixRuntimeHeaderIndex(
  headers,
  headerName,
  required
) {

  const index =
    headers.indexOf(headerName);

  if (
    index === -1 &&
    required === true
  ) {

    throw new Error(
      "Required header missing: " +
      headerName
    );

  }

  return index;

}


/************************************************
 * READ SHEET RECORDS
 *
 * RETURNS:
 * [
 *   {
 *     headerName: value
 *   }
 * ]
 ************************************************/

function mairixRuntimeReadRecords(sheetName) {

  const sheet =
    mairixRuntimeGetSheet(
      sheetName
    );

  const lastRow =
    sheet.getLastRow();

  const headers =
    mairixRuntimeGetHeaders(
      sheet
    );

  if (lastRow < 2) {
    return [];
  }

  const values =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        headers.length
      )
      .getDisplayValues();

  return values.map(function(row) {

    const record = {};

    headers.forEach(
      function(header, index) {

        record[header] =
          row[index];

      }
    );

    return record;

  });

}


/************************************************
 * FIND USER BY USER ID
 *
 * OFFICIAL IDENTITY SOURCE:
 *
 * USERS
 ************************************************/

function mairixRuntimeGetUserById(userId) {

  const normalizedUserId =
    mairixRuntimeCleanText(userId);

  if (!normalizedUserId) {

    throw new Error(
      "User ID is required."
    );

  }

  const users =
    mairixRuntimeReadRecords(
      MAIRIX_RUNTIME_MERGE.SHEETS.USERS
    );

  for (
    let i = 0;
    i < users.length;
    i++
  ) {

    const row =
      users[i];

    if (
      mairixRuntimeCleanText(
        row["User ID"]
      ) === normalizedUserId
    ) {

      return mairixRuntimeNormalizeUser(
        row
      );

    }

  }

  return null;

}


/************************************************
 * FIND USER BY EMAIL
 ************************************************/

function mairixRuntimeGetUserByEmail(email) {

  const normalizedEmail =
    mairixRuntimeCleanText(email)
      .toLowerCase();

  if (!normalizedEmail) {
    return null;
  }

  const users =
    mairixRuntimeReadRecords(
      MAIRIX_RUNTIME_MERGE.SHEETS.USERS
    );

  for (
    let i = 0;
    i < users.length;
    i++
  ) {

    const row =
      users[i];

    const rowEmail =
      mairixRuntimeCleanText(
        row["Gmail"]
      ).toLowerCase();

    if (
      rowEmail === normalizedEmail
    ) {

      return mairixRuntimeNormalizeUser(
        row
      );

    }

  }

  return null;

}


/************************************************
 * NORMALIZE USER
 *
 * THIS IS THE SINGLE OFFICIAL
 * RUNTIME USER OBJECT.
 ************************************************/

function mairixRuntimeNormalizeUser(user) {

  if (!user) {

    throw new Error(
      "User object is required."
    );

  }

  const userId =
    mairixRuntimeCleanText(
      user.userId ||
      user["User ID"]
    );

  const employeeCode =
    mairixRuntimeCleanText(
      user.employeeCode ||
      user["Employee Code"]
    );

  const email =
    mairixRuntimeCleanText(
      user.email ||
      user["Gmail"]
    );

  const userName =
    mairixRuntimeCleanText(
      user.userName ||
      user.name ||
      user["Name"]
    );

  const roleCode =
    mairixRuntimeNormalizeKey(
      user.roleCode ||
      user["Role Code"] ||
      user.role
    );

  const roleName =
    mairixRuntimeCleanText(
      user.roleName ||
      user["Role Name"]
    );

  const activeValue =
    user.active !== undefined
      ? user.active
      : user["Active"];

  if (!userId) {

    throw new Error(
      "User has no User ID."
    );

  }

  if (!employeeCode) {

    throw new Error(
      "User has no Employee Code: " +
      userId
    );

  }

  if (!roleCode) {

    throw new Error(
      "User has no Role Code: " +
      userId
    );

  }

  return {

    userId:
      userId,

    employeeCode:
      employeeCode,

    userName:
      userName,

    email:
      email,

    role:
      roleCode,

    roleCode:
      roleCode,

    roleName:
      roleName,

    active:
      mairixRuntimeIsTrue(
        activeValue
      )

  };

}


/************************************************
 * GET CURRENT USER ID
 ************************************************/

function mairixRuntimeGetCurrentUserId() {

  const userId =
    PropertiesService
      .getUserProperties()
      .getProperty(
        MAIRIX_RUNTIME_MERGE.SESSION_PROPERTY
      );

  return mairixRuntimeCleanText(
    userId
  );

}


/************************************************
 * GET CURRENT USER
 *
 * OFFICIAL RUNTIME RESOLVER.
 ************************************************/

function mairixRuntimeGetCurrentUser() {

  const userId =
    mairixRuntimeGetCurrentUserId();

  if (!userId) {

    return null;

  }

  const user =
    mairixRuntimeGetUserById(
      userId
    );

  if (!user) {

    PropertiesService
      .getUserProperties()
      .deleteProperty(
        MAIRIX_RUNTIME_MERGE
          .SESSION_PROPERTY
      );

    return null;

  }

  if (
    user.active !== true
  ) {

    PropertiesService
      .getUserProperties()
      .deleteProperty(
        MAIRIX_RUNTIME_MERGE
          .SESSION_PROPERTY
      );

    return null;

  }

  return user;

}


/************************************************
 * REQUIRE CURRENT USER
 *
 * CANONICAL INTERNAL VERSION
 ************************************************/

function mairixRuntimeRequireCurrentUser() {

  const user =
    mairixRuntimeGetCurrentUser();

  if (
    !user ||
    !user.userId
  ) {

    throw new Error(
      "No authenticated MAIRIX user session."
    );

  }

  return user;

}


/************************************************
 * GET ROLE
 ************************************************/

function mairixRuntimeGetRole(roleCode) {

  const normalizedRoleCode =
    mairixRuntimeNormalizeKey(
      roleCode
    );

  if (!normalizedRoleCode) {

    throw new Error(
      "Role Code is required."
    );

  }

  const roles =
    mairixRuntimeReadRecords(
      MAIRIX_RUNTIME_MERGE.SHEETS.ROLES
    );

  for (
    let i = 0;
    i < roles.length;
    i++
  ) {

    const row =
      roles[i];

    if (

      mairixRuntimeNormalizeKey(
        row["Role Code"]
      ) === normalizedRoleCode

    ) {

      return {

        roleCode:
          normalizedRoleCode,

        roleName:
          mairixRuntimeCleanText(
            row["Role Name"]
          ),

        active:
          mairixRuntimeIsTrue(
            row["Active"]
          )

      };

    }

  }

  return null;

}


/************************************************
 * GET ROLE PERMISSION ROW
 *
 * ROLE_PERMISSIONS
 *
 * Role Code
 * Permission Code
 * View
 * Create
 * Edit
 * Delete
 * Export
 * Active
 ************************************************/

function mairixRuntimeGetRolePermission(

  roleCode,

  permissionCode

) {

  const normalizedRoleCode =
    mairixRuntimeNormalizeKey(
      roleCode
    );

  const normalizedPermissionCode =
    mairixRuntimeNormalizeKey(
      permissionCode
    );

  const records =
    mairixRuntimeReadRecords(
      MAIRIX_RUNTIME_MERGE
        .SHEETS
        .ROLE_PERMISSIONS
    );

  for (
    let i = 0;
    i < records.length;
    i++
  ) {

    const row =
      records[i];

    const rowRoleCode =
      mairixRuntimeNormalizeKey(
        row["Role Code"]
      );

    const rowPermissionCode =
      mairixRuntimeNormalizeKey(
        row["Permission Code"]
      );

    if (

      rowRoleCode === normalizedRoleCode &&

      rowPermissionCode ===
        normalizedPermissionCode

    ) {

      return {

        roleCode:
          rowRoleCode,

        permissionCode:
          rowPermissionCode,

        view:
          mairixRuntimeIsTrue(
            row["View"]
          ),

        create:
          mairixRuntimeIsTrue(
            row["Create"]
          ),

        edit:
          mairixRuntimeIsTrue(
            row["Edit"]
          ),

        delete:
          mairixRuntimeIsTrue(
            row["Delete"]
          ),

        export:
          mairixRuntimeIsTrue(
            row["Export"]
          ),

        active:
          mairixRuntimeIsTrue(
            row["Active"]
          )

      };

    }

  }

  return null;

}


/************************************************
 * ACTION RESOLUTION
 ************************************************/

function mairixRuntimePermissionActionAllowed(
  permission,
  action
) {

  if (!permission) {
    return false;
  }

  if (
    permission.active !== true
  ) {
    return false;
  }

  const normalizedAction =
    mairixRuntimeNormalizeKey(
      action
    );

  switch (
    normalizedAction
  ) {

    case "VIEW":
      return permission.view === true;

    case "CREATE":
      return permission.create === true;

    case "EDIT":
      return permission.edit === true;

    case "DELETE":
      return permission.delete === true;

    case "EXPORT":
      return permission.export === true;

    default:

      throw new Error(
        "Unsupported permission action: " +
        action
      );

  }

}


/************************************************
 * OFFICIAL RUNTIME HAS PERMISSION
 *
 * IMPORTANT:
 *
 * DOES NOT READ EMPLOYEES.
 *
 * PATH:
 *
 * USER ID
 *    ↓
 * USERS
 *    ↓
 * ROLE CODE
 *    ↓
 * ROLE_PERMISSIONS
 ************************************************/

function mairixRuntimeMergedHasPermission(

  userId,

  permissionCode,

  action

) {

  const user =
    mairixRuntimeGetUserById(
      userId
    );

  if (!user) {

    throw new Error(
      "MAIRIX user not found: " +
      userId
    );

  }

  if (
    user.active !== true
  ) {

    return false;

  }


  const role =
    mairixRuntimeGetRole(
      user.roleCode
    );

  if (!role) {

    throw new Error(
      "Role not found: " +
      user.roleCode
    );

  }

  if (
    role.active !== true
  ) {

    return false;

  }


  const permission =
    mairixRuntimeGetRolePermission(

      user.roleCode,

      permissionCode

    );


  return mairixRuntimePermissionActionAllowed(

    permission,

    action

  );

}


/************************************************
 * REQUIRE PERMISSION
 *
 * NEW MERGED VERSION
 ************************************************/

function mairixRuntimeMergedRequirePermission(

  userId,

  permissionCode,

  action

) {

  const allowed =
    mairixRuntimeMergedHasPermission(

      userId,

      permissionCode,

      action

    );

  if (
    allowed !== true
  ) {

    throw new Error(

      "Permission denied: " +

      permissionCode +

      " / " +

      action

    );

  }

  return true;

}

function mairixRuntimeCurrentUserHasPermission(

  action

  const user =

function mairixRuntimeCurrentUserRequirePermission(

) {

}


/************************************************
 * USER WAREHOUSES
 *
 * OFFICIAL WAREHOUSE SCOPE
 ************************************************/

function mairixRuntimeGetUserWarehouses(
  userId
) {

  const normalizedUserId =
    mairixRuntimeCleanText(
      userId
    );

  const records =
    mairixRuntimeReadRecords(
      MAIRIX_RUNTIME_MERGE
        .SHEETS
        .USER_WAREHOUSES
    );

  const result = [];

  for (
    let i = 0;
    i < records.length;
    i++
  ) {

    const row =
      records[i];

    if (

      mairixRuntimeCleanText(
        row["User ID"]
      ) !== normalizedUserId

    ) {
      continue;
    }

    if (
      !mairixRuntimeIsTrue(
        row["Active"]
      )
    ) {
      continue;
    }

    result.push({

      userId:
        normalizedUserId,

      employeeCode:
        mairixRuntimeCleanText(
          row["Employee Code"]
        ),

      warehouseCode:
        mairixRuntimeCleanText(
          row["Warehouse Code"]
        ),

      warehouseName:
        mairixRuntimeCleanText(
          row["Warehouse Name"]
        ),

      active:
        true

    });

  }

  return result;

}


/************************************************
 * WAREHOUSE ACCESS
 ************************************************/

function mairixRuntimeMergedUserHasWarehouseAccess(

  userId,

  warehouseCode

) {

  const normalizedWarehouseCode =
    mairixRuntimeCleanText(
      warehouseCode
    );

  if (!normalizedWarehouseCode) {

    return false;

  }

  const warehouses =
    mairixRuntimeGetUserWarehouses(
      userId
    );

  for (
    let i = 0;
    i < warehouses.length;
    i++
  ) {

    if (

      mairixRuntimeCleanText(
        warehouses[i].warehouseCode
      ) === normalizedWarehouseCode

    ) {

      return true;

    }

  }

  return false;

}


/************************************************
 * FULL ACCESS CONTEXT
 *
 * Useful for UI and transaction engines.
 ************************************************/

function mairixRuntimeGetAccessContext() {

  const user =
    mairixRuntimeRequireCurrentUser();

  const warehouses =
    mairixRuntimeGetUserWarehouses(
      user.userId
    );

  return {

    user: user,

    warehouses: warehouses

  };

}

/************************************************
 * MAIRIX ERP
 *
 * RUNTIME PERMISSION ENGINE
 *
 * FILE:
 * MAIRIX_RuntimePermissionEngine.gs
 *
 * PURPOSE:
 *
 * Central runtime authorization engine.
 *
 * ARCHITECTURE:
 *
 * CURRENT SESSION
 *        ↓
 * requireCurrentUser()
 *        ↓
 * USERS
 *        ↓
 * Role Code
 *        ↓
 * ROLE_PERMISSIONS
 *        ↓
 * Permission + Action
 *        ↓
 * ALLOW / DENY
 *
 *
 * IMPORTANT ARCHITECTURE RULE:
 *
 * EMPLOYEES DOES NOT OWN USER IDs.
 *
 * EMPLOYEES:
 * - Employee identity
 * - Job title
 * - Department
 * - Phone
 * - Email
 *
 * USERS:
 * - User ID
 * - Employee Code
 * - Login identity
 * - Role Code
 * - Password
 * - Active
 *
 ************************************************/


/************************************************
 * CONSTANTS
 ************************************************/

const MAIRIX_RUNTIME_PERMISSION = Object.freeze({

  SHEETS: {

    USERS: "USERS",

    ROLE_PERMISSIONS:
      "ROLE_PERMISSIONS"

  },


  ACTION_COLUMNS: {

    VIEW: "View",

    CREATE: "Create",

    EDIT: "Edit",

    DELETE: "Delete",

    EXPORT: "Export"

  },


  VALID_ACTIONS: [

    "VIEW",
    "CREATE",
    "EDIT",
    "DELETE",
    "EXPORT"

  ]

});


/************************************************
 * TEXT HELPER
 ************************************************/

function mairixRuntimeCleanText(value) {

  if (
    value === null ||
    value === undefined
  ) {

    return "";

  }


  return String(value)
    .trim();

}


/************************************************
 * NORMALIZE ACTION
 ************************************************/

function mairixRuntimeNormalizeAction(
  action
) {

  const normalized =
    mairixRuntimeCleanText(
      action
    ).toUpperCase();


  if (!normalized) {

    throw new Error(
      "Permission action is required."
    );

  }


  /*
   * Compatibility aliases.
   */

  const aliases = {

    READ: "VIEW",

    ADD: "CREATE",

    UPDATE: "EDIT",

    REMOVE: "DELETE",

    DOWNLOAD: "EXPORT"

  };


  const finalAction =
    aliases[normalized] ||
    normalized;


  if (
    MAIRIX_RUNTIME_PERMISSION
      .VALID_ACTIONS
      .indexOf(
        finalAction
      ) === -1
  ) {

    throw new Error(
      "Invalid permission action: " +
      action
    );

  }


  return finalAction;

}


/************************************************
 * BOOLEAN HELPER
 ************************************************/

function mairixRuntimeIsTrue(
  value
) {

  if (
    value === true ||
    value === 1
  ) {

    return true;

  }


  const normalized =
    mairixRuntimeCleanText(
      value
    ).toUpperCase();


  return (

    normalized === "TRUE" ||
    normalized === "1" ||
    normalized === "YES" ||
    normalized === "Y"

  );

}

    );

  }

}


/************************************************
 * GET USER FROM USERS SHEET
 *
 * IMPORTANT:
 *
 * This function NEVER reads EMPLOYEES
 * to find a User ID.
 *
 ************************************************/

function mairixRuntimeGetUserById(
  userId
) {

  const normalizedUserId =
    mairixRuntimeCleanText(
      userId
    );


  if (!normalizedUserId) {

    throw new Error(
      "User ID is required."
    );

  }


  const sheet =
    mairixRuntimeGetRequiredSheet(
      MAIRIX_RUNTIME_PERMISSION
        .SHEETS
        .USERS
    );


  if (
    sheet.getLastRow() < 2
  ) {

    return null;

  }


  const headers =
    mairixRuntimeGetHeaders(
      sheet
    );


  const userIdIndex =
    mairixRuntimeRequireHeader(
      headers,
      "User ID",
      "USERS"
    );


  const employeeCodeIndex =
    headers.indexOf(
      "Employee Code"
    );


  const emailIndex =
    headers.indexOf(
      "Gmail"
    );


  const userNameIndex =
    headers.indexOf(
      "Name"
    );


  /*
   * Compatibility:
   * Some schemas may still use Name.
   */

  const nameIndex =
    userNameIndex !== -1
      ? userNameIndex
      : headers.indexOf(
          "Name"
        );


  const roleCodeIndex =
    mairixRuntimeRequireHeader(
      headers,
      "Role Code",
      "USERS"
    );


  const roleNameIndex =
    headers.indexOf(
      "Role Name"
    );


  const activeIndex =
    headers.indexOf(
      "Active"
    );


  const values =
    sheet
      .getRange(
        2,
        1,
        sheet.getLastRow() - 1,
        sheet.getLastColumn()
      )
      .getDisplayValues();


  for (
    let i = 0;
    i < values.length;
    i++
  ) {

    const row =
      values[i];


    const rowUserId =
      mairixRuntimeCleanText(
        row[userIdIndex]
      );


    if (
      rowUserId !==
      normalizedUserId
    ) {

      continue;

    }


    const active =
      activeIndex === -1
        ? true
        : mairixRuntimeIsTrue(
            row[activeIndex]
          );


    return {

      userId:
        rowUserId,

      employeeCode:
        employeeCodeIndex === -1
          ? ""
          : mairixRuntimeCleanText(
              row[employeeCodeIndex]
            ),

      userName:
        nameIndex === -1
          ? ""
          : mairixRuntimeCleanText(
              row[nameIndex]
            ),

      email:
        emailIndex === -1
          ? ""
          : mairixRuntimeCleanText(
              row[emailIndex]
            ),

      roleCode:
        mairixRuntimeCleanText(
          row[roleCodeIndex]
        ).toUpperCase(),

      roleName:
        roleNameIndex === -1
          ? ""
          : mairixRuntimeCleanText(
              row[roleNameIndex]
            ),

      active:
        active

    };

  }


  return null;

}


/************************************************
 * REQUIRE ACTIVE USER
 ************************************************/

function mairixRuntimeRequireActiveUser(
  userId
) {

  const user =
    mairixRuntimeGetUserById(
      userId
    );


  if (!user) {

    throw new Error(
      "MAIRIX user not found: " +
      userId
    );

  }


  if (
    user.active !== true
  ) {

    throw new Error(
      "MAIRIX user is inactive: " +
      user.userId
    );

  }


  if (
    !user.roleCode
  ) {

    throw new Error(
      "MAIRIX user has no Role Code: " +
      user.userId
    );

  }


  return user;

}

/************************************************
 * GET ROLE PERMISSION ROW
 *
 * ROLE_PERMISSIONS structure:
 *
 * Role Code
 * Permission Code
 * View
 * Create
 * Edit
 * Delete
 * Export
 * Active
 *
 ************************************************/

function mairixRuntimeGetRolePermission(
  roleCode,
  permissionCode
) {

  const normalizedRoleCode =
    mairixRuntimeCleanText(
      roleCode
    ).toUpperCase();


  const normalizedPermissionCode =
    mairixRuntimeCleanText(
      permissionCode
    ).toUpperCase();


  if (!normalizedRoleCode) {

    throw new Error(
      "Role Code is required."
    );

  }


  if (!normalizedPermissionCode) {

    throw new Error(
      "Permission Code is required."
    );

  }


  const sheet =
    mairixRuntimeGetRequiredSheet(
      MAIRIX_RUNTIME_PERMISSION
        .SHEETS
        .ROLE_PERMISSIONS
    );


  if (
    sheet.getLastRow() < 2
  ) {

    return null;

  }


  const headers =
    mairixRuntimeGetHeaders(
      sheet
    );


  const roleCodeIndex =
    mairixRuntimeRequireHeader(
      headers,
      "Role Code",
      "ROLE_PERMISSIONS"
    );


  const permissionCodeIndex =
    mairixRuntimeRequireHeader(
      headers,
      "Permission Code",
      "ROLE_PERMISSIONS"
    );


  const viewIndex =
    mairixRuntimeRequireHeader(
      headers,
      "View",
      "ROLE_PERMISSIONS"
    );


  const createIndex =
    mairixRuntimeRequireHeader(
      headers,
      "Create",
      "ROLE_PERMISSIONS"
    );


  const editIndex =
    mairixRuntimeRequireHeader(
      headers,
      "Edit",
      "ROLE_PERMISSIONS"
    );


  const deleteIndex =
    mairixRuntimeRequireHeader(
      headers,
      "Delete",
      "ROLE_PERMISSIONS"
    );


  const exportIndex =
    mairixRuntimeRequireHeader(
      headers,
      "Export",
      "ROLE_PERMISSIONS"
    );


  const activeIndex =
    headers.indexOf(
      "Active"
    );


  const values =
    sheet
      .getRange(
        2,
        1,
        sheet.getLastRow() - 1,
        sheet.getLastColumn()
      )
      .getDisplayValues();


  for (
    let i = 0;
    i < values.length;
    i++
  ) {

    const row =
      values[i];


    const rowRoleCode =
      mairixRuntimeCleanText(
        row[roleCodeIndex]
      ).toUpperCase();


    const rowPermissionCode =
      mairixRuntimeCleanText(
        row[permissionCodeIndex]
      ).toUpperCase();


    if (
      rowRoleCode !==
      normalizedRoleCode
    ) {

      continue;

    }


    if (
      rowPermissionCode !==
      normalizedPermissionCode
    ) {

      continue;

    }


    return {

      roleCode:
        rowRoleCode,

      permissionCode:
        rowPermissionCode,

      view:
        mairixRuntimeIsTrue(
          row[viewIndex]
        ),

      create:
        mairixRuntimeIsTrue(
          row[createIndex]
        ),

      edit:
        mairixRuntimeIsTrue(
          row[editIndex]
        ),

      delete:
        mairixRuntimeIsTrue(
          row[deleteIndex]
        ),

      export:
        mairixRuntimeIsTrue(
          row[exportIndex]
        ),

      active:
        activeIndex === -1
          ? true
          : mairixRuntimeIsTrue(
              row[activeIndex]
            )

    };

  }


  return null;

}


/************************************************
 * GET ACTION VALUE
 ************************************************/

function mairixRuntimeGetPermissionActionValue(
  permission,
  action
) {

  const normalizedAction =
    mairixRuntimeNormalizeAction(
      action
    );


  switch (
    normalizedAction
  ) {

    case "VIEW":

      return (
        permission.view === true
      );


    case "CREATE":

      return (
        permission.create === true
      );


    case "EDIT":

      return (
        permission.edit === true
      );


    case "DELETE":

      return (
        permission.delete === true
      );


    case "EXPORT":

      return (
        permission.export === true
      );


    default:

      return false;

  }

}


/************************************************
 * MAIN:
 * HAS PERMISSION
 *
 * USER
 * ↓
 * USERS
 * ↓
 * ROLE
 * ↓
 * ROLE_PERMISSIONS
 * ↓
 * ACTION
 *
 ************************************************/

function mairixRuntimeHasPermission(
  userId,
  permissionCode,
  action
) {

  const user =
    mairixRuntimeRequireActiveUser(
      userId
    );


  const permission =
    mairixRuntimeGetRolePermission(
      user.roleCode,
      permissionCode
    );


  /*
   * Missing permission row
   * = DENY
   */

  if (!permission) {

    return false;

  }


  /*
   * Inactive permission
   * = DENY
   */

  if (
    permission.active !== true
  ) {

    return false;

  }


  return (
    mairixRuntimeGetPermissionActionValue(
      permission,
      action
    ) === true
  );

}


/************************************************
 * REQUIRE PERMISSION
 ************************************************/

function mairixRuntimeRequirePermission(
  userId,
  permissionCode,
  action
) {

  const allowed =
    mairixRuntimeHasPermission(
      userId,
      permissionCode,
      action
    );


  if (!allowed) {

    /*
     * Rejection logging is optional.
     * It must NEVER break authorization.
     */

    try {

      if (
        typeof mairixRuntimeRecordRejection ===
        "function"
      ) {

        mairixRuntimeRecordRejection(
          userId,
          permissionCode,
          "PERMISSION_DENIED:" +
          mairixRuntimeNormalizeAction(
            action
          )
        );

      }

    } catch (loggingError) {

      Logger.log(
        "Permission rejection logging failed: " +
        String(loggingError)
      );

    }


    throw new Error(
      "Permission denied: " +
      permissionCode +
      " / " +
      mairixRuntimeNormalizeAction(
        action
      )
    );

  }


  return true;

}

/************************************************
 * SAFE TEXT
 ************************************************/

function mairixPermissionCleanText(value) {

  if (
    value === null ||
    value === undefined
  ) {

    return "";

  }

  return String(value).trim();

}


/************************************************
 * NORMALIZE BOOLEAN
 ************************************************/

function mairixPermissionBoolean(value) {

  if (value === true) {
    return true;
  }

  if (value === 1) {
    return true;
  }

  const normalized =
    String(
      value === null ||
      value === undefined
        ? ""
        : value
    )
      .trim()
      .toUpperCase();

  return (

    normalized === "TRUE" ||
    normalized === "YES" ||
    normalized === "Y" ||
    normalized === "1"

  );

}


/************************************************
 * GET OR CREATE SHEET
 ************************************************/

function mairixGetOrCreatePermissionSheet(
  sheetName
) {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  let sheet =
    ss.getSheetByName(
      sheetName
    );

  if (!sheet) {

    sheet =
      ss.insertSheet(
        sheetName
      );

  }

  return sheet;

}


/************************************************
 * RESET SHEET STRUCTURE
 *
 * Clears existing data completely.
 *
 * IMPORTANT:
 * This is intentionally destructive because
 * the project currently has no legacy data
 * that needs migration.
 ************************************************/

function mairixResetPermissionSheet(
  sheetName,
  headers
) {

  const sheet =
    mairixGetOrCreatePermissionSheet(
      sheetName
    );

  sheet.clearContents();
  sheet.clearFormats();

  sheet
    .getRange(
      1,
      1,
      1,
      headers.length
    )
    .setValues([
      headers
    ]);

  sheet.setFrozenRows(1);

  sheet
    .getRange(
      1,
      1,
      1,
      headers.length
    )
    .setFontWeight("bold");

  return sheet;

}

/************************************************
 * WRITE PERMISSIONS
 ************************************************/

function mairixWriteDefaultPermissions() {

  const sheet =
    mairixResetPermissionSheet(
      MAIRIX_PERMISSION_SCHEMA.SHEETS.PERMISSIONS,
      MAIRIX_PERMISSIONS_HEADERS
    );

  const permissions =
    mairixGetDefaultPermissions();

  if (
    permissions.length > 0
  ) {

    sheet
      .getRange(
        2,
        1,
        permissions.length,
        permissions[0].length
      )
      .setValues(
        permissions
      );

  }

  sheet.autoResizeColumns(
    1,
    MAIRIX_PERMISSIONS_HEADERS.length
  );

  return {

    success: true,

    count:
      permissions.length

  };

}

/************************************************
 * EMPTY ROLE PERMISSION MATRIX
 *
 * Creates every role against every permission.
 *
 * Default:
 *
 * VIEW   = FALSE
 * CREATE = FALSE
 * EDIT   = FALSE
 * DELETE = FALSE
 * EXPORT = FALSE
 *
 * Active = TRUE
 *
 ************************************************/

function mairixCreateEmptyRolePermissionMatrix() {

  const permissions =
    mairixGetDefaultPermissions();

  const roles =
    mairixGetDefaultRoleCodes();

  const now =
    new Date();

  const rows = [];

  roles.forEach(
    function(roleCode) {

      permissions.forEach(
        function(permission) {

          rows.push([

            roleCode,

            permission[0],

            false,

            false,

            false,

            false,

            false,

            true,

            now,

            now

          ]);

        }
      );

    }
  );

  return rows;

}


/************************************************
 * FIND MATRIX ROW
 ************************************************/

function mairixFindPermissionMatrixRow(
  rows,
  roleCode,
  permissionCode
) {

  for (
    let i = 0;
    i < rows.length;
    i++
  ) {

    if (

      rows[i][0] === roleCode &&
      rows[i][1] === permissionCode

    ) {

      return rows[i];

    }

  }

  return null;

}


/************************************************
 * SET ROLE PERMISSION
 ************************************************/

function mairixSetDefaultRolePermission(
  rows,
  roleCode,
  permissionCode,
  permissions
) {

  const row =
    mairixFindPermissionMatrixRow(
      rows,
      roleCode,
      permissionCode
    );

  if (!row) {

    return;

  }

  const config =
    permissions || {};

  row[2] =
    config.view === true;

  row[3] =
    config.create === true;

  row[4] =
    config.edit === true;

  row[5] =
    config.delete === true;

  row[6] =
    config.export === true;

}

/************************************************
 * FIX DISCOUNT MATRIX
 *
 * ROLE_PERMISSIONS must be updated after
 * the normal defaults are created.
 ************************************************/

function mairixUpdateDiscountPermissionMatrix() {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();

  const sheet =
    ss.getSheetByName(
      MAIRIX_PERMISSION_SCHEMA
        .SHEETS
        .ROLE_PERMISSIONS
    );

  if (!sheet) {

    throw new Error(
      "ROLE_PERMISSIONS sheet not found."
    );

  }

  const lastRow =
    sheet.getLastRow();

  if (
    lastRow < 2
  ) {

    return {

      success: true,

      updated: 0

    };

  }

  const values =
    sheet
      .getRange(
        1,
        1,
        lastRow,
        sheet.getLastColumn()
      )
      .getValues();

  const headers =
    values[0];

  const roleIndex =
    headers.indexOf(
      "Role Code"
    );

  const permissionIndex =
    headers.indexOf(
      "Permission Code"
    );

  const viewIndex =
    headers.indexOf(
      "View"
    );

  const createIndex =
    headers.indexOf(
      "Create"
    );

  if (

    roleIndex === -1 ||
    permissionIndex === -1 ||
    viewIndex === -1 ||
    createIndex === -1

  ) {

    throw new Error(
      "ROLE_PERMISSIONS structure is invalid."
    );

  }

  const limits =
    mairixGetDefaultRoleDiscountLimits();

  const limitMap =
    {};

  limits.forEach(
    function(row) {

      limitMap[
        row[0]
      ] =
        row[1] === true;

    }
  );

  let updated = 0;

  for (
    let i = 1;
    i < values.length;
    i++
  ) {

    const row =
      values[i];

    if (
      row[permissionIndex] !==
      "SALES_DISCOUNT"
    ) {

      continue;

    }

    const roleCode =
      row[roleIndex];

    const enabled =
      limitMap[roleCode] === true;

    row[viewIndex] =
      enabled;

    row[createIndex] =
      enabled;

    updated++;

  }

  sheet
    .getRange(
      1,
      1,
      values.length,
      values[0].length
    )
    .setValues(
      values
    );

  return {

    success: true,

    updated:
      updated

  };

}
