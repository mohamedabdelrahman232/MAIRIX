/************************************************************
 * NORMALIZE SEARCH
 ************************************************************/

function mairixUIDropdownNormalizeSearch(
  value
) {

  return String(
    value ||
    ""
  )
    .trim()
    .toLowerCase();

}


/************************************************************
 * GENERIC SERVER RESULT NORMALIZER
 ************************************************************/

function mairixUIDropdownExtractResults(
  result
) {

  if (!result) {

    return [];

  }


  if (
    result.success === true
  ) {

    if (
      Array.isArray(
        result.data
      )
    ) {

      return result.data;

    }


    if (
      result.data &&
      Array.isArray(
        result.data.results
      )
    ) {

      return result.data.results;

    }


    if (
      Array.isArray(
        result.results
      )
    ) {

      return result.results;

    }

  }


  return [];

}


/************************************************************
 * PRODUCT DROPDOWN
 *
 * Search:
 *
 * Product Code
 * Barcode
 * Product Name
 ************************************************************/

function mairixUICreateProductDropdown(
  options
) {

  options =
    options ||
    {};


  return mairixUICreateDropdown(
    {

      placeholder:
        options.placeholder ||
        "ابحث بالباركود أو اسم الصنف",


      emptyMessage:
        "الصنف غير موجود.",


      getPrimary:
        function(item) {

          return (
            item.productName ||
            item.name ||
            item.itemName ||
            ""
          );

        },


      getSecondary:
        function(item) {

          var productCode =
            item.productCode ||
            item.itemCode ||
            item.code ||
            "";

          var barcode =
            item.barcode ||
            "";

          var category =
            item.categoryName ||
            item.category ||
            "";


          var parts = [];


          if (
            productCode
          ) {

            parts.push(
              "كود: " +
              productCode
            );

          }


          if (
            barcode
          ) {

            parts.push(
              "باركود: " +
              barcode
            );

          }


          if (
            category
          ) {

            parts.push(
              category
            );

          }


          return parts.join(
            " — "
          );

        },


      getDisplayValue:
        function(item) {

          return (
            item.productName ||
            item.name ||
            item.itemName ||
            ""
          );

        },


      onSearch:
        function(
          query,
          render
        ) {

          mairixUIDropdownSearchProducts(
            query,
            render
          );

        },


      onSelect:
        function(item) {

          if (
            typeof options.onSelect ===
            "function"
          ) {

            options.onSelect(
              item
            );

          }

        }

    }
  );

}


/************************************************************
 * PRODUCT SEARCH
 ************************************************************/

function mairixUIDropdownSearchProducts(
  query,
  render
) {

  var normalized =
    mairixUIDropdownNormalizeSearch(
      query
    );


  if (
    normalized.length === 0
  ) {

    render(
      []
    );

    return;

  }


  if (
    typeof mairixUIApiSearchProducts !==
    "function"
  ) {

    render(
      []
    );

    return;

  }


  mairixUIApiSearchProducts(
    normalized,
    function(result) {

      var results =
        mairixUIDropdownExtractResults(
          result
        );


      render(
        results
      );

    }
  );

}

/************************************************************
 * SELECTED RECORD VALIDATION
 *
 * Prevents the UI from treating typed text
 * as a selected Master Data record.
 ************************************************************/

function mairixUIDropdownRequireSelection(
  dropdown,
  message
) {

  if (
    !dropdown ||
    typeof dropdown.getSelected !==
    "function"
  ) {

    throw new Error(
      message ||
      "يجب اختيار سجل من القائمة."
    );

  }


  var selected =
    dropdown.getSelected();


  if (
    !selected
  ) {

    throw new Error(
      message ||
      "يجب اختيار سجل من القائمة."
    );

  }


  return selected;

}
