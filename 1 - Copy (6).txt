/************************************************
 * MAIRIX ERP
 * 12-MAIRIX_Purchases.gs
 *
 * PURCHASE ENGINE
 *
 * RESPONSIBILITIES:
 *
 * 1. Create purchase transactions
 * 2. Validate supplier
 * 3. Validate warehouse
 * 4. Validate items
 * 5. Update inventory through centralized movement
 * 6. Calculate purchase taxes through Tax Engine
 * 7. Create supplier payable / cash effect
 * 8. Create centralized financial reference
 * 9. Support cash / credit purchases
 * 10. Keep purchase document immutable after posting
 *
 * DESIGN PRINCIPLES:
 *
 * - White-sheet architecture
 * - No dependency on legacy data
 * - Central transaction reference
 * - Central inventory movement
 * - Central financial movement
 * - Central tax calculation
 * - Unlimited tax types
 * - Warehouse-aware
 * - Supplier balance separated from cash
 * - Every reference is traceable
 *
 * IMPORTANT:
 *
 * This file does NOT directly manipulate balances.
 *
 * Inventory:
 *   -> MAIRIX_InventoryEngine
 *
 * Taxes:
 *   -> MAIRIX_TaxEngine
 *
 * Financial:
 *   -> MAIRIX_FinancialEngine
 *
 * Master data:
 *   -> MAIRIX_MasterData
 *
 ************************************************/


/************************************************
 * CONFIGURATION
 ************************************************/

const MAIRIX_PURCHASES = {

  SHEET:
    "PURCHASES",

  LINES_SHEET:
    "PURCHASE_LINES",

  DOCUMENT_PREFIX:
    "PUR",

  STATUS_POSTED:
    "POSTED",

  STATUS_CANCELLED:
    "CANCELLED",

  PAYMENT_CASH:
    "CASH",

  PAYMENT_CREDIT:
    "CREDIT"

};


/************************************************
 * SETUP
 *
 * Creates the purchase sheets on a blank workbook.
 ************************************************/

function setupMAIRIXPurchases() {

  const ss =
    SpreadsheetApp.getActiveSpreadsheet();


  let headerSheet =
    ss.getSheetByName(
      MAIRIX_PURCHASES.SHEET
    );


  let linesSheet =
    ss.getSheetByName(
      MAIRIX_PURCHASES.LINES_SHEET
    );


  if (!headerSheet) {

    headerSheet =
      ss.insertSheet(
        MAIRIX_PURCHASES.SHEET
      );

  }


  if (!linesSheet) {

    linesSheet =
      ss.insertSheet(
        MAIRIX_PURCHASES.LINES_SHEET
      );

  }


  const headers = [

    "Purchase ID",
    "Reference No",
    "Date",
    "Supplier Code",
    "Supplier Name",
    "Warehouse Code",
    "Warehouse Name",
    "Payment Type",
    "Due Date",
    "Currency",

    "Subtotal",
    "Discount",
    "Taxable Amount",
    "Tax Total",
    "Net Total",

    "Cash Effect",
    "Payable Effect",

    "Tax Reference",
    "Inventory Reference",
    "Financial Reference",

    "User ID",
    "User Name",

    "Status",
    "Created At"

  ];


  const lineHeaders = [

    "Purchase ID",
    "Line No",

    "Item Code",
    "Item Name",

    "Quantity",
    "Unit",

    "Unit Cost",
    "Gross Amount",
    "Discount",

    "Taxable Amount",
    "Tax Total",
    "Net Amount",

    "Inventory Effect",

    "Tax Reference",

    "Warehouse Code",

    "Created At"

  ];


  if (
    headerSheet.getLastRow() === 0
  ) {

    headerSheet
      .getRange(
        1,
        1,
        1,
        headers.length
      )
      .setValues([
        headers
      ]);

  } else {

    headerSheet
      .getRange(
        1,
        1,
        1,
        headers.length
      )
      .setValues([
        headers
      ]);

  }


  if (
    linesSheet.getLastRow() === 0
  ) {

    linesSheet
      .getRange(
        1,
        1,
        1,
        lineHeaders.length
      )
      .setValues([
        lineHeaders
      ]);

  } else {

    linesSheet
      .getRange(
        1,
        1,
        1,
        lineHeaders.length
      )
      .setValues([
        lineHeaders
      ]);

  }


  headerSheet.setFrozenRows(1);
  linesSheet.setFrozenRows(1);


  return {

    success: true,

    purchaseSheet:
      MAIRIX_PURCHASES.SHEET,

    linesSheet:
      MAIRIX_PURCHASES.LINES_SHEET

  };

}


/************************************************
 * CREATE PURCHASE
 *
 * DATA STRUCTURE:
 *
 * {
 *   referenceNo,
 *   date,
 *   supplierCode,
 *   warehouseCode,
 *   paymentType,
 *   dueDate,
 *   currency,
 *   discount,
 *   lines: [
 *     {
 *       itemCode,
 *       quantity,
 *       unitCost,
 *       discount
 *     }
 *   ]
 * }
 ************************************************/

function createPurchase(data) {

  if (!data) {

    throw new Error(
      "Purchase data is required."
    );

  }


  /**********************************************
   * CURRENT USER
   **********************************************/

  const user =
    mairixPurchaseRequireUser();


  /**********************************************
   * PERMISSION
   **********************************************/

  mairixPurchaseRequirePermission(
    user.userId,
    "PURCHASES",
    "Create"
  );


  /**********************************************
   * VALIDATE INPUT
   **********************************************/

  if (
    !Array.isArray(data.lines) ||
    data.lines.length === 0
  ) {

    throw new Error(
      "Purchase must contain at least one item."
    );

  }


  const supplierCode =
    mairixPurchaseRequired(
      data.supplierCode,
      "Supplier Code"
    );


  const warehouseCode =
    mairixPurchaseRequired(
      data.warehouseCode,
      "Warehouse Code"
    );


  const paymentType =
    mairixPurchaseNormalizePaymentType(
      data.paymentType
    );


  const purchaseDate =
    data.date
      ? new Date(data.date)
      : new Date();


  if (
    isNaN(
      purchaseDate.getTime()
    )
  ) {

    throw new Error(
      "Invalid purchase date."
    );

  }


  /**********************************************
   * MASTER DATA
   **********************************************/

  const supplier =
    mairixPurchaseGetSupplier(
      supplierCode
    );


  if (!supplier) {

    throw new Error(
      "Supplier not found: " +
      supplierCode
    );

  }


  const warehouse =
    mairixPurchaseGetWarehouse(
      warehouseCode
    );


  if (!warehouse) {

    throw new Error(
      "Warehouse not found: " +
      warehouseCode
    );

  }


  /**********************************************
   * PURCHASE ID
   **********************************************/

  const purchaseId =
    mairixGeneratePurchaseId();


  /**********************************************
   * PREPARE LINES
   **********************************************/

  const preparedLines = [];

  let subtotal = 0;

  let lineNo = 1;


  data.lines.forEach(
    function(line) {

      if (!line) {
        return;
      }


      const itemCode =
        mairixPurchaseRequired(
          line.itemCode,
          "Item Code"
        );


      const quantity =
        Number(
          line.quantity
        );


      const unitCost =
        Number(
          line.unitCost
        );


      if (
        !isFinite(quantity) ||
        quantity <= 0
      ) {

        throw new Error(
          "Invalid quantity for item: " +
          itemCode
        );

      }


      if (
        !isFinite(unitCost) ||
        unitCost < 0
      ) {

        throw new Error(
          "Invalid unit cost for item: " +
          itemCode
        );

      }


      const item =
        mairixPurchaseGetItem(
          itemCode
        );


      if (!item) {

        throw new Error(
          "Item not found: " +
          itemCode
        );

      }


      const grossAmount =
        quantity *
        unitCost;


      const lineDiscount =
        Number(
          line.discount || 0
        );


      if (
        !isFinite(lineDiscount) ||
        lineDiscount < 0 ||
        lineDiscount > grossAmount
      ) {

        throw new Error(
          "Invalid discount for item: " +
          itemCode
        );

      }


      const taxableAmount =
        grossAmount -
        lineDiscount;


      preparedLines.push({

        lineNo:
          lineNo++,

        itemCode:
          itemCode,

        itemName:
          mairixPurchaseItemName(item),

        quantity:
          quantity,

        unit:
          mairixPurchaseItemUnit(item),

        unitCost:
          unitCost,

        grossAmount:
          grossAmount,

        discount:
          lineDiscount,

        taxableAmount:
          taxableAmount

      });


      subtotal +=
        taxableAmount;

    }
  );


  /**********************************************
   * DOCUMENT DISCOUNT
   **********************************************/

  const documentDiscount =
    Number(
      data.discount || 0
    );


  if (
    !isFinite(documentDiscount) ||
    documentDiscount < 0 ||
    documentDiscount > subtotal
  ) {

    throw new Error(
      "Invalid purchase discount."
    );

  }


  const taxableBeforeTax =
    subtotal -
    documentDiscount;


  /**********************************************
   * TAX ENGINE
   *
   * The tax engine determines:
   *
   * - tax types
   * - rates
   * - tax amounts
   * - tax signs
   * - final tax base
   *
   * This purchase engine never hardcodes
   * a tax type.
   **********************************************/

  const taxResult =
    mairixPurchaseCalculateTaxes({

      transactionType:
        "PURCHASE",

      transactionId:
        purchaseId,

      date:
        purchaseDate,

      taxableAmount:
        taxableBeforeTax,

      lines:
        preparedLines,

      supplierCode:
        supplierCode

    });


  const taxTotal =
    Number(
      taxResult.totalTax || 0
    );


  const netTotal =
    taxableBeforeTax +
    taxTotal;


  /**********************************************
   * FINANCIAL EFFECT
   *
   * Purchase:
   *
   * Inventory / expense:
   *   positive economic acquisition
   *
   * Cash:
   *   negative
   *
   * Supplier payable:
   *   positive liability
   *
   * The centralized Financial Engine
   * owns the actual ledger entries.
   **********************************************/

  let cashEffect = 0;
  let payableEffect = 0;


  if (
    paymentType ===
    MAIRIX_PURCHASES.PAYMENT_CASH
  ) {

    cashEffect =
      -netTotal;

  } else {

    payableEffect =
      netTotal;

  }


  /**********************************************
   * INVENTORY
   *
   * Purchase quantity:
   *
   * POSITIVE
   *
   * Inventory value:
   * calculated centrally by Inventory Engine.
   **********************************************/

  const inventoryResult =
  mairixPurchasePostInventory({

    purchaseId:
      purchaseId,

    date:
      purchaseDate,

    warehouseCode:
      warehouseCode,

    lines:
      preparedLines,

    supplierCode:
      supplierCode,

    userId:
      user.userId,

    userName:
      user.name || ""

  });


  /**********************************************
   * FINANCIAL POSTING
   **********************************************/

  const financialResult =
    mairixPurchasePostFinancial({

      purchaseId:
        purchaseId,

      date:
        purchaseDate,

      supplierCode:
        supplierCode,

      supplierName:
        mairixPurchaseSupplierName(
          supplier
        ),

      paymentType:
        paymentType,

      dueDate:
        data.dueDate
          ? new Date(data.dueDate)
          : null,

      subtotal:
        subtotal,

      discount:
        documentDiscount,

      taxableAmount:
        taxableBeforeTax,

      taxTotal:
        taxTotal,

      netTotal:
        netTotal,

      cashEffect:
        cashEffect,

      payableEffect:
        payableEffect,

      userId:
        user.userId

    });


  /**********************************************
   * SAVE PURCHASE HEADER
   **********************************************/

  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        MAIRIX_PURCHASES.SHEET
      );


  if (!sheet) {

    throw new Error(
      "PURCHASES sheet not found."
    );

  }


  const now =
    new Date();


  const row = [

    purchaseId,

    data.referenceNo ||
      purchaseId,

    purchaseDate,

    supplierCode,

    mairixPurchaseSupplierName(
      supplier
    ),

    warehouseCode,

    mairixPurchaseWarehouseName(
      warehouse
    ),

    paymentType,

    data.dueDate
      ? new Date(data.dueDate)
      : "",

    data.currency ||
      "EGP",

    subtotal,

    documentDiscount,

    taxableBeforeTax,

    taxTotal,

    netTotal,

    cashEffect,

    payableEffect,

    taxResult.reference ||
      "",

    inventoryResult.reference ||
      "",

    financialResult.reference ||
      "",

    user.userId,

    user.name ||
      "",

    MAIRIX_PURCHASES.STATUS_POSTED,

    now

  ];


  sheet
    .appendRow(row);


  /**********************************************
   * SAVE PURCHASE LINES
   **********************************************/

  mairixPurchaseSaveLines(
    purchaseId,
    warehouseCode,
    preparedLines,
    taxResult,
    now
  );


  /**********************************************
   * CENTRAL TRANSACTION REFERENCE
   **********************************************/

  mairixPurchaseRegisterReference({

    reference:
      purchaseId,

    type:
      "PURCHASE",

    date:
      purchaseDate,

    warehouseCode:
      warehouseCode,

    supplierCode:
      supplierCode,

    financialReference:
      financialResult.reference ||
      "",

    inventoryReference:
      inventoryResult.reference ||
      "",

    taxReference:
      taxResult.reference ||
      "",

    userId:
      user.userId

  });


  /**********************************************
   * RETURN
   **********************************************/

  return {

    success:
      true,

    purchaseId:
      purchaseId,

    referenceNo:
      data.referenceNo ||
      purchaseId,

    supplierCode:
      supplierCode,

    warehouseCode:
      warehouseCode,

    paymentType:
      paymentType,

    subtotal:
      subtotal,

    discount:
      documentDiscount,

    taxableAmount:
      taxableBeforeTax,

    taxTotal:
      taxTotal,

    netTotal:
      netTotal,

    cashEffect:
      cashEffect,

    payableEffect:
      payableEffect,

    taxReference:
      taxResult.reference ||
      "",

    inventoryReference:
      inventoryResult.reference ||
      "",

    financialReference:
      financialResult.reference ||
      "",

    status:
      MAIRIX_PURCHASES.STATUS_POSTED

  };

}


/************************************************
 * SAVE PURCHASE LINES
 ************************************************/

function mairixPurchaseSaveLines(
  purchaseId,
  warehouseCode,
  lines,
  taxResult,
  createdAt
) {

  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        MAIRIX_PURCHASES.LINES_SHEET
      );


  if (!sheet) {

    throw new Error(
      "PURCHASE_LINES sheet not found."
    );

  }


  const rows =
    lines.map(
      function(line) {

        const lineTax =
          mairixPurchaseFindLineTax(
            taxResult,
            line.lineNo
          );


        const taxTotal =
          lineTax.totalTax || 0;


        const netAmount =
          line.taxableAmount +
          taxTotal;


        return [

          purchaseId,

          line.lineNo,

          line.itemCode,

          line.itemName,

          line.quantity,

          line.unit,

          line.unitCost,

          line.grossAmount,

          line.discount,

          line.taxableAmount,

          taxTotal,

          netAmount,

          line.quantity,

          lineTax.reference ||
            "",

          warehouseCode,

          createdAt

        ];

      }
    );


  if (!rows.length) {
    return;
  }


  sheet
    .getRange(
      sheet.getLastRow() + 1,
      1,
      rows.length,
      rows[0].length
    )
    .setValues(rows);

}


/************************************************
 * FIND LINE TAX
 ************************************************/

function mairixPurchaseFindLineTax(
  taxResult,
  lineNo
) {

  if (
    !taxResult ||
    !Array.isArray(
      taxResult.lines
    )
  ) {

    return {
      totalTax: 0,
      reference: ""
    };

  }


  for (
    let i = 0;
    i < taxResult.lines.length;
    i++
  ) {

    if (
      Number(
        taxResult.lines[i].lineNo
      ) === Number(lineNo)
    ) {

      return {
        totalTax:
          Number(
            taxResult.lines[i].totalTax || 0
          ),

        reference:
          taxResult.lines[i].reference ||
          ""
      };

    }

  }


  return {
    totalTax: 0,
    reference: ""
  };

}


/************************************************
 * CANCEL PURCHASE
 *
 * Purchase cancellation is not a direct delete.
 *
 * A reversal transaction must be created.
 ************************************************/

function cancelPurchase(
  purchaseId,
  reason
) {

  const user =
    mairixPurchaseRequireUser();


  mairixPurchaseRequirePermission(
    user.userId,
    "PURCHASES",
    "Edit"
  );


  const purchase =
    mairixPurchaseGetPurchase(
      purchaseId
    );


  if (!purchase) {

    throw new Error(
      "Purchase not found: " +
      purchaseId
    );

  }


  if (
    purchase.status ===
    MAIRIX_PURCHASES.STATUS_CANCELLED
  ) {

    throw new Error(
      "Purchase is already cancelled."
    );

  }


  /*
   * The actual reversal is delegated
   * to the centralized engines.
   */

  const inventoryReversal =
    mairixPurchaseReverseInventory(
      purchase
    );


  const financialReversal =
    mairixPurchaseReverseFinancial(
      purchase
    );


  const taxReversal =
    mairixPurchaseReverseTax(
      purchase
    );


  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        MAIRIX_PURCHASES.SHEET
      );


  const purchaseColumn =
    mairixgetColumnNumber(
      sheet,
      "Purchase ID"
    );


  const statusColumn =
    mairixgetColumnNumber(
      sheet,
      "Status"
    );


  const lastRow =
    sheet.getLastRow();


  for (
    let r = 2;
    r <= lastRow;
    r++
  ) {

    const id =
      mairixCleanText(
        sheet
          .getRange(
            r,
            purchaseColumn
          )
          .getDisplayValue()
      );


    if (
      id !== purchaseId
    ) {

      continue;

    }


    sheet
      .getRange(
        r,
        statusColumn
      )
      .setValue(
        MAIRIX_PURCHASES.STATUS_CANCELLED
      );


    break;

  }


  mairixPurchaseRegisterReference({

    reference:
      purchaseId,

    type:
      "PURCHASE_CANCELLATION",

    date:
      new Date(),

    warehouseCode:
      purchase.warehouseCode,

    supplierCode:
      purchase.supplierCode,

    financialReference:
      financialReversal.reference ||
      "",

    inventoryReference:
      inventoryReversal.reference ||
      "",

    taxReference:
      taxReversal.reference ||
      "",

    userId:
      user.userId,

    reason:
      reason || ""

  });


  return {

    success:
      true,

    purchaseId:
      purchaseId,

    status:
      MAIRIX_PURCHASES.STATUS_CANCELLED,

    inventoryReference:
      inventoryReversal.reference ||
      "",

    financialReference:
      financialReversal.reference ||
      "",

    taxReference:
      taxReversal.reference ||
      ""

  };

}


/************************************************
 * GET PURCHASE
 ************************************************/

function mairixPurchaseGetPurchase(
  purchaseId
) {

  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        MAIRIX_PURCHASES.SHEET
      );


  if (!sheet) {
    return null;
  }


  const lastRow =
    sheet.getLastRow();


  if (lastRow < 2) {
    return null;
  }


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        headers.length
      )
      .getValues();


  for (
    let i = 0;
    i < rows.length;
    i++
  ) {

    const row =
      rows[i];


    if (
      mairixCleanText(
        row[
          headers.indexOf(
            "Purchase ID"
          )
        ]
      ) !==
      mairixCleanText(purchaseId)
    ) {

      continue;

    }


    const result = {};


    headers.forEach(
      function(header, index) {

        result[
          mairixCleanText(header)
        ] =
          row[index];

      }
    );


    /*
     * Normalize common fields.
     */

    result.purchaseId =
      mairixCleanText(
        result["Purchase ID"]
      );

    result.supplierCode =
      mairixCleanText(
        result["Supplier Code"]
      );

    result.warehouseCode =
      mairixCleanText(
        result["Warehouse Code"]
      );

    result.status =
      mairixCleanText(
        result["Status"]
      );


    return result;

  }


  return null;

}


/************************************************
 * PURCHASE ID
 ************************************************/

function mairixGeneratePurchaseId() {

  const dateKey =
    Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      "yyyyMMdd"
    );


  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        MAIRIX_PURCHASES.SHEET
      );


  let counter = 1;


  if (
    sheet &&
    sheet.getLastRow() >= 2
  ) {

    const idColumn =
      mairixgetColumnNumber(
        sheet,
        "Purchase ID"
      );


    const values =
      sheet
        .getRange(
          2,
          idColumn,
          sheet.getLastRow() - 1,
          1
        )
        .getDisplayValues();


    const prefix =
      MAIRIX_PURCHASES.DOCUMENT_PREFIX +
      "-" +
      dateKey +
      "-";


    values.forEach(
      function(row) {

        const value =
          mairixCleanText(
            row[0]
          );


        if (
          value.indexOf(prefix) !== 0
        ) {

          return;

        }


        const number =
          parseInt(
            value.substring(
              prefix.length
            ),
            10
          );


        if (
          !isNaN(number) &&
          number >= counter
        ) {

          counter =
            number + 1;

        }

      }
    );

  }


  return (
    MAIRIX_PURCHASES.DOCUMENT_PREFIX +
    "-" +
    dateKey +
    "-" +
    String(counter)
      .padStart(5, "0")
  );

}


/************************************************
 * NORMALIZE PAYMENT TYPE
 ************************************************/

function mairixPurchaseNormalizePaymentType(
  value
) {

  const type =
    mairixCleanText(
      value
    ).toUpperCase();


  if (
    type ===
    MAIRIX_PURCHASES.PAYMENT_CASH
  ) {

    return type;

  }


  if (
    type ===
    MAIRIX_PURCHASES.PAYMENT_CREDIT
  ) {

    return type;

  }


  throw new Error(
    "Payment Type must be CASH or CREDIT."
  );

}


/************************************************
 * REQUIRED VALUE
 ************************************************/

function mairixPurchaseRequired(
  value,
  label
) {

  const normalized =
    mairixCleanText(value);


  if (!normalized) {

    throw new Error(
      label +
      " is required."
    );

  }


  return normalized;

}


/************************************************
 * CURRENT USER
 ************************************************/

function mairixPurchaseRequireUser() {

  if (
    typeof requireCurrentUser ===
    "function"
  ) {

    return requireCurrentUser();

  }


  if (
    typeof getCurrentUser ===
    "function"
  ) {

    const user =
      getCurrentUser();


    if (!user) {

      throw new Error(
        "Current user not found."
      );

    }


    return user;

  }


  throw new Error(
    "MAIRIX authentication engine is not available."
  );

}

/************************************************
 * GENERIC MASTER LOOKUP
 ************************************************/

function mairixPurchaseLookupMaster(
  sheetName,
  keyColumn,
  keyValue
) {

  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        sheetName
      );


  if (!sheet) {

    throw new Error(
      "Master data sheet not found: " +
      sheetName
    );

  }


  const data =
    getSheetDataAsObjects(
      sheet
    );


  for (
    let i = 0;
    i < data.length;
    i++
  ) {

    if (
      mairixCleanText(
        data[i][keyColumn]
      ) ===
      mairixCleanText(keyValue)
    ) {

      return data[i];

    }

  }


  return null;

}

/************************************************
 * INVENTORY ENGINE BRIDGE
 *
 * FINAL MAIRIX INVENTORY INTEGRATION
 *
 * PURCHASE
 *   -> INVENTORY_LEDGER
 *   -> Positive Quantity
 *   -> WAC Update
 *   -> INVENTORY_BALANCES Rebuild
 ************************************************/

function mairixPurchasePostInventory(
  data
) {

  if (!data) {

    throw new Error(
      "Purchase inventory data is required."
    );

  }


  if (
    typeof recordMAIRIXInventoryMovement !==
    "function"
  ) {

    throw new Error(
      "MAIRIX Inventory Engine is not available. " +
      "Required function: recordMAIRIXInventoryMovement"
    );

  }


  const purchaseId =
    mairixPurchaseRequired(
      data.purchaseId,
      "Purchase ID"
    );


  const warehouseCode =
    mairixPurchaseRequired(
      data.warehouseCode,
      "Warehouse Code"
    );


  if (
    !Array.isArray(data.lines) ||
    data.lines.length === 0
  ) {

    throw new Error(
      "Purchase inventory requires at least one line."
    );

  }


  let warehouseName =
    "";


  try {

    const warehouse =
      mairixPurchaseGetWarehouse(
        warehouseCode
      );


    warehouseName =
      mairixPurchaseWarehouseName(
        warehouse
      );

  } catch (error) {

    warehouseName =
      "";

  }


  const movements = [];


  data.lines.forEach(
    function(line, index) {

      if (!line) {

        return;

      }


      const itemCode =
        mairixPurchaseRequired(
          line.itemCode,
          "Item Code"
        );


      const quantity =
        Number(
          line.quantity
        );


      const unitCost =
        Number(
          line.unitCost
        );


      if (
        !isFinite(quantity) ||
        quantity <= 0
      ) {

        throw new Error(
          "Invalid purchase quantity for item: " +
          itemCode
        );

      }


      if (
        !isFinite(unitCost) ||
        unitCost < 0
      ) {

        throw new Error(
          "Invalid purchase unit cost for item: " +
          itemCode
        );

      }


      const lineId =
        mairixCleanText(
          line.lineId ||
          line.lineNo ||
          (index + 1)
        );


      /*
       * Prevent duplicate inventory posting.
       *
       * The Inventory Engine itself also protects
       * against duplicates by Reference ID + Line ID.
       */

      const result =
        recordMAIRIXInventoryMovement({

          referenceId:
            purchaseId,

          lineId:
            lineId,

          movementType:
            "PURCHASE",

          itemCode:
            itemCode,

          itemName:
            mairixCleanText(
              line.itemName
            ),

          warehouseCode:
            warehouseCode,

          warehouseName:
            warehouseName,

          quantity:
            quantity,

          unitCost:
            unitCost,

          transactionDate:
            data.date ||
            new Date(),

          userId:
            mairixCleanText(
              data.userId ||
              ""
            ),

          userName:
            mairixCleanText(
              data.userName ||
              ""
            )

        });


      movements.push(
        result
      );

    }
  );


  if (!movements.length) {

    throw new Error(
      "No inventory movements were created for purchase: " +
      purchaseId
    );

  }


  /*
   * Rebuild derived warehouse balances.
   *
   * INVENTORY_LEDGER remains the central source
   * of truth.
   */

  let balancesResult = null;


  if (
    typeof rebuildMAIRIXInventoryBalances ===
    "function"
  ) {

    balancesResult =
      rebuildMAIRIXInventoryBalances();

  }


  return {

    success:
      true,

    reference:
      purchaseId,

    purchaseId:
      purchaseId,

    warehouseCode:
      warehouseCode,

    movementCount:
      movements.length,

    movements:
      movements,

    balances:
      balancesResult

  };

}

/************************************************
 * INVENTORY REVERSAL BRIDGE
 *
 * PURCHASE CANCELLATION
 *
 * Original Purchase:
 *   PURCHASE
 *   Quantity Effect = POSITIVE
 *
 * Cancellation:
 *   PURCHASE_RETURN
 *   Quantity Effect = NEGATIVE
 ************************************************/

function mairixPurchaseReverseInventory(
  purchase
) {

  if (!purchase) {

    throw new Error(
      "Purchase is required for inventory reversal."
    );

  }


  if (
    typeof recordMAIRIXInventoryMovement !==
    "function"
  ) {

    throw new Error(
      "MAIRIX Inventory Engine is not available. " +
      "Required function: recordMAIRIXInventoryMovement"
    );

  }


  const purchaseId =
    mairixPurchaseRequired(
      purchase.purchaseId,
      "Purchase ID"
    );


  const warehouseCode =
    mairixPurchaseRequired(
      purchase.warehouseCode,
      "Warehouse Code"
    );


  const linesSheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        MAIRIX_PURCHASES.LINES_SHEET
      );


  if (!linesSheet) {

    throw new Error(
      "PURCHASE_LINES sheet not found."
    );

  }


  const lastRow =
    linesSheet.getLastRow();


  if (lastRow < 2) {

    throw new Error(
      "No purchase lines found for: " +
      purchaseId
    );

  }


  const headers =
    linesSheet
      .getRange(
        1,
        1,
        1,
        linesSheet.getLastColumn()
      )
      .getDisplayValues()[0]
      .map(
        function(header) {

          return String(header).trim();

        }
      );


  const rows =
    linesSheet
      .getRange(
        2,
        1,
        lastRow - 1,
        headers.length
      )
      .getValues();


  const purchaseIdIndex =
    headers.indexOf(
      "Purchase ID"
    );


  const lineNoIndex =
    headers.indexOf(
      "Line No"
    );


  const itemCodeIndex =
    headers.indexOf(
      "Item Code"
    );


  const itemNameIndex =
    headers.indexOf(
      "Item Name"
    );


  const quantityIndex =
    headers.indexOf(
      "Quantity"
    );


  const unitCostIndex =
    headers.indexOf(
      "Unit Cost"
    );


  const warehouseIndex =
    headers.indexOf(
      "Warehouse Code"
    );


  if (
    purchaseIdIndex === -1 ||
    lineNoIndex === -1 ||
    itemCodeIndex === -1 ||
    quantityIndex === -1 ||
    unitCostIndex === -1
  ) {

    throw new Error(
      "PURCHASE_LINES headers are incomplete."
    );

  }


  const reversalReference =
    "PURCHASE-CANCEL-" +
    purchaseId;


  /*
   * Prevent duplicate cancellation posting.
   */

  const movements = [];


  rows.forEach(
    function(row) {

      const rowPurchaseId =
        mairixCleanText(
          row[purchaseIdIndex]
        );


      if (
        rowPurchaseId !==
        purchaseId
      ) {

        return;

      }


      const itemCode =
        mairixCleanText(
          row[itemCodeIndex]
        );


      const quantity =
        Number(
          row[quantityIndex]
        );


      const unitCost =
        Number(
          row[unitCostIndex]
        );


      if (
        !itemCode ||
        !isFinite(quantity) ||
        quantity <= 0
      ) {

        return;

      }


      const lineId =
        mairixCleanText(
          row[lineNoIndex]
        );


      const lineWarehouse =
        warehouseIndex === -1
          ? warehouseCode
          : (
              mairixCleanText(
                row[warehouseIndex]
              ) ||
              warehouseCode
            );


      /*
       * PURCHASE_RETURN is the exact inventory
       * reversal of PURCHASE.
       */

      const result =
        recordMAIRIXInventoryMovement({

          referenceId:
            reversalReference,

          lineId:
            lineId,

          movementType:
            "PURCHASE_RETURN",

          itemCode:
            itemCode,

          itemName:
            itemNameIndex === -1
              ? ""
              : mairixCleanText(
                  row[itemNameIndex]
                ),

          warehouseCode:
            lineWarehouse,

          warehouseName:
            getMAIRIXWarehouseName(
              lineWarehouse
            ),

          quantity:
            quantity,

          /*
           * For PURCHASE_RETURN the Inventory
           * Engine uses current WAC for the issue.
           */
          unitCost:
            unitCost,

          transactionDate:
            new Date(),

          userId:
            "",

          userName:
            "SYSTEM"

        });


      movements.push(
        result
      );

    }
  );


  if (!movements.length) {

    throw new Error(
      "No inventory reversal movements were created for purchase: " +
      purchaseId
    );

  }


  let balancesResult = null;


  if (
    typeof rebuildMAIRIXInventoryBalances ===
    "function"
  ) {

    balancesResult =
      rebuildMAIRIXInventoryBalances();

  }


  return {

    success:
      true,

    reference:
      reversalReference,

    purchaseId:
      purchaseId,

    movementCount:
      movements.length,

    movements:
      movements,

    balances:
      balancesResult

  };

}
