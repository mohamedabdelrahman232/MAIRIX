/************************************************************
 * COMPONENT FACTORY
 *
 * Central DOM creation helper.
 ************************************************************/

function mairixUIComponentCreate(
  tagName,
  className
) {

  var element =
    document.createElement(
      tagName
    );


  if (
    className
  ) {

    element.className =
      className;

  }


  return element;

}


/************************************************************
 * SAFE TEXT
 *
 * Database values are inserted as text.
 *
 * Never use innerHTML for Master Data.
 ************************************************************/

function mairixUISetText(
  element,
  value
) {

  if (
    !element
  ) {

    return;

  }


  element.textContent =
    value === null ||
    value === undefined
      ? ""
      : String(
          value
        );

}


/************************************************************
 * CLEAR ELEMENT
 ************************************************************/

function mairixUIClear(
  element
) {

  if (
    !element
  ) {

    return;

  }


  while (
    element.firstChild
  ) {

    element.removeChild(
      element.firstChild
    );

  }

}


/************************************************************
 * CREATE STANDARD INPUT
 ************************************************************/

function mairixUICreateInput(
  options
) {

  options =
    options ||
    {};


  var input =
    mairixUIComponentCreate(
      "input",
      "mairix-input"
    );


  input.type =
    options.type ||
    "text";


  if (
    options.id
  ) {

    input.id =
      options.id;

  }


  if (
    options.name
  ) {

    input.name =
      options.name;

  }


  if (
    options.placeholder
  ) {

    input.placeholder =
      options.placeholder;

  }


  if (
    options.autocomplete
  ) {

    input.autocomplete =
      options.autocomplete;

  }


  if (
    options.value !== undefined
  ) {

    input.value =
      options.value;

  }


  if (
    options.readonly === true
  ) {

    input.readOnly =
      true;

    input.classList.add(
      "mairix-readonly"
    );

  }


  if (
    options.disabled === true
  ) {

    input.disabled =
      true;

  }


  if (
    options.required === true
  ) {

    input.required =
      true;

  }


  return input;

}

  data,

    mairixUIComponentCreate(

    );

/************************************************************
 * QUANTITY STEPPER
 *
 * UI state only.
 *
 * Final quantity validation belongs to the
 * transaction engine.
 ************************************************************/

function mairixUICreateQuantityStepper(
  options
) {

  options =
    options ||
    {};


  var min =
    Number.isFinite(
      Number(
        options.min
      )
    )
      ? Number(
          options.min
        )
      : 1;


  var max =
    Number.isFinite(
      Number(
        options.max
      )
    )
      ? Number(
          options.max
        )
      : null;


  var step =
    Number.isFinite(
      Number(
        options.step
      )
    ) &&
    Number(
      options.step
    ) > 0
      ? Number(
          options.step
        )
      : 1;


  var initial =
    Number.isFinite(
      Number(
        options.value
      )
    )
      ? Number(
          options.value
        )
      : min;


  var wrapper =
    mairixUIComponentCreate(
      "div",
      "mairix-stepper"
    );


  var minus =
    mairixUICreateButton(
      {

        label:
          "−",

        small:
          true

      }
    );


  minus.classList.add(
    "mairix-stepper-button"
  );


  var valueElement =
    mairixUIComponentCreate(
      "span",
      "mairix-stepper-value"
    );


  var plus =
    mairixUICreateButton(
      {

        label:
          "+",

        small:
          true

      }
    );


  plus.classList.add(
    "mairix-stepper-button"
  );


  var value =
    initial;


  function normalize() {

    if (
      value < min
    ) {

      value =
        min;

    }


    if (
      max !== null &&
      value > max
    ) {

      value =
        max;

    }

  }


  function render() {

    normalize();


    mairixUISetText(
      valueElement,
      value
    );


    minus.disabled =
      value <= min;


    plus.disabled =
      max !== null &&
      value >= max;


    if (
      typeof options.onChange ===
      "function"
    ) {

      options.onChange(
        value
      );

    }

  }


  minus.addEventListener(
    "click",
    function() {

      value -=
        step;

      render();

    }
  );


  plus.addEventListener(
    "click",
    function() {

      value +=
        step;

      render();

    }
  );


  wrapper.appendChild(
    minus
  );

  wrapper.appendChild(
    valueElement
  );

  wrapper.appendChild(
    plus
  );


  render();


  wrapper.getValue =
    function() {

      return value;

    };


  wrapper.setValue =
    function(newValue) {

      var parsed =
        Number(
          newValue
        );


      if (
        !Number.isFinite(
          parsed
        )
      ) {

        return;

      }


      value =
        parsed;


      render();

    };


  wrapper.getMin =
    function() {

      return min;

    };


  wrapper.getMax =
    function() {

      return max;

    };


  return wrapper;

}


/************************************************************
 * GENERIC DROPDOWN FACTORY
 *
 * Important:
 *
 * Selecting an item does NOT create anything.
 *
 * The selected object is temporary UI state.
 ************************************************************/

function mairixUICreateDropdown(
  options
) {

  options =
    options ||
    {};


  var wrapper =
    mairixUIComponentCreate(
      "div",
      "mairix-dropdown"
    );


  var input =
    mairixUICreateInput(
      {

        placeholder:
          options.placeholder ||
          "اختار...",

        autocomplete:
          "off"

      }
    );


  var menu =
    mairixUIComponentCreate(
      "div",
      "mairix-dropdown-menu"
    );


  wrapper.appendChild(
    input
  );

  wrapper.appendChild(
    menu
  );


  var selected =
    null;


  var requestSequence =
    0;


  function close() {

    menu.classList.remove(
      "is-open"
    );

  }


  function open() {

    menu.classList.add(
      "is-open"
    );

  }


  function clearMenu() {

    mairixUIClear(
      menu
    );

  }


  function renderItems(
    results
  ) {

    clearMenu();


    if (
      !results ||
      results.length === 0
    ) {

      var empty =
        mairixUIComponentCreate(
          "div",
          "mairix-dropdown-empty"
        );


      mairixUISetText(
        empty,
        options.emptyMessage ||
        "لا توجد نتائج."
      );


      menu.appendChild(
        empty
      );


      open();

      return;

    }


    results.forEach(
      function(item) {

        var button =
          mairixUIComponentCreate(
            "button",
            "mairix-dropdown-item"
          );


        button.type =
          "button";


        var primary =
          mairixUIComponentCreate(
            "div",
            "mairix-dropdown-primary"
          );


        var secondary =
          mairixUIComponentCreate(
            "div",
            "mairix-dropdown-secondary"
          );


        var primaryValue =
          typeof options.getPrimary ===
          "function"

            ? options.getPrimary(
                item
              )

            : item;


        var secondaryValue =
          typeof options.getSecondary ===
          "function"

            ? options.getSecondary(
                item
              )

            : "";


        mairixUISetText(
          primary,
          primaryValue
        );


        mairixUISetText(
          secondary,
          secondaryValue
        );


        button.appendChild(
          primary
        );


        if (
          secondaryValue
        ) {

          button.appendChild(
            secondary
          );

        }


        button.addEventListener(
          "click",
          function() {

            selected =
              item;


            var displayValue =
              typeof options.getDisplayValue ===
              "function"

                ? options.getDisplayValue(
                    item
                  )

                : primaryValue;


            input.value =
              displayValue ||
              "";


            close();


            if (
              typeof options.onSelect ===
              "function"
            ) {

              options.onSelect(
                item
              );

            }

          }
        );


        menu.appendChild(
          button
        );

      }
    );


    open();

  }


  function search(
    query
  ) {

    if (
      typeof options.onSearch !==
      "function"
    ) {

      renderItems(
        []
      );

      return;

    }


    var sequence =
      ++requestSequence;


    options.onSearch(
      query,
      function(results) {

        /*
         * Ignore an older response if the
         * user already typed another query.
         */

        if (
          sequence !==
          requestSequence
        ) {

          return;

        }


        renderItems(
          Array.isArray(
            results
          )
            ? results
            : []
        );

      }
    );

  }


  input.addEventListener(
    "input",
    function() {

      selected =
        null;


      search(
        input.value
      );

    }
  );


  input.addEventListener(
    "focus",
    function() {

      if (
        input.value
      ) {

        search(
          input.value
        );

      }

    }
  );


  document.addEventListener(
    "click",
    function(event) {

      if (
        !wrapper.contains(
          event.target
        )
      ) {

        close();

      }

    }
  );


  wrapper.getValue =
    function() {

      return selected;

    };


  wrapper.getSelected =
    function() {

      return selected;

    };


  wrapper.getInput =
    function() {

      return input;

    };


  wrapper.clear =
    function() {

      selected =
        null;

      input.value =
        "";

      clearMenu();

      close();

    };


  wrapper.setSelected =
    function(item) {

      selected =
        item || null;


      if (
        selected
      ) {

        var displayValue =
          typeof options.getDisplayValue ===
          "function"

            ? options.getDisplayValue(
                selected
              )

            : selected.name ||
              selected.productName ||
              selected.customerName ||
              selected.vendorName ||
              selected.warehouseName ||
              "";


        input.value =
          displayValue;

      } else {

        input.value =
          "";

      }

    };


  return wrapper;

}


/************************************************************
 * STATIC SELECT
 *
 * Controlled values only.
 *
 * Example:
 * Discount:
 * 0 / 5 / 7 / 10
 *
 * This is NOT Master Data.
 ************************************************************/

function mairixUICreateStaticSelect(
  options
) {

  options =
    options ||
    {};


  var select =
    mairixUIComponentCreate(
      "select",
      "mairix-select"
    );


  if (
    options.id
  ) {

    select.id =
      options.id;

  }


  if (
    options.name
  ) {

    select.name =
      options.name;

  }


  if (
    options.disabled === true
  ) {

    select.disabled =
      true;

  }


  if (
    options.required === true
  ) {

    select.required =
      true;

  }


  if (
    options.placeholder
  ) {

    var placeholder =
      mairixUIComponentCreate(
        "option"
      );


    placeholder.value =
      "";


    placeholder.disabled =
      true;


    placeholder.selected =
      true;


    mairixUISetText(
      placeholder,
      options.placeholder
    );


    select.appendChild(
      placeholder
    );

  }


  var items =
    Array.isArray(
      options.items
    )
      ? options.items
      : [];


  items.forEach(
    function(item) {

      var option =
        mairixUIComponentCreate(
          "option"
        );


      var value =
        item &&
        typeof item ===
        "object"

          ? item.value

          : item;


      var label =
        item &&
        typeof item ===
        "object"

          ? item.label

          : item;


      option.value =
        value === undefined ||
        value === null
          ? ""
          : String(
              value
            );


      mairixUISetText(
        option,
        label
      );


      if (
        options.value !==
          undefined &&
        String(
          options.value
        ) ===
        option.value
      ) {

        option.selected =
          true;

      }


      select.appendChild(
        option
      );

    }
  );


  if (
    typeof options.onChange ===
    "function"
  ) {

    select.addEventListener(
      "change",
      function() {

        options.onChange(
          select.value
        );

      }
    );

  }


  return select;

}

/************************************************************
 * PRODUCT SEARCH BRIDGE
 *
 * Preferred contract:
 *
 * Existing UI API facade:
 * mairixUIApiSearchProducts()
 *
 * Fallback:
 * apiSearch({
 *   type: "PRODUCT"
 * })
 *
 * No Sheet access here.
 ************************************************************/

function mairixUIComponentSearchProducts(
  query,
  render,
  options
) {

  var normalized =
    String(
      query ||
      ""
    )
      .trim();


  if (
    !normalized
  ) {

    render(
      []
    );

    return;

  }


  if (
    typeof mairixUIApiSearchProducts ===
    "function"
  ) {

    mairixUIApiSearchProducts(
      normalized,
      function(result) {

        if (
          !result ||
          result.success !== true
        ) {

          render(
            []
          );

          return;

        }


        render(
          Array.isArray(
            result.data
          )
            ? result.data
            : []
        );

      }
    );

    return;

  }


  /*
   * Generic API router fallback.
   */

  if (
    typeof google !==
      "undefined" &&
    google.script &&
    google.script.run &&
    typeof google.script.run
      .apiSearch ===
      "function"
  ) {

    google
      .script
      .run
      .withSuccessHandler(
        function(result) {

          if (
            !result ||
            result.success !== true
          ) {

            render(
              []
            );

            return;

          }


          render(
            Array.isArray(
              result.data
            )
              ? result.data
              : []
          );

        }
      )
      .withFailureHandler(
        function() {

          render(
            []
          );

        }
      )
      .apiSearch(
        {

          type:
            "PRODUCT",

          query:
            normalized

        }
      );

    return;

  }


  render(
    []
  );

}

/************************************************************
 * SAFE SELECTED RECORD ACCESS
 ************************************************************/

function mairixUIGetSelectedCode(
  component,
  keys
) {

  if (
    !component ||
    typeof component.getSelected !==
    "function"
  ) {

    return "";

  }


  var item =
    component.getSelected();


  if (
    !item
  ) {

    return "";

  }


  var candidateKeys =
    Array.isArray(
      keys
    )
      ? keys
      : [];


  for (
    var i = 0;
    i < candidateKeys.length;
    i++
  ) {

    var value =
      item[
        candidateKeys[i]
      ];


    if (
      value !==
        undefined &&
      value !==
        null &&
      String(
        value
      ).trim()
    ) {

      return String(
        value
      ).trim();

    }

  }


  return "";

}

function mairixUICreateTransactionItemRow(
  options

  options =

    {};

              item;


            if (
              typeof options.onProductSelect ===
              "function"
            ) {

              options.onProductSelect(
                item,
                row
              );

            }

          }

      }
    );


  var quantity =
    mairixUICreateQuantityStepper(
      {

        min:
          options.minQuantity ||
          1,

        max:
          options.maxQuantity,

        value:
          options.quantity ||
          1,

        onChange:
          function(value) {

            if (
              typeof options.onQuantityChange ===
              "function"
            ) {

              options.onQuantityChange(
                value,
                row
              );

            }

          }

      }
    );


  row.appendChild(
    product
  );


  row.appendChild(
    quantity
  );


  row.productSelector =
    product;


  row.quantityStepper =
    quantity;


  row.getSelectedProduct =
    function() {

      return (
        typeof product.getSelected ===
        "function"
          ? product.getSelected()
          : null
      );

    };


  row.getQuantity =
    function() {

      return (
        typeof quantity.getValue ===
        "function"
          ? quantity.getValue()
          : 0
      );

    };


  row.clear =
    function() {

      if (
        typeof product.clear ===
        "function"
      ) {

        product.clear();

      }


      if (
        typeof quantity.setValue ===
        "function"
      ) {

        quantity.setValue(
          options.quantity ||
          1
        );

      }


      row._selectedProduct =
        null;

    };


  return row;

}


/************************************************************
 * TRANSACTION ROW COLLECTION
 *
 * Used by future Sales/Purchase screens.
 *
 * Minimum rows can be set by caller.
 *
 * No business calculations are performed here.
 ************************************************************/

function mairixUICreateTransactionRowCollection(
  options
) {

  options =
    options ||
    {};


  var container =
    mairixUIComponentCreate(
      "div",
      "mairix-transaction-rows"
    );


  var minimumRows =
    Number.isFinite(
      Number(
        options.minimumRows
      )
    )
      ? Number(
          options.minimumRows
        )
      : 5;


  var rows =
    [];


  function createRow() {

    var row =
      mairixUICreateTransactionItemRow(
        options
      );


    rows.push(
      row
    );


    container.appendChild(
      row
    );


    if (
      typeof options.onRowCreate ===
      "function"
    ) {

      options.onRowCreate(
        row,
        rows.length
      );

    }


    return row;

  }


  function ensureMinimumRows() {

    while (
      rows.length <
      minimumRows
    ) {

      createRow();

    }

  }


  container.addRow =
    function() {

      return createRow();

    };


  container.getRows =
    function() {

      return rows.slice();

    };


  container.getSelectedRows =
    function() {

      return rows.filter(
        function(row) {

          return !!row.getSelectedProduct();

        }
      );

    };


  container.clear =
    function() {

      rows.forEach(
        function(row) {

          row.clear();

        }
      );

    };


  ensureMinimumRows();


  return container;

}
