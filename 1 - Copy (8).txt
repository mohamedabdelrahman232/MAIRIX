/************************************************
 * MAIRIX ERP
 * 10-MAIRIX_Sales.gs
 *
 * SALES TRANSACTION ENGINE
 *
 * MAIRIX
 * -------
 * Sales is a CENTRAL transaction.
 *
 * Every sale creates ONE central reference and
 * connects:
 *
 *   SALE
 *    |
 *    +-- Customer
 *    +-- Salesman
 *    +-- Warehouse
 *    +-- Items
 *    +-- Discount
 *    +-- Loyalty Points
 *    +-- Taxes
 *    +-- Inventory Movement
 *    +-- Financial Movement
 *    +-- Cash / Credit classification
 *    +-- Financial Statement impact
 *
 *
 * IMPORTANT DESIGN
 * ----------------
 *
 * 1) Never trust UI calculations.
 * 2) Server recalculates totals.
 * 3) Server determines tax.
 * 4) Server determines inventory effect.
 * 5) Server determines financial effect.
 * 6) One Central Reference links all effects.
 * 7) Cash balance and credit balances remain separate.
 * 8) Warehouse is part of every stock movement.
 * 9) Loyalty is calculated from final sold quantity.
 * 10) Salesman commission is generated by the
 *     appropriate engine and is NOT shown here
 *     unless the engine requires it.
 *
 ************************************************/


/************************************************
 * MAIRIX SALES
 * CENTRAL SALES CONFIGURATION
 ************************************************/

const MAIRIX_SALES = {

  SHEET: "SALES",

  ITEMS_SHEET: "SALES_ITEMS",

  TRANSACTION_TYPE: "SALE",

  INVENTORY_SIGN: -1,

  FINANCIAL_SIGN: 1,

  DEFAULT_DISCOUNT: 0,

  DEFAULT_LOYALTY_POINTS_PER_UNIT: 1,

  DEFAULT_ALLOW_NEGATIVE_STOCK: false,

  DEFAULT_LOYALTY_ENABLED: true

};


/************************************************
 * PUBLIC ENTRY POINT
 ************************************************/

function createMairixSale(data) {

  const referenceContext =
    createMairixReferenceContext(
      "SALE"
    );

  return withMairixTransactionLock(
    function() {

      return _createMairixSaleLocked(
        data,
        referenceContext
      );

    },
    10000
  );

}


/************************************************
 * INTERNAL SALE CREATION
 *
 * AUTHORITATIVE SALES FLOW
 *
 * UI MAY SEND:
 * - customer
 * - salesman
 * - warehouse
 * - payment method
 * - requested invoice discount
 * - item/barcode
 * - quantity
 *
 * SERVER DETERMINES:
 * - valid discount
 * - price
 * - offer
 * - item discount
 * - invoice discount eligibility
 * - stock policy
 * - loyalty
 * - financial effect
 ************************************************/

function _createMairixSaleLocked(
  data,
  referenceContext
) {

  /**********************************************
   * 1. BASIC VALIDATION
   **********************************************/

  if (!data) {

    throw new Error(
      "Sale data is required."
    );

  }


  if (
    !Array.isArray(data.items) ||
    data.items.length === 0
  ) {

    throw new Error(
      "At least one sale item is required."
    );

  }


  /**********************************************
   * 2. CENTRAL REFERENCE
   **********************************************/

  const saleContext =
    assertMairixChildContext(
      referenceContext,
      "SALE"
    );


  const saleId =
    saleContext.referenceId;


  /**********************************************
   * 3. CURRENT USER
   **********************************************/

  const currentUser =
    requireCurrentUser();


  requireSalesCreate(
    currentUser.userId
  );


  /**********************************************
   * 4. SETTINGS SNAPSHOT
   **********************************************/

  const salesSettings =
    mairixGetSalesRuntimeSettings();


  /**********************************************
   * 5. TRANSACTION DATE
   **********************************************/

  const transactionDate =
    mairixResolveTransactionDate(
      data.transactionDate
    );


  /**********************************************
   * 6. NORMALIZE HEADER
   **********************************************/

  const customerCode =
    mairixCleanText(
      data.customerCode
    );


  const salesmanCode =
    mairixCleanText(
      data.salesmanCode
    );


  const warehouseCode =
    mairixCleanText(
      data.warehouseCode
    );


  const paymentMethod =
    mairixCleanText(
      data.paymentMethod
    ).toUpperCase();


  const requestedDiscountPercent =
    mairixNormalizePercentage(
      data.discountPercent
    );


  if (!customerCode) {

    throw new Error(
      "Customer Code is required."
    );

  }


  if (!salesmanCode) {

    throw new Error(
      "Salesman Code is required."
    );

  }


  if (!warehouseCode) {

    throw new Error(
      "Warehouse Code is required."
    );

  }


  if (
    paymentMethod !== "CASH" &&
    paymentMethod !== "CREDIT"
  ) {

    throw new Error(
      "Payment Method must be CASH or CREDIT."
    );

  }


  /**********************************************
   * 7. VALIDATE DISCOUNT AGAINST SETTINGS
   **********************************************/

  mairixValidateSalesDiscount(
    requestedDiscountPercent,
    salesSettings
  );


  /**********************************************
   * 8. MASTER DATA
   **********************************************/

  const customer =
    mairixGetCustomer(
      customerCode
    );


  if (!customer) {

    throw new Error(
      "Customer not found: " +
      customerCode
    );

  }


  const salesman =
  mairixResolveSalesmanIdentity(
    salesmanCode
  );


if (!salesman) {

  throw new Error(
    "Salesman not found: " +
    salesmanCode
  );

}


  const warehouse =
    mairixGetWarehouse(
      warehouseCode
    );


  if (!warehouse) {

    throw new Error(
      "Warehouse not found: " +
      warehouseCode
    );

  }


  /**********************************************
   * 9. WAREHOUSE ACCESS
   **********************************************/

  requireMairixWarehouseAccess(
    currentUser.userId,
    warehouseCode
  );


  /**********************************************
   * 10. OPENING LOYALTY
   **********************************************/

  const openingLoyaltyPoints =
    salesSettings.loyaltyEnabled
      ? mairixGetCustomerLoyaltyPoints(
          customerCode
        )
      : 0;


  /**********************************************
   * 11. PREPARE PRICED ITEMS
   *
   * SERVER IS AUTHORITATIVE
   **********************************************/

  const preparedItems = [];

  let grossSubtotal = 0;

  let totalQuantity = 0;

  let totalOfferSavings = 0;

  let totalItemDiscountSavings = 0;


  data.items.forEach(
    function(inputItem, index) {

      const preparedItem =
        mairixPrepareSaleItem({

          inputItem:
            inputItem,

          lineNumber:
            index + 1,

          warehouseCode:
            warehouseCode,

          transactionDate:
            transactionDate,

          requestedInvoiceDiscountPercent:
            requestedDiscountPercent,

          settings:
            salesSettings

        });


      preparedItems.push(
        preparedItem
      );


      grossSubtotal +=
        Number(
          preparedItem.grossAmount || 0
        );


      totalQuantity +=
        Number(
          preparedItem.quantity || 0
        );


      totalOfferSavings +=
        Number(
          preparedItem.offerSavings || 0
        );


      totalItemDiscountSavings +=
        Number(
          preparedItem.itemDiscountSavings || 0
        );

    }
  );


  /**********************************************
   * 12. APPLY INVOICE DISCOUNT
   *
   * IMPORTANT:
   * This function is the ONLY place where
   * invoice discount is distributed to items.
   **********************************************/

  const invoiceDiscountResult =
    mairixApplyInvoiceDiscountToPreparedItems(

      preparedItems,

      requestedDiscountPercent

    );


  grossSubtotal =
    roundMairixMoney(
      grossSubtotal
    );


  const invoiceDiscountEligibleSubtotal =
    roundMairixMoney(
      Number(
        invoiceDiscountResult.eligibleSubtotal || 0
      )
    );


  const invoiceDiscountAmount =
    roundMairixMoney(
      Number(
        invoiceDiscountResult.invoiceDiscountAmount || 0
      )
    );


  /**********************************************
   * 13. FINAL SAVINGS
   **********************************************/

  const totalSavings =
    roundMairixMoney(

      totalOfferSavings +
      totalItemDiscountSavings +
      invoiceDiscountAmount

    );


  /**********************************************
   * 14. TAXABLE SUBTOTAL
   *
   * Final sum from prepared items after:
   * - offer pricing
   * - item discounts
   * - invoice discount
   **********************************************/

  const taxableSubtotal =
    roundMairixMoney(

      preparedItems.reduce(
        function(total, item) {

          return (
            total +
            Number(
              item.netAmountBeforeTax || 0
            )
          );

        },
        0
      )

    );


  /**********************************************
   * 15. TAX ENGINE
   **********************************************/

  const taxResult =
    mairixCalculateSaleTaxes({

      transactionType:
        MAIRIX_SALES.TRANSACTION_TYPE,

      customerCode:
        customerCode,

      warehouseCode:
        warehouseCode,

      subtotal:
        taxableSubtotal,

      items:
        preparedItems,

      discountAmount:
        roundMairixMoney(

          totalItemDiscountSavings +
          invoiceDiscountAmount

        )

    });


  const totalTax =
    roundMairixMoney(
      mairixExtractTotalTax(
        taxResult
      )
    );


  /**********************************************
   * 16. FINAL TOTAL
   **********************************************/

  const netTotal =
    roundMairixMoney(
      taxableSubtotal +
      totalTax
    );


  if (
    netTotal < 0
  ) {

    throw new Error(
      "Sale total cannot be negative."
    );

  }


  /**********************************************
   * 17. FINANCIAL EFFECT
   **********************************************/

  const financialEffect =
    MAIRIX_SALES.FINANCIAL_SIGN *
    netTotal;


  const cashEffect =
    paymentMethod === "CASH"
      ? financialEffect
      : 0;


  const customerCreditEffect =
    paymentMethod === "CREDIT"
      ? financialEffect
      : 0;


  /**********************************************
   * 18. INVENTORY EFFECT
   **********************************************/

  const inventoryEffect =
    MAIRIX_SALES.INVENTORY_SIGN;


  /**********************************************
   * 19. LOYALTY
   **********************************************/

  const loyaltyPointsAdded =
    salesSettings.loyaltyEnabled
      ? roundMairixNumber(

          totalQuantity *
          salesSettings.pointsPerUnit

        )
      : 0;


  /**********************************************
   * 20. BUILD HEADER
   **********************************************/

  const saleHeader =
    _mairixBuildSaleHeader({

      saleId:
        saleId,

      date:
        transactionDate,

      customerCode:
        customerCode,

      customerName:
        mairixCleanText(
          customer.name ||
          customer["Customer Name"] ||
          ""
        ),

      salesmanCode:
        salesmanCode,

      salesmanName:
        mairixCleanText(
          salesman.name ||
          salesman["Salesman Name"] ||
          ""
        ),

      warehouseCode:
        warehouseCode,

      warehouseName:
        mairixCleanText(
          warehouse.name ||
          warehouse["Warehouse Name"] ||
          ""
        ),

      paymentMethod:
        paymentMethod,

      grossSubtotal:
        grossSubtotal,

      discountPercent:
        requestedDiscountPercent,

      discountAmount:
        roundMairixMoney(

          totalItemDiscountSavings +
          invoiceDiscountAmount

        ),

      taxableSubtotal:
        taxableSubtotal,

      totalTax:
        totalTax,

      netTotal:
        netTotal,

      totalQuantity:
        totalQuantity,

      inventoryEffect:
        inventoryEffect,

      financialEffect:
        financialEffect,

      cashEffect:
        cashEffect,

      customerCreditEffect:
        customerCreditEffect,

      loyaltyPointsEffect:
        loyaltyPointsAdded,

      userId:
        currentUser.userId,

      userName:
        currentUser.name,

      referenceNote:
        mairixCleanText(
          data.referenceNote
        )

    });


  /**********************************************
   * OPTIONAL EXTRA PRICING INFORMATION
   **********************************************/

  saleHeader["Invoice Discount Eligible Subtotal"] =
    invoiceDiscountEligibleSubtotal;


  saleHeader["Offer Savings"] =
    roundMairixMoney(
      totalOfferSavings
    );


  saleHeader["Item Discount Savings"] =
    roundMairixMoney(
      totalItemDiscountSavings
    );


  saleHeader["Invoice Discount Amount"] =
    invoiceDiscountAmount;


  saleHeader["Total Savings"] =
    totalSavings;


  /**********************************************
   * 21. WRITE SALE HEADER
   **********************************************/

  mairixWriteSaleHeader(
    saleHeader
  );


  /**********************************************
   * 22. BUILD SALES ITEMS
   *
   * HEADER-DRIVEN OBJECT ROWS
   **********************************************/

  const itemRows =
    preparedItems.map(
      function(item) {

        return _mairixBuildSaleItemRow({

          saleId:
            saleId,

          date:
            transactionDate,

          lineNumber:
            item.lineNumber,

          itemCode:
            item.itemCode,

          itemName:
            item.itemName,

          barcode:
            item.barcode,

          warehouseCode:
            warehouseCode,

          transactionType:
            MAIRIX_SALES.TRANSACTION_TYPE,

          quantity:
            item.quantity,

          originalUnitPrice:
            item.originalUnitPrice,

          offerApplied:
            item.offerApplied,

          offerPrice:
            item.offerPrice,

          offerSavings:
            item.offerSavings,

          itemDiscountPercent:
            item.itemDiscountPercent,

          itemDiscountAmount:
            item.itemDiscountAmount,

          invoiceDiscountEligible:
            item.invoiceDiscountEligible,

          invoiceDiscountPercent:
            item.invoiceDiscountPercent,

          invoiceDiscountAmount:
            item.invoiceDiscountAmount,

          totalDiscountAmount:
            item.totalDiscountAmount,

          grossAmount:
            item.grossAmount,

          netAmountBeforeTax:
            item.netAmountBeforeTax,

          finalUnitPrice:
            item.finalUnitPrice,

          totalSavings:
            item.totalSavings

        });

      }
    );


  mairixWriteSaleItems(
    itemRows
  );


  /**********************************************
   * 23. INVENTORY
   **********************************************/

  preparedItems.forEach(
    function(item) {

      mairixCreateInventoryMovement({

        referenceId:
          saleId,

        referenceType:
          "SALE",

        date:
          transactionDate,

        warehouseCode:
          warehouseCode,

        itemCode:
          item.itemCode,

        quantity:
          item.quantity,

        quantityEffect:
          -1 *
          Number(
            item.quantity || 0
          ),

        movementType:
          "SALE",

        userId:
          currentUser.userId

      });

    }
  );


  /**********************************************
   * 24. FINANCIAL
   **********************************************/

  mairixCreateFinancialMovement({

    referenceId:
      saleId,

    referenceType:
      "SALE",

    date:
      transactionDate,

    customerCode:
      customerCode,

    salesmanCode:
      salesmanCode,

    warehouseCode:
      warehouseCode,

    paymentMethod:
      paymentMethod,

    grossAmount:
      grossSubtotal,

    discountAmount:
      roundMairixMoney(

        totalItemDiscountSavings +
        invoiceDiscountAmount

      ),

    taxAmount:
      totalTax,

    netAmount:
      netTotal,

    financialEffect:
      financialEffect,

    cashEffect:
      cashEffect,

    creditEffect:
      customerCreditEffect,

    userId:
      currentUser.userId

  });


  /**********************************************
   * 25. TAX COMMIT
   **********************************************/

  mairixCommitSaleTaxes({

    referenceId:
      saleId,

    transactionType:
      "SALE",

    date:
      transactionDate,

    customerCode:
      customerCode,

    warehouseCode:
      warehouseCode,

    taxableAmount:
      taxableSubtotal,

    totalTax:
      totalTax,

    taxResult:
      taxResult,

    userId:
      currentUser.userId

  });


  /**********************************************
   * 26. LOYALTY
   **********************************************/

  if (
    salesSettings.loyaltyEnabled &&
    loyaltyPointsAdded !== 0
  ) {

    mairixApplyLoyaltyMovement({

      referenceId:
        saleId,

      customerCode:
        customerCode,

      pointsEffect:
        loyaltyPointsAdded,

      movementType:
        "SALE",

      date:
        transactionDate,

      userId:
        currentUser.userId

    });

  }


  /**********************************************
   * 27. COMMISSION
   **********************************************/

  if (
    salesSettings.commissionEnabled
  ) {

    mairixCreateSalesmanCommission({

      referenceId:
        saleId,

      salesmanCode:
        salesmanCode,

      customerCode:
        customerCode,

      netSalesAmount:
        taxableSubtotal,

      date:
        transactionDate,

      userId:
        currentUser.userId

    });

  }


  /**********************************************
   * 28. CENTRAL AUDIT
   **********************************************/

  mairixWriteSalesAudit({

    referenceId:
      saleId,

    transactionType:
      "SALE",

    customerCode:
      customerCode,

    salesmanCode:
      salesmanCode,

    warehouseCode:
      warehouseCode,

    paymentMethod:
      paymentMethod,

    netTotal:
      netTotal,

    userId:
      currentUser.userId

  });


  /**********************************************
   * 29. RETURN RESULT
   **********************************************/

  return {

    success:
      true,

    referenceId:
      saleId,

    saleId:
      saleId,

    date:
      transactionDate,


    customer: {

      code:
        customerCode,

      name:
        mairixCleanText(
          customer.name ||
          customer["Customer Name"] ||
          ""
        ),

      loyaltyPointsBefore:
        openingLoyaltyPoints,

      loyaltyPointsAdded:
        loyaltyPointsAdded,

      loyaltyPointsAfter:
        openingLoyaltyPoints +
        loyaltyPointsAdded

    },


    salesman: {

      code:
        salesmanCode,

      name:
        mairixCleanText(
          salesman.name ||
          salesman["Salesman Name"] ||
          ""
        )

    },


    warehouse: {

      code:
        warehouseCode,

      name:
        mairixCleanText(
          warehouse.name ||
          warehouse["Warehouse Name"] ||
          ""
        )

    },


    paymentMethod:
      paymentMethod,


    grossSubtotal:
      grossSubtotal,


    invoiceDiscountEligibleSubtotal:
      invoiceDiscountEligibleSubtotal,


    invoiceDiscountPercent:
      requestedDiscountPercent,


    itemDiscountAmount:
      roundMairixMoney(
        totalItemDiscountSavings
      ),


    invoiceDiscountAmount:
      invoiceDiscountAmount,


    totalDiscountAmount:
      roundMairixMoney(

        totalItemDiscountSavings +
        invoiceDiscountAmount

      ),


    offerSavings:
      roundMairixMoney(
        totalOfferSavings
      ),


    totalSavings:
      totalSavings,


    taxableSubtotal:
      taxableSubtotal,


    totalTax:
      totalTax,


    netTotal:
      netTotal,


    totalQuantity:
      totalQuantity,


    loyaltyPointsAdded:
      loyaltyPointsAdded,


    inventoryEffect:
      inventoryEffect,


    financialEffect:
      financialEffect,


    cashEffect:
      cashEffect,


    customerCreditEffect:
      customerCreditEffect,


    items:
      preparedItems,


    tax:
      taxResult,


    message:
      "Sale created successfully."

  };

}


/************************************************
 * SALE HEADER BUILDER
 ************************************************/

function _mairixBuildSaleHeader(data) {

  return {

    "Reference ID":
      data.saleId,

    "Transaction Type":
      "SALE",

    "Date":
      data.date,

    "Customer Code":
      data.customerCode,

    "Customer Name":
      data.customerName,

    "Salesman Code":
      data.salesmanCode,

    "Salesman Name":
      data.salesmanName,

    "Warehouse Code":
      data.warehouseCode,

    "Warehouse Name":
      data.warehouseName,

    "Payment Method":
      data.paymentMethod,

    "Gross Subtotal":
      data.grossSubtotal,

    "Discount %":
      data.discountPercent,

    "Discount Amount":
      data.discountAmount,

    "Taxable Subtotal":
      data.taxableSubtotal,

    "Total Tax":
      data.totalTax,

    "Net Total":
      data.netTotal,

    "Total Quantity":
      data.totalQuantity,

    /*
     * Central movement signs
     */

    "Inventory Quantity Effect":
      data.inventoryEffect *
      data.totalQuantity,

    "Financial Effect":
      data.financialEffect,

    "Cash Effect":
      data.cashEffect,

    "Credit Effect":
      data.customerCreditEffect,

    /*
     * Loyalty
     */

    "Loyalty Points Effect":
      data.loyaltyPointsEffect,

    /*
     * User
     */

    "User ID":
      data.userId,

    "User Name":
      data.userName,

    "Reference Note":
      data.referenceNote

  };

}


/************************************************
 * SALE ITEM BUILDER
 *
 * RETURNS OBJECT
 *
 * SALES_ITEMS WRITER MAPS VALUES
 * BY SHEET HEADERS
 ************************************************/

function _mairixBuildSaleItemRow(data) {

  return {

    "Reference ID":
      data.saleId,

    "Date":
      data.date,

    "Line Number":
      data.lineNumber,

    "Item Code":
      data.itemCode,

    "Item Name":
      data.itemName,

    "Barcode":
      data.barcode || "",

    "Warehouse Code":
      data.warehouseCode,

    "Transaction Type":
      data.transactionType,

    "Quantity":
      data.quantity,


    /**********************************************
     * ORIGINAL PRICING
     **********************************************/

    "Original Unit Price":
      roundMairixMoney(
        data.originalUnitPrice
      ),


    /**********************************************
     * OFFER
     **********************************************/

    "Offer Applied":
      data.offerApplied === true,

    "Offer Price":
      roundMairixMoney(
        data.offerPrice
      ),

    "Offer Savings":
      roundMairixMoney(
        data.offerSavings
      ),


    /**********************************************
     * ITEM DISCOUNT
     **********************************************/

    "Item Discount %":
      Number(
        data.itemDiscountPercent || 0
      ),

    "Item Discount Amount":
      roundMairixMoney(
        data.itemDiscountAmount
      ),


    /**********************************************
     * INVOICE DISCOUNT
     **********************************************/

    "Invoice Discount Eligible":
      data.invoiceDiscountEligible === true,

    "Invoice Discount %":
      Number(
        data.invoiceDiscountPercent || 0
      ),

    "Invoice Discount Amount":
      roundMairixMoney(
        data.invoiceDiscountAmount
      ),


    /**********************************************
     * FINAL PRICING
     **********************************************/

    "Final Unit Price":
      roundMairixMoney(
        data.finalUnitPrice
      ),

    "Gross Amount":
      roundMairixMoney(
        data.grossAmount
      ),

    "Total Discount Amount":
      roundMairixMoney(
        data.totalDiscountAmount
      ),

    "Net Amount":
      roundMairixMoney(
        data.netAmountBeforeTax
      ),

    "Total Savings":
      roundMairixMoney(
        data.totalSavings
      ),


    /**********************************************
     * CENTRAL INVENTORY SIGN
     **********************************************/

    "Inventory Quantity Effect":
      -1 *
      Number(
        data.quantity || 0
      )

  };

}

/************************************************
 * MONEY ROUNDING
 ************************************************/

function roundMairixMoney(
  value
) {

  return Math.round(
    Number(value || 0) *
    100
  ) / 100;

}


/************************************************
 * MASTER DATA ADAPTERS
 *
 * These adapters intentionally isolate the
 * Sales Engine from the physical structure
 * of MasterData.
 ************************************************/

/************************************************
 * CUSTOMER MASTER DATA ADAPTER
 ************************************************/

function mairixGetCustomer(
  customerCode
) {

  if (
    typeof getMairixCustomer ===
    "function"
  ) {

    return getMairixCustomer(
      customerCode
    );

  }


  if (
    typeof getMairixCustomerByCode ===
    "function"
  ) {

    return getMairixCustomerByCode(
      customerCode
    );

  }


  if (
    typeof getCustomerByCode ===
    "function"
  ) {

    return getCustomerByCode(
      customerCode
    );

  }


  throw new Error(
    "Customer Master Data adapter is not available."
  );

}


/************************************************
 * SALESMAN MASTER DATA ADAPTER
 *
 * AUTHORITATIVE IDENTITY FLOW:
 *
 * SALESMAN CODE
 *      SAL-0000001
 *            ↓
 * SALES_PROFILES
 *            ↓
 * EMPLOYEE CODE
 *      EMP-0000001
 *            ↓
 * EMPLOYEES
 *
 ************************************************/

function mairixGetSalesman(
  salesmanCode
) {

  salesmanCode =
    mairixCleanText(
      salesmanCode
    );


  if (!salesmanCode) {

    return null;

  }


  /**********************************************
   * 1. PRIMARY SOURCE
   *
   * SALES_PROFILES
   **********************************************/

  if (
    typeof mairixGetSalesProfileBySalesmanCode ===
    "function"
  ) {

    const profile =
      mairixGetSalesProfileBySalesmanCode(
        salesmanCode
      );


    if (profile) {

      return profile;

    }

  }


  /**********************************************
   * 2. COMPATIBILITY LOOKUP
   *
   * Existing function if available
   **********************************************/

  if (
    typeof getMairixSalesmanByCode ===
    "function"
  ) {

    const salesman =
      getMairixSalesmanByCode(
        salesmanCode
      );


    if (salesman) {

      return salesman;

    }

  }


  if (
    typeof getSalesmanByCode ===
    "function"
  ) {

    const salesman =
      getSalesmanByCode(
        salesmanCode
      );


    if (salesman) {

      return salesman;

    }

  }


  /**********************************************
   * 3. DIRECT SALES_PROFILES FALLBACK
   *
   * Prevent dependency on another adapter.
   **********************************************/

  const sheetName =
    "SALES_PROFILES";


  if (
    typeof mairixGetSheet !==
    "function"
  ) {

    throw new Error(
      "MAIRIX sheet adapter is not available."
    );

  }


  const sheet =
    mairixGetSheet(
      sheetName
    );


  if (!sheet) {

    return null;

  }


  const values =
    sheet.getDataRange()
      .getValues();


  if (
    !values ||
    values.length < 2
  ) {

    return null;

  }


  const headers =
    values[0].map(
      function(header) {

        return mairixCleanText(
          header
        );

      }
    );


  const salesmanCodeIndex =
    headers.indexOf(
      "Salesman Code"
    );


  if (
    salesmanCodeIndex === -1
  ) {

    throw new Error(
      "Required header missing: SALES_PROFILES -> Salesman Code"
    );

  }


  const employeeCodeIndex =
    headers.indexOf(
      "Employee Code"
    );


  const userIdIndex =
    headers.indexOf(
      "User ID"
    );


  const employeeNameIndex =
    headers.indexOf(
      "Employee Name"
    );


  const emailIndex =
    headers.indexOf(
      "Gmail"
    ) !== -1

      ? headers.indexOf("Gmail")

      : headers.indexOf("Email");


  const roleCodeIndex =
    headers.indexOf(
      "Role Code"
    );


  const activeIndex =
    headers.indexOf(
      "Active"
    );


  for (
    let rowIndex = 1;
    rowIndex < values.length;
    rowIndex++
  ) {

    const row =
      values[rowIndex];


    const rowSalesmanCode =
      mairixCleanText(
        row[salesmanCodeIndex]
      );


    if (
      rowSalesmanCode !==
      salesmanCode
    ) {

      continue;

    }


    const active =
      activeIndex !== -1
        ? row[activeIndex]
        : true;


    if (
      active === false ||
      String(active)
        .toUpperCase() ===
        "FALSE"
    ) {

      return null;

    }


    return {

      salesmanCode:
        salesmanCode,

      employeeCode:
        employeeCodeIndex !== -1
          ? mairixCleanText(
              row[employeeCodeIndex]
            )
          : "",

      userId:
        userIdIndex !== -1
          ? mairixCleanText(
              row[userIdIndex]
            )
          : "",

      name:
        employeeNameIndex !== -1
          ? mairixCleanText(
              row[employeeNameIndex]
            )
          : "",

      employeeName:
        employeeNameIndex !== -1
          ? mairixCleanText(
              row[employeeNameIndex]
            )
          : "",

      email:
        emailIndex !== -1
          ? mairixCleanText(
              row[emailIndex]
            )
          : "",

      roleCode:
        roleCodeIndex !== -1
          ? mairixCleanText(
              row[roleCodeIndex]
            )
          : "SALESMAN",

      active:
        true

    };

  }


  /**********************************************
   * NOT FOUND
   **********************************************/

  return null;

}

/************************************************
 * INVENTORY STOCK ADAPTER
 *
 * SALES -> INVENTORY INTEGRATION
 *
 * Single responsibility:
 * Resolve current available stock from the
 * central Inventory Engine.
 ************************************************/

function mairixGetWarehouseStock(
  warehouseCode,
  itemCode
) {

  warehouseCode =
    mairixCleanText(
      warehouseCode
    );


  itemCode =
    mairixCleanText(
      itemCode
    );


  if (!warehouseCode) {

    throw new Error(
      "Warehouse Code is required for stock lookup."
    );

  }


  if (!itemCode) {

    throw new Error(
      "Item Code is required for stock lookup."
    );

  }


  /**********************************************
   * PRIORITY 1
   *
   * Central inventory balance engine
   **********************************************/

  if (
    typeof mairixGetInventoryBalance ===
    "function"
  ) {

    const result =
      mairixGetInventoryBalance(
        warehouseCode,
        itemCode
      );


    if (
      result !== null &&
      result !== undefined
    ) {

      if (
        typeof result === "number"
      ) {

        return Number(result);

      }


      return Number(
        result.availableStock ||
        result.quantity ||
        result.balance ||
        result.stock ||
        0
      );

    }

  }


  /**********************************************
   * PRIORITY 2
   *
   * Alternative central inventory functions
   **********************************************/

  if (
    typeof getMairixInventoryBalance ===
    "function"
  ) {

    const result =
      getMairixInventoryBalance(
        warehouseCode,
        itemCode
      );


    return Number(
      result &&
      (
        result.availableStock ||
        result.quantity ||
        result.balance ||
        result.stock
      ) ||
      0
    );

  }


  if (
    typeof getInventoryBalance ===
    "function"
  ) {

    const result =
      getInventoryBalance(
        warehouseCode,
        itemCode
      );


    return Number(
      result &&
      (
        result.availableStock ||
        result.quantity ||
        result.balance ||
        result.stock
      ) ||
      0
    );

  }


  /**********************************************
   * PRIORITY 3
   *
   * INVENTORY_BALANCES SHEET
   *
   * Fallback reader only.
   * Does NOT create another inventory engine.
   **********************************************/

  const sheet =
    mairixGetSheet(
      "INVENTORY_BALANCES"
    );


  if (!sheet) {

    throw new Error(
      "Inventory Engine is not connected and INVENTORY_BALANCES sheet was not found."
    );

  }


  const values =
    sheet.getDataRange()
      .getValues();


  if (
    !values ||
    values.length < 2
  ) {

    return 0;

  }


  const headers =
    values[0].map(
      function(header) {

        return mairixCleanText(
          header
        );

      }
    );


  const warehouseIndex =
    headers.indexOf(
      "Warehouse Code"
    );


  const itemIndex =
    headers.indexOf(
      "Item Code"
    );


  let balanceIndex =
    headers.indexOf(
      "Available Stock"
    );


  if (balanceIndex === -1) {

    balanceIndex =
      headers.indexOf(
        "Quantity"
      );

  }


  if (balanceIndex === -1) {

    balanceIndex =
      headers.indexOf(
        "Balance"
      );

  }


  if (
    warehouseIndex === -1 ||
    itemIndex === -1 ||
    balanceIndex === -1
  ) {

    throw new Error(
      "INVENTORY_BALANCES headers are not compatible with stock lookup."
    );

  }


  for (
    let rowIndex = 1;
    rowIndex < values.length;
    rowIndex++
  ) {

    const row =
      values[rowIndex];


    const rowWarehouse =
      mairixCleanText(
        row[warehouseIndex]
      );


    const rowItem =
      mairixCleanText(
        row[itemIndex]
      );


    if (
      rowWarehouse === warehouseCode &&
      rowItem === itemCode
    ) {

      return Number(
        row[balanceIndex] || 0
      );

    }

  }


  return 0;

}


/************************************************
 * WAREHOUSE ACCESS
 *
 * CENTRAL COMPATIBILITY ADAPTER
 *
 * AUTHORITATIVE PURPOSE:
 * Verify that the current user has access
 * to the requested warehouse.
 *
 * COMPATIBILITY ORDER:
 *
 * 1. requireWarehousePermission()
 * 2. hasWarehousePermission()
 * 3. USER_WAREHOUSES sheet
 * 4. USER_WAREHOUSE_PERMISSIONS sheet
 *
 * SECURITY:
 * Never silently grants warehouse access.
 ************************************************/

function requireMairixWarehouseAccess(
  userId,
  warehouseCode
) {

  const normalizedUserId =
    mairixCleanText(userId);

  const normalizedWarehouseCode =
    mairixCleanText(warehouseCode);


  /**********************************************
   * BASIC VALIDATION
   **********************************************/

  if (!normalizedUserId) {

    throw new Error(
      "Warehouse access error: User ID is required."
    );

  }


  if (!normalizedWarehouseCode) {

    throw new Error(
      "Warehouse access error: Warehouse Code is required."
    );

  }


  /**********************************************
   * 1. AUTHORITATIVE ENGINE
   **********************************************/

  if (
    typeof requireWarehousePermission ===
    "function"
  ) {

    try {

      requireWarehousePermission(
        normalizedUserId,
        normalizedWarehouseCode,
        "CREATE"
      );

      return true;

    } catch (error) {

      /*
       * Do not immediately stop here.
       *
       * Different legacy permission engines may
       * use different action naming conventions
       * or implementations.
       *
       * Continue to the central sheet fallback.
       */

    }

  }


  /**********************************************
   * 2. BOOLEAN PERMISSION ENGINE
   **********************************************/

  if (
    typeof hasWarehousePermission ===
    "function"
  ) {

    try {

      const allowed =
        hasWarehousePermission(
          normalizedUserId,
          normalizedWarehouseCode
        );


      if (allowed === true) {

        return true;

      }

    } catch (error) {

      /*
       * Continue to central sheet validation.
       */

    }

  }


  /**********************************************
   * 3. USER_WAREHOUSES
   *
   * Expected relationship:
   *
   * User ID
   * Employee Code
   * Warehouse Code
   * Warehouse Name
   * Active
   **********************************************/

  const userWarehousesResult =
    mairixCheckWarehouseAccessFromSheet_(
      "USER_WAREHOUSES",
      normalizedUserId,
      normalizedWarehouseCode
    );


  if (
    userWarehousesResult.allowed
  ) {

    return true;

  }


  /**********************************************
   * 4. USER_WAREHOUSE_PERMISSIONS
   *
   * Compatibility with legacy architecture.
   **********************************************/

  const legacyWarehousesResult =
    mairixCheckWarehouseAccessFromSheet_(
      "USER_WAREHOUSE_PERMISSIONS",
      normalizedUserId,
      normalizedWarehouseCode
    );


  if (
    legacyWarehousesResult.allowed
  ) {

    return true;

  }


  /**********************************************
   * DENY
   **********************************************/

  throw new Error(

    "User does not have access to warehouse: " +
    normalizedWarehouseCode +
    " (User: " +
    normalizedUserId +
    ")"

  );

}



/************************************************
 * CHECK WAREHOUSE ACCESS FROM SHEET
 *
 * INTERNAL COMPATIBILITY HELPER
 ************************************************/

function mairixCheckWarehouseAccessFromSheet_(
  sheetName,
  userId,
  warehouseCode
) {

  const result = {

    allowed: false,

    foundSheet: false,

    foundUser: false,

    reason: ""

  };


  const spreadsheet =
    SpreadsheetApp.getActiveSpreadsheet();


  const sheet =
    spreadsheet.getSheetByName(
      sheetName
    );


  /*
   * Sheet does not exist.
   */

  if (!sheet) {

    result.reason =
      "Sheet not found: " +
      sheetName;

    return result;

  }


  result.foundSheet = true;


  const lastRow =
    sheet.getLastRow();


  const lastColumn =
    sheet.getLastColumn();


  if (
    lastRow < 2 ||
    lastColumn < 1
  ) {

    result.reason =
      "Sheet is empty: " +
      sheetName;

    return result;

  }


  const values =
    sheet.getRange(
      1,
      1,
      lastRow,
      lastColumn
    ).getValues();


  const headers =
    values[0].map(
      function(header) {

        return mairixCleanText(
          header
        ).toUpperCase();

      }
    );


  /**********************************************
   * RESOLVE HEADER INDEXES
   **********************************************/

  const userIdIndex =
    mairixFindHeaderIndex_(
      headers,
      [
        "USER ID",
        "USERID"
      ]
    );


  const warehouseCodeIndex =
    mairixFindHeaderIndex_(
      headers,
      [
        "WAREHOUSE CODE",
        "WAREHOUSECODE"
      ]
    );


  const activeIndex =
    mairixFindHeaderIndex_(
      headers,
      [
        "ACTIVE",
        "IS ACTIVE",
        "STATUS"
      ]
    );


  /*
   * Required structure is unavailable.
   */

  if (
    userIdIndex === -1 ||
    warehouseCodeIndex === -1
  ) {

    result.reason =
      "Required warehouse permission headers " +
      "not found in: " +
      sheetName;

    return result;

  }


  /**********************************************
   * SEARCH ASSIGNMENTS
   **********************************************/

  for (
    let rowIndex = 1;
    rowIndex < values.length;
    rowIndex++
  ) {

    const row =
      values[rowIndex];


    const rowUserId =
      mairixCleanText(
        row[userIdIndex]
      );


    const rowWarehouseCode =
      mairixCleanText(
        row[warehouseCodeIndex]
      );


    if (
      rowUserId !== userId ||
      rowWarehouseCode !== warehouseCode
    ) {

      continue;

    }


    result.foundUser = true;


    /********************************************
     * ACTIVE VALIDATION
     ********************************************/

    if (
      activeIndex === -1
    ) {

      /*
       * If the assignment exists and there is
       * no Active column, treat the assignment
       * itself as valid.
       */

      result.allowed = true;

      result.reason =
        "Warehouse assignment found.";

      return result;

    }


    const activeValue =
      row[activeIndex];


    if (
      mairixIsWarehouseAssignmentActive_(
        activeValue
      )
    ) {

      result.allowed = true;

      result.reason =
        "Active warehouse assignment found.";

      return result;

    }


    result.reason =
      "Warehouse assignment is inactive.";

  }


  if (!result.reason) {

    result.reason =
      "No warehouse assignment found.";

  }


  return result;

}



/************************************************
 * FIND HEADER INDEX
 ************************************************/

function mairixFindHeaderIndex_(
  headers,
  candidates
) {

  for (
    let i = 0;
    i < candidates.length;
    i++
  ) {

    const candidate =
      mairixCleanText(
        candidates[i]
      ).toUpperCase();


    const index =
      headers.indexOf(
        candidate
      );


    if (index !== -1) {

      return index;

    }

  }


  return -1;

}



/************************************************
 * ACTIVE VALUE NORMALIZER
 ************************************************/

function mairixIsWarehouseAssignmentActive_(
  value
) {

  if (
    value === true
  ) {

    return true;

  }


  const normalized =
    mairixCleanText(
      value
    ).toUpperCase();


  return (

    normalized === "TRUE" ||
    normalized === "YES" ||
    normalized === "Y" ||
    normalized === "1" ||
    normalized === "ACTIVE"

  );

}

/************************************************
 * SALE SHEET WRITER
 ************************************************/

function mairixWriteSaleHeader(
  data
) {

  const sheet =
    mairixgetSheet(
      MAIRIX_SALES.SHEET
    );


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const row =
    new Array(
      headers.length
    ).fill("");


  headers.forEach(
    function(header, index) {

      const key =
        mairixCleanText(
          header
        );


      if (
        data[key] !== undefined
      ) {

        row[index] =
          data[key];

      }

    }
  );


  sheet
    .getRange(
      sheet.getLastRow() + 1,
      1,
      1,
      row.length
    )
    .setValues([
      row
    ]);

}


/************************************************
 * SALES ITEMS WRITER
 *
 * HEADER-DRIVEN
 *
 * DOES NOT DEPEND ON COLUMN ORDER
 ************************************************/

function mairixWriteSaleItems(
  rows
) {

  if (
    !Array.isArray(rows) ||
    rows.length === 0
  ) {

    return;

  }


  const sheet =
    mairixgetSheet(
      MAIRIX_SALES.ITEMS_SHEET
    );


  const lastColumn =
    sheet.getLastColumn();


  if (
    lastColumn < 1
  ) {

    throw new Error(
      "SALES_ITEMS sheet has no headers."
    );

  }


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        lastColumn
      )
      .getDisplayValues()[0]
      .map(
        function(header) {

          return mairixCleanText(
            header
          );

        }
      );


  /**********************************************
   * SAFETY:
   * Reference ID MUST EXIST
   **********************************************/

  if (
    headers.indexOf(
      "Reference ID"
    ) === -1
  ) {

    throw new Error(
      "SALES_ITEMS sheet must contain 'Reference ID' column."
    );

  }


  const values =
    rows.map(
      function(data) {

        return headers.map(
          function(header) {

            if (
              data[header] !== undefined
            ) {

              return data[header];

            }

            return "";

          }
        );

      }
    );


  const startRow =
    sheet.getLastRow() + 1;


  sheet
    .getRange(
      startRow,
      1,
      values.length,
      headers.length
    )
    .setValues(
      values
    );

}

/************************************************
 * GET ACTIVE SALES DISCOUNT OPTIONS
 *
 * SETTINGS_OPTIONS:
 *
 * KEY
 * OPTION VALUE
 * LABEL
 * SORT ORDER
 * ACTIVE
 ************************************************/

function mairixGetActiveSalesDiscountOptions(
  optionsKey
) {

  const normalizedKey =
    mairixCleanText(
      optionsKey ||
      "SALES_DISCOUNT_PERCENT"
    ).toUpperCase();


  const sheet =
    mairixSettingsGetOptionsSheet();


  if (!sheet) {

    return [];

  }


  const lastRow =
    sheet.getLastRow();


  if (lastRow < 2) {

    return [];

  }


  const values =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        5
      )
      .getValues();


  const discounts =
    values
      .filter(
        function(row) {

          const key =
            mairixCleanText(
              row[0]
            ).toUpperCase();


          const active =
            row[4];


          return (

            key === normalizedKey &&

            (
              active === true ||

              String(
                active
              ).toUpperCase() ===
              "TRUE"

            )

          );

        }
      )
      .map(
        function(row) {

          return Number(
            row[1]
          );

        }
      )
      .filter(
        function(value) {

          return (

            isFinite(value) &&

            value >= 0 &&

            value <= 100

          );

        }
      )
      .sort(
        function(a, b) {

          return a - b;

        }
      );


  return discounts;

}

/************************************************
 * VALIDATE SALES DISCOUNT
 ************************************************/

function mairixValidateSalesDiscount(
  discountPercent,
  settings
) {

  const discount =
    Number(
      discountPercent
    );


  if (
    !isFinite(discount) ||
    discount < 0 ||
    discount > 100
  ) {

    throw new Error(
      "Invalid sales discount percentage."
    );

  }


  /**********************************************
   * DISCOUNTS DISABLED
   **********************************************/

  if (
    !settings.discountEnabled
  ) {

    if (discount !== 0) {

      throw new Error(
        "Sales discounts are currently disabled."
      );

    }

    return true;

  }


  /**********************************************
   * ZERO DISCOUNT POLICY
   **********************************************/

  if (
    discount === 0
  ) {

    if (
      !settings.discountAllowZero
    ) {

      throw new Error(
        "Zero discount is not currently allowed."
      );

    }

    return true;

  }


  /**********************************************
   * APPROVED OPTIONS POLICY
   **********************************************/

  if (
    settings.discountRequireApprovedOption
  ) {

    if (

      settings.activeDiscounts.indexOf(
        discount
      ) === -1

    ) {

      throw new Error(

        "Discount percentage is not approved: " +
        discount

      );

    }

  }


  return true;

}

function mairixGetAuthoritativeSalePrice(
  masterItem,
  inputItem
) {

  const possiblePrices = [

    masterItem.salePrice,

    masterItem["Sale Price"],

    masterItem.price,

    masterItem["Price"],

    masterItem.sellingPrice,

    masterItem["Selling Price"]

  ];


  for (
    let i = 0;
    i < possiblePrices.length;
    i++
  ) {

    const price =
      Number(
        possiblePrices[i]
      );


    if (
      isFinite(price) &&
      price >= 0
    ) {

      return roundMairixMoney(
        price
      );

    }

  }


  throw new Error(

    "No authoritative sale price found for item: " +
    (
      masterItem.code ||
      masterItem["Item Code"] ||
      inputItem.itemCode
    )

  );

}

function mairixGetItemDiscountPercent(
  masterItem,
  inputItem
) {

  const possibleValues = [

    masterItem.itemDiscountPercent,

    masterItem["Item Discount Percent"],

    masterItem.discountPercent,

    masterItem["Discount Percent"],

    inputItem.itemDiscountPercent

  ];


  for (
    let i = 0;
    i < possibleValues.length;
    i++
  ) {

    const value =
      Number(
        possibleValues[i]
      );


    if (
      isFinite(value) &&
      value > 0 &&
      value <= 100
    ) {

      return value;

    }

  }


  return 0;

}

/************************************************
 * PREPARE SALE ITEM
 *
 * CENTRAL PRICING ENGINE
 *
 * RULES:
 *
 * 1. Never trust UI price.
 * 2. Server gets original price.
 * 3. Offer is evaluated first.
 * 4. Item discount is evaluated second.
 * 5. Invoice discount eligibility is determined
 *    by the final pricing policy.
 ************************************************/

function mairixPrepareSaleItem(
  context
) {

  if (!context) {

    throw new Error(
      "Sale item context is required."
    );

  }


  const inputItem =
    context.inputItem || {};


  const lineNumber =
    Number(
      context.lineNumber || 0
    );


  /**********************************************
   * ITEM CODE
   **********************************************/

  const itemCode =
    mairixCleanText(
      inputItem.itemCode ||
      inputItem["Item Code"]
    );


  if (!itemCode) {

    throw new Error(
      "Item Code is required at line " +
      lineNumber
    );

  }


  /**********************************************
   * BARCODE
   **********************************************/

  const barcode =
    mairixCleanText(
      inputItem.barcode ||
      inputItem["Barcode"]
    );


  /**********************************************
   * QUANTITY
   **********************************************/

  const quantity =
    Number(
      inputItem.quantity ||
      inputItem["Quantity"]
    );


  if (
    !isFinite(quantity) ||
    quantity <= 0
  ) {

    throw new Error(
      "Invalid quantity for item: " +
      itemCode
    );

  }


  /**********************************************
   * MASTER ITEM
   **********************************************/

  const masterItem =
    mairixGetItem(
      itemCode
    );


  if (!masterItem) {

    throw new Error(
      "Item not found: " +
      itemCode
    );

  }


  /**********************************************
   * AVAILABLE STOCK
   **********************************************/

  const availableStock =
    Number(
      mairixGetWarehouseStock(
        context.warehouseCode,
        itemCode
      ) || 0
    );


  const allowNegativeStock =
    context.settings &&
    context.settings.allowNegativeStock === true;


  if (
    !allowNegativeStock &&
    availableStock < quantity
  ) {

    throw new Error(

      "Insufficient stock for item: " +
      itemCode +
      " | Available: " +
      availableStock +
      " | Requested: " +
      quantity

    );

  }


  /**********************************************
   * AUTHORITATIVE ORIGINAL PRICE
   **********************************************/

  const originalUnitPrice =
    mairixGetAuthoritativeSalePrice(
      masterItem,
      inputItem
    );


  /**********************************************
   * OFFER
   **********************************************/

  let offer =
    null;


  if (
    context.settings &&
    context.settings.offersEnabled
  ) {

    offer =
      mairixResolveActiveSaleOffer({

        itemCode:
          itemCode,

        barcode:
          barcode,

        date:
          context.transactionDate,

        originalUnitPrice:
          originalUnitPrice

      });

  }


  let offerApplied =
    false;


  let offerPrice =
    0;


  let offerSavings =
    0;


  let pricingUnitPrice =
    originalUnitPrice;


  if (
    offer &&
    offer.active === true
  ) {

    const resolvedOfferPrice =
      Number(
        offer.finalPrice
      );


    if (
      isFinite(resolvedOfferPrice) &&
      resolvedOfferPrice >= 0 &&
      resolvedOfferPrice <
      originalUnitPrice
    ) {

      offerApplied =
        true;


      offerPrice =
        roundMairixMoney(
          resolvedOfferPrice
        );


      pricingUnitPrice =
        offerPrice;


      offerSavings =
        roundMairixMoney(

          (
            originalUnitPrice -
            offerPrice
          ) *
          quantity

        );

    }

  }


  /**********************************************
   * OFFER POLICY
   **********************************************/

  const offerBlocksOtherDiscounts =
    context.settings &&
    context.settings.offerBlocksOtherDiscounts === true;


  /**********************************************
   * ITEM DISCOUNT
   **********************************************/

  let itemDiscountPercent =
    0;


  let itemDiscountAmount =
    0;


  let itemDiscountSavings =
    0;


  if (
    !offerApplied ||
    !offerBlocksOtherDiscounts
  ) {

    itemDiscountPercent =
      mairixGetItemDiscountPercent(
        masterItem,
        inputItem
      );


    if (
      itemDiscountPercent > 0
    ) {

      itemDiscountAmount =
        roundMairixMoney(

          pricingUnitPrice *
          quantity *
          itemDiscountPercent /
          100

        );


      itemDiscountSavings =
        itemDiscountAmount;

    }

  }


  /**********************************************
   * GROSS AMOUNT
   *
   * BEFORE ITEM / INVOICE DISCOUNT
   **********************************************/

  const grossAmount =
    roundMairixMoney(
      pricingUnitPrice *
      quantity
    );


  /**********************************************
   * INVOICE DISCOUNT ELIGIBILITY
   *
   * DEFAULT POLICY:
   *
   * OFFER + BLOCKING SETTING
   *       => NOT ELIGIBLE
   *
   * ITEM DISCOUNT
   *       => NOT ELIGIBLE
   **********************************************/

  let invoiceDiscountEligible =
    true;


  if (
    itemDiscountPercent > 0
  ) {

    invoiceDiscountEligible =
      false;

  }


  if (
    offerApplied &&
    offerBlocksOtherDiscounts
  ) {

    invoiceDiscountEligible =
      false;

  }


  /**********************************************
   * PRE-INVOICE NET
   **********************************************/

  const netAmountAfterItemDiscount =
    roundMairixMoney(

      grossAmount -
      itemDiscountAmount

    );


  /**********************************************
   * ITEM NAME
   **********************************************/

  const itemName =
    mairixCleanText(

      masterItem.name ||
      masterItem["Item Name"] ||
      masterItem.itemName ||
      ""

    );


  /**********************************************
   * RETURN COMPLETE PRICING OBJECT
   **********************************************/

  return {

    lineNumber:
      lineNumber,

    itemCode:
      itemCode,

    itemName:
      itemName,

    barcode:
      barcode,

    warehouseCode:
      context.warehouseCode,

    quantity:
      quantity,

    availableStock:
      availableStock,


    /********************************************
     * ORIGINAL PRICE
     ********************************************/

    originalUnitPrice:
      originalUnitPrice,


    /********************************************
     * OFFER
     ********************************************/

    offerApplied:
      offerApplied,

    offerPrice:
      offerPrice,

    offerSavings:
      offerSavings,


    /********************************************
     * ITEM DISCOUNT
     ********************************************/

    itemDiscountPercent:
      itemDiscountPercent,

    itemDiscountAmount:
      itemDiscountAmount,

    itemDiscountSavings:
      itemDiscountSavings,


    /********************************************
     * INVOICE DISCOUNT
     ********************************************/

    invoiceDiscountEligible:
      invoiceDiscountEligible,

    invoiceDiscountPercent:
      0,

    invoiceDiscountAmount:
      0,


    /********************************************
     * AMOUNTS
     ********************************************/

    grossAmount:
      grossAmount,

    totalDiscountAmount:
      itemDiscountAmount,

    netAmountBeforeTax:
      netAmountAfterItemDiscount,


    /********************************************
     * TEMPORARY FINAL PRICE
     *
     * Will be recalculated after invoice
     * discount distribution.
     ********************************************/

    finalUnitPrice:
      quantity > 0
        ? roundMairixMoney(

            netAmountAfterItemDiscount /
            quantity

          )
        : 0,


    /********************************************
     * SAVINGS
     ********************************************/

    preInvoiceSavings:
      roundMairixMoney(

        offerSavings +
        itemDiscountSavings

      ),

    totalSavings:
      roundMairixMoney(

        offerSavings +
        itemDiscountSavings

      )

  };

}

  context
) {

  /**********************************************
   * ADAPTER
   *
   * Replace this block only when connecting

   **********************************************/

  if (

    });

  }

        context.itemCode,

      barcode:
        context.barcode,

      date:
        context.date,

      originalUnitPrice:
        context.originalUnitPrice

    });

  }


  return null;

}

function mairixResolveTransactionDate(
  inputDate
) {

  if (!inputDate) {

    return new Date();

  }


  const parsedDate =
    new Date(
      inputDate
    );


  if (
    isNaN(
      parsedDate.getTime()
    )
  ) {

    throw new Error(
      "Invalid transaction date."
    );

  }


  return parsedDate;

}

  value
) {

  if (
    value === null ||
    value === undefined ||
    value === ""
  ) {

    return 0;

  }


  const result =
    Number(value);


  if (
    !isFinite(result) ||
    result < 0 ||
    result > 100
  ) {

    throw new Error(
      "Invalid percentage."
    );

  }


  return result;

}

function roundMairixNumber(
  value
) {

  const number =
    Number(value);


  if (
    !isFinite(number)
  ) {

    return 0;

  }


  return Math.round(
    number
  );

}

/************************************************
 * SAFE BOOLEAN NORMALIZER
 ************************************************/

function mairixToBoolean(
  value,
  defaultValue
) {

  if (
    value === true ||
    value === false
  ) {

    return value;

  }


  if (
    value === null ||
    value === undefined ||
    value === ""
  ) {

    return defaultValue;

  }


  const normalized =
    String(value)
      .trim()
      .toUpperCase();


  if (
    normalized === "TRUE" ||
    normalized === "YES" ||
    normalized === "1"
  ) {

    return true;

  }


  if (
    normalized === "FALSE" ||
    normalized === "NO" ||
    normalized === "0"
  ) {

    return false;

  }


  return defaultValue;

}

/************************************************
 * APPLY INVOICE DISCOUNT TO PREPARED SALE ITEMS
 *
 * RULES:
 *
 * 1. Only eligible items receive invoice discount.
 * 2. Offer items are excluded when policy blocks it.
 * 3. Item-discounted items are excluded.
 * 4. Discount is distributed proportionally.
 * 5. Rounding difference is assigned to last item.
 ************************************************/

function mairixApplyInvoiceDiscountToPreparedItems(
  preparedItems,
  invoiceDiscountPercent
) {

  if (
    !Array.isArray(preparedItems) ||
    preparedItems.length === 0
  ) {

    return {
      eligibleSubtotal: 0,
      invoiceDiscountPercent: 0,
      invoiceDiscountAmount: 0
    };

  }


  const discountPercent =
    Number(invoiceDiscountPercent || 0);


  if (
    !isFinite(discountPercent) ||
    discountPercent <= 0
  ) {

    preparedItems.forEach(
      function(item) {

        item.invoiceDiscountPercent = 0;
        item.invoiceDiscountAmount = 0;

        item.totalDiscountAmount =
          roundMairixMoney(
            Number(item.itemDiscountAmount || 0)
          );

        item.totalSavings =
          roundMairixMoney(

            Number(item.offerSavings || 0) +
            Number(item.itemDiscountAmount || 0)

          );

      }
    );


    return {
      eligibleSubtotal: 0,
      invoiceDiscountPercent: 0,
      invoiceDiscountAmount: 0
    };

  }


  /**********************************************
   * GET ELIGIBLE ITEMS
   **********************************************/

  const eligibleItems =
    preparedItems.filter(
      function(item) {

        return (
          item.invoiceDiscountEligible === true
        );

      }
    );


  if (
    eligibleItems.length === 0
  ) {

    return {
      eligibleSubtotal: 0,
      invoiceDiscountPercent: discountPercent,
      invoiceDiscountAmount: 0
    };

  }


  /**********************************************
   * CALCULATE ELIGIBLE SUBTOTAL
   *
   * Uses amount after item-level pricing.
   **********************************************/

  const eligibleSubtotal =
    eligibleItems.reduce(
      function(total, item) {

        return (
          total +
          Number(
            item.netAmountBeforeTax || 0
          )
        );

      },
      0
    );


  if (
    eligibleSubtotal <= 0
  ) {

    return {
      eligibleSubtotal: 0,
      invoiceDiscountPercent: discountPercent,
      invoiceDiscountAmount: 0
    };

  }


  /**********************************************
   * TOTAL INVOICE DISCOUNT
   **********************************************/

  const totalInvoiceDiscount =
    roundMairixMoney(

      eligibleSubtotal *
      discountPercent /
      100

    );


  let distributedDiscount =
    0;


  /**********************************************
   * APPLY DISCOUNT PROPORTIONALLY
   **********************************************/

  eligibleItems.forEach(
    function(item, index) {

      let itemInvoiceDiscount = 0;


      const isLastItem =
        index ===
        eligibleItems.length - 1;


      if (isLastItem) {

        /******************************************
         * ROUNDING DIFFERENCE
         ******************************************/

        itemInvoiceDiscount =
          roundMairixMoney(

            totalInvoiceDiscount -
            distributedDiscount

          );

      } else {

        const itemAmount =
          Number(
            item.netAmountBeforeTax || 0
          );


        const ratio =
          itemAmount /
          eligibleSubtotal;


        itemInvoiceDiscount =
          roundMairixMoney(

            totalInvoiceDiscount *
            ratio

          );

      }


      distributedDiscount =
        roundMairixMoney(

          distributedDiscount +
          itemInvoiceDiscount

        );


      /********************************************
       * UPDATE ITEM
       ********************************************/

      item.invoiceDiscountPercent =
        discountPercent;


      item.invoiceDiscountAmount =
        itemInvoiceDiscount;


      item.netAmountBeforeTax =
        roundMairixMoney(

          Number(
            item.netAmountBeforeTax || 0
          ) -
          itemInvoiceDiscount

        );


      item.finalUnitPrice =
        Number(item.quantity || 0) > 0

          ? roundMairixMoney(

              item.netAmountBeforeTax /
              Number(item.quantity)

            )

          : 0;


      item.totalDiscountAmount =
        roundMairixMoney(

          Number(item.itemDiscountAmount || 0) +
          itemInvoiceDiscount

        );


      item.totalSavings =
        roundMairixMoney(

          Number(item.offerSavings || 0) +
          Number(item.itemDiscountAmount || 0) +
          itemInvoiceDiscount

        );

    }
  );


  /**********************************************
   * RESET NON-ELIGIBLE ITEMS
   **********************************************/

  preparedItems.forEach(
    function(item) {

      if (
        item.invoiceDiscountEligible === true
      ) {

        return;

      }


      item.invoiceDiscountPercent = 0;
      item.invoiceDiscountAmount = 0;


      item.totalDiscountAmount =
        roundMairixMoney(
          Number(item.itemDiscountAmount || 0)
        );


      item.totalSavings =
        roundMairixMoney(

          Number(item.offerSavings || 0) +
          Number(item.itemDiscountAmount || 0)

        );

    }
  );


  return {

    eligibleSubtotal:
      roundMairixMoney(
        eligibleSubtotal
      ),


    invoiceDiscountPercent:
      discountPercent,


    invoiceDiscountAmount:
      totalInvoiceDiscount

  };

}

/************************************************
 * SALES PRICING INTEGRATION TEST
 *
 * TESTS:
 *
 * 1. Normal item + invoice discount
 * 2. Offer item
 * 3. Item discount
 * 4. Mixed invoice
 * 5. Invoice discount distribution
 *
 * RUN:
 *

/************************************************
 * FIND ONE ROW BY COLUMN VALUE
 ************************************************/

function mairixFindRowByValue(
  sheetName,
  columnName,
  searchValue
) {

  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        sheetName
      );


  if (!sheet) {

    throw new Error(
      "Sheet not found: " +
      sheetName
    );

  }


  const values =
    sheet.getDataRange()
      .getValues();


  if (
    !values ||
    values.length < 2
  ) {

    return null;

  }


  const headers =
    values[0].map(
      function(header) {

        return String(
          header
        ).trim();

      }
    );


  const columnIndex =
    headers.indexOf(
      columnName
    );


  if (
    columnIndex === -1
  ) {

    throw new Error(
      "Column not found in " +
      sheetName +
      ": " +
      columnName
    );

  }


  const normalizedSearchValue =
    String(
      searchValue
    ).trim();


  for (
    let i = 1;
    i < values.length;
    i++
  ) {

    const cellValue =
      String(
        values[i][columnIndex]
      ).trim();


    if (
      cellValue ===
      normalizedSearchValue
    ) {

      const rowObject = {};


      headers.forEach(
        function(header, index) {

          rowObject[header] =
            values[i][index];

        }
      );


      return rowObject;

    }

  }


  return null;

}

/************************************************
 * GET FIRST ACTIVE MASTER CODE
 *
 * AUTO-DETECTS COMMON CODE COLUMNS
 ************************************************/

function mairixGetFirstActiveMasterCode(
  sheetName
) {

  if (
    !sheetName ||
    typeof sheetName !== "string"
  ) {

    throw new Error(
      "Master sheet name is required. Received: " +
      String(sheetName)
    );

  }


  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        sheetName
      );


  if (!sheet) {

    throw new Error(
      "Master sheet not found: " +
      sheetName
    );

  }
  const values =
    sheet.getDataRange()
      .getValues();


  if (
    values.length < 2
  ) {

    return null;

  }


  const headers =
    values[0].map(
      function(header) {

        return String(
          header
        ).trim()
        .toUpperCase();

      }
    );


  const possibleCodeColumns = [

    "CUSTOMER CODE",
    "SALESMAN CODE",
    "WAREHOUSE CODE",
    "CODE"

  ];


  let codeIndex = -1;


  for (
    let i = 0;
    i < possibleCodeColumns.length;
    i++
  ) {

    const index =
      headers.indexOf(
        possibleCodeColumns[i]
      );


    if (
      index !== -1
    ) {

      codeIndex = index;

      break;

    }

  }


  if (
    codeIndex === -1
  ) {

    throw new Error(
      "No Code column found in: " +
      sheetName
    );

  }


  const statusIndex =
    headers.indexOf(
      "STATUS"
    );


  for (
    let i = 1;
    i < values.length;
    i++
  ) {

    const code =
      String(
        values[i][codeIndex] || ""
      ).trim();


    if (!code) {

      continue;

    }


    if (
      statusIndex !== -1
    ) {

      const status =
        String(
          values[i][statusIndex] || ""
        )
        .trim()
        .toUpperCase();


      if (
        status &&
        status !== "ACTIVE"
      ) {

        continue;

      }

    }


    return code;

  }


  return null;

}

/************************************************
 * GET FIRST SELLABLE BARCODE
 *
 * REAL DATA ONLY
 *
 * DOES NOT INVENT TEST ITEMS.
 ************************************************/

function mairixGetFirstSellableBarcode() {

  const sheet =
    SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName(
        "ITEMS"
      );


  if (!sheet) {

    throw new Error(
      "ITEMS sheet not found."
    );

  }


  const lastRow =
    sheet.getLastRow();


  if (
    lastRow < 2
  ) {

    throw new Error(

      "REAL SALE TEST PRECHECK FAILED: " +
      "ITEMS contains no real item records. " +
      "Create at least one active item before running a real sale test."

    );

  }


  const lastColumn =
    sheet.getLastColumn();


  const values =
    sheet
      .getRange(
        1,
        1,
        lastRow,
        lastColumn
      )
      .getValues();


  const headers =
    values[0].map(
      function(header) {

        return String(
          header || ""
        )
        .trim()
        .toUpperCase();

      }
    );


  /**********************************************
   * FIND ITEM CODE COLUMN
   **********************************************/

  const possibleItemCodeColumns = [

    "ITEM CODE",

    "ITEM_CODE",

    "CODE"

  ];


  let itemCodeIndex = -1;


  for (
    let i = 0;
    i < possibleItemCodeColumns.length;
    i++
  ) {

    const index =
      headers.indexOf(
        possibleItemCodeColumns[i]
      );


    if (
      index !== -1
    ) {

      itemCodeIndex = index;

      break;

    }

  }


  if (
    itemCodeIndex === -1
  ) {

    throw new Error(

      "ITEMS sheet has no recognized Item Code column. " +
      "Headers: " +
      JSON.stringify(headers)

    );

  }


  /**********************************************
   * FIND BARCODE COLUMN
   **********************************************/

  const possibleBarcodeColumns = [

    "BARCODE",

    "BAR CODE",

    "BARCODE 1",

    "PRIMARY BARCODE"

  ];


  let barcodeIndex = -1;


  for (
    let i = 0;
    i < possibleBarcodeColumns.length;
    i++
  ) {

    const index =
      headers.indexOf(
        possibleBarcodeColumns[i]
      );


    if (
      index !== -1
    ) {

      barcodeIndex = index;

      break;

    }

  }


  /**********************************************
   * OPTIONAL STATUS COLUMNS
   **********************************************/

  const statusIndex =
    headers.indexOf(
      "STATUS"
    );


  const activeIndex =
    headers.indexOf(
      "ACTIVE"
    );


  /**********************************************
   * FIND FIRST VALID ITEM
   **********************************************/

  for (
    let rowIndex = 1;
    rowIndex < values.length;
    rowIndex++
  ) {

    const row =
      values[rowIndex];


    const itemCode =
      String(
        row[itemCodeIndex] || ""
      ).trim();


    if (!itemCode) {

      continue;

    }


    /********************************************
     * STATUS CHECK
     ********************************************/

    if (
      statusIndex !== -1
    ) {

      const status =
        String(
          row[statusIndex] || ""
        )
        .trim()
        .toUpperCase();


      if (
        status === "INACTIVE" ||
        status === "DISABLED" ||
        status === "CANCELLED"
      ) {

        continue;

      }

    }


    /********************************************
     * ACTIVE CHECK
     ********************************************/

    if (
      activeIndex !== -1
    ) {

      const activeValue =
        String(
          row[activeIndex]
        )
        .trim()
        .toUpperCase();


      if (
        activeValue === "FALSE" ||
        activeValue === "NO" ||
        activeValue === "0"
      ) {

        continue;

      }

    }


    /********************************************
     * BARCODE
     ********************************************/

    const barcode =
      barcodeIndex !== -1
        ? String(
            row[barcodeIndex] || ""
          ).trim()
        : "";


    /********************************************
     * RETURN
     ********************************************/

    return {

      itemCode:
        itemCode,

      barcode:
        barcode || itemCode

    };

  }


  throw new Error(

    "REAL SALE TEST PRECHECK FAILED: " +
    "No active valid item record was found in ITEMS."

  );

}

    );

  }

   **********************************************/

  const beforeStock =
    Number(
      beforeSnapshot.stock || 0
    );


  const afterStock =
    Number(
      afterSnapshot.stock || 0
    );


  if (
    beforeStock !== 0 ||
    afterStock !== 0
  ) {

    mairixSalesPricingAssertMoney(

      afterStock,

    );

  }


  /**********************************************
   * SUCCESS
   **********************************************/

  const testResult = {

    success:
      true,

    saleId:
      saleId,

      testContext,

      result,

      beforeSnapshot,

  );

/************************************************
 * MAIRIX SALES PATCH
 * Target: 10-MAIRIX_Sales.gs
 *
 * FIXES:
 * 1) Barcode-only sale input
 * 2) Item Code / Barcode consistency validation
 * 3) Flexible ITEMS header resolution
 * 4) Correct Loyalty/Commission test sheet aliases
 *
 * INSTALL:
 * Replace ONLY the existing mairixPrepareSaleItem()
 * function with the implementation below, then add
 * mairixResolveSaleItemIdentity() and helpers.
 ************************************************/

function mairixResolveSaleItemIdentity(inputItem, lineNumber) {
  inputItem = inputItem || {};

  const requestedItemCode = mairixCleanText(
    inputItem.itemCode ||
    inputItem["Item Code"] ||
    inputItem.code ||
    inputItem["Code"]
  );

  const requestedBarcode = mairixCleanText(
    inputItem.barcode ||
    inputItem["Barcode"]
  );

  if (!requestedItemCode && !requestedBarcode) {
    throw new Error(
      "Item Code or Barcode is required at line " +
      Number(lineNumber || 0)
    );
  }

  let itemByCode = null;
  let itemByBarcode = null;

  if (requestedItemCode) {
    itemByCode = mairixGetItem(requestedItemCode);

    if (!itemByCode) {
      throw new Error(
        "Item not found by Item Code at line " +
        Number(lineNumber || 0) +
        ": " +
        requestedItemCode
      );
    }
  }

  if (requestedBarcode) {
    itemByBarcode = mairixFindItemByBarcode(requestedBarcode);

    if (!itemByBarcode) {
      throw new Error(
        "Item not found by Barcode at line " +
        Number(lineNumber || 0) +
        ": " +
        requestedBarcode
      );
    }
  }

  if (itemByCode && itemByBarcode) {
    const codeFromCodeLookup =
      mairixGetItemCodeFromRecord(itemByCode);

    const codeFromBarcodeLookup =
      mairixGetItemCodeFromRecord(itemByBarcode);

    if (
      codeFromCodeLookup &&
      codeFromBarcodeLookup &&
      codeFromCodeLookup !== codeFromBarcodeLookup
    ) {
      throw new Error(
        "Item Code and Barcode refer to different items at line " +
        Number(lineNumber || 0) +
        "."
      );
    }
  }

  const masterItem =
    itemByCode ||
    itemByBarcode;

  const resolvedItemCode =
    mairixGetItemCodeFromRecord(masterItem) ||
    requestedItemCode;

  const resolvedBarcode =
    mairixGetItemBarcodeFromRecord(masterItem) ||
    requestedBarcode;

  if (!resolvedItemCode) {
    throw new Error(
      "Could not resolve authoritative Item Code at line " +
      Number(lineNumber || 0)
    );
  }

  return {
    itemCode: resolvedItemCode,
    barcode: resolvedBarcode,
    masterItem: masterItem
  };
}


function mairixGetItemCodeFromRecord(item) {
  if (!item) return "";

  return mairixCleanText(
    item.itemCode ||
    item["Item Code"] ||
    item.code ||
    item["Code"]
  );
}


function mairixGetItemBarcodeFromRecord(item) {
  if (!item) return "";

  return mairixCleanText(
    item.barcode ||
    item["Barcode"]
  );
}


function mairixFindItemByBarcode(barcode) {
  barcode = mairixCleanText(barcode);

  if (!barcode) {
    return null;
  }

  if (typeof getMairixItemByBarcode === "function") {
    const result = getMairixItemByBarcode(barcode);
    if (result) return result;
  }

  if (typeof getItemByBarcode === "function") {
    const result = getItemByBarcode(barcode);
    if (result) return result;
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("ITEMS");

  if (!sheet || sheet.getLastRow() < 2) {
    return null;
  }

  const headers =
    sheet.getRange(
      1,
      1,
      1,
      sheet.getLastColumn()
    ).getDisplayValues()[0];

  const barcodeIndex =
    headers.indexOf("Barcode");

  if (barcodeIndex < 0) {
    throw new Error(
      'ITEMS sheet does not contain required header: "Barcode"'
    );
  }

  const values =
    sheet.getRange(
      2,
      1,
      sheet.getLastRow() - 1,
      sheet.getLastColumn()
    ).getDisplayValues();

  for (let i = 0; i < values.length; i++) {
    if (
      mairixCleanText(values[i][barcodeIndex]) === barcode
    ) {
      const record = {};

      headers.forEach(function(header, index) {
        record[header] = values[i][index];
      });

      record.itemCode =
        mairixGetItemCodeFromRecord(record);

      record.barcode =
        mairixGetItemBarcodeFromRecord(record);

      return record;
    }
  }

  return null;
}


/************************************************
 * REPLACEMENT:
 * mairixPrepareSaleItem
 ************************************************/

function mairixPrepareSaleItem(context) {

  if (!context) {
    throw new Error(
      "Sale item context is required."
    );
  }

  const inputItem =
    context.inputItem || {};

  const lineNumber =
    Number(context.lineNumber || 0);

  /**********************************************
   * AUTHORITATIVE ITEM IDENTITY
   *
   * UI may send Item Code OR Barcode.
   **********************************************/

  const identity =
    mairixResolveSaleItemIdentity(
      inputItem,
      lineNumber
    );

  const itemCode =
    identity.itemCode;

  const barcode =
    identity.barcode;

  const masterItem =
    identity.masterItem;

  /**********************************************
   * QUANTITY
   **********************************************/

  const quantity =
    Number(
      inputItem.quantity ||
      inputItem["Quantity"]
    );

  if (
    !isFinite(quantity) ||
    quantity <= 0
  ) {
    throw new Error(
      "Invalid quantity for item: " +
      itemCode
    );
  }

  /**********************************************
   * AVAILABLE STOCK
   **********************************************/

  const availableStock =
    Number(
      mairixGetWarehouseStock(
        context.warehouseCode,
        itemCode
      ) || 0
    );

  const allowNegativeStock =
    context.settings &&
    context.settings.allowNegativeStock === true;

  if (
    !allowNegativeStock &&
    availableStock < quantity
  ) {
    throw new Error(
      "Insufficient stock for item: " +
      itemCode +
      " | Available: " +
      availableStock +
      " | Requested: " +
      quantity
    );
  }

  /**********************************************
   * AUTHORITATIVE ORIGINAL PRICE
   **********************************************/

  const originalUnitPrice =
    mairixGetAuthoritativeSalePrice(
      masterItem,
      inputItem
    );

  /**********************************************
   * OFFER
   **********************************************/

  let offer = null;

  if (
    context.settings &&
    context.settings.offersEnabled
  ) {
    offer =
      mairixResolveActiveSaleOffer({
        itemCode: itemCode,
        barcode: barcode,
        date: context.transactionDate,
        originalUnitPrice: originalUnitPrice
      });
  }

  let offerApplied = false;
  let offerPrice = 0;
  let offerSavings = 0;
  let pricingUnitPrice = originalUnitPrice;

  if (
    offer &&
    offer.active === true
  ) {
    const resolvedOfferPrice =
      Number(offer.finalPrice);

    if (
      isFinite(resolvedOfferPrice) &&
      resolvedOfferPrice >= 0 &&
      resolvedOfferPrice < originalUnitPrice
    ) {
      offerApplied = true;

      offerPrice =
        roundMairixMoney(
          resolvedOfferPrice
        );

      pricingUnitPrice = offerPrice;

      offerSavings =
        roundMairixMoney(
          (
            originalUnitPrice -
            offerPrice
          ) * quantity
        );
    }
  }

  /**********************************************
   * OFFER POLICY
   **********************************************/

  const offerBlocksOtherDiscounts =
    context.settings &&
    context.settings.offerBlocksOtherDiscounts === true;

  /**********************************************
   * ITEM DISCOUNT
   **********************************************/

  let itemDiscountPercent = 0;
  let itemDiscountAmount = 0;
  let itemDiscountSavings = 0;

  if (
    !offerApplied ||
    !offerBlocksOtherDiscounts
  ) {
    itemDiscountPercent =
      mairixGetItemDiscountPercent(
        masterItem,
        inputItem
      );

    if (itemDiscountPercent > 0) {
      itemDiscountAmount =
        roundMairixMoney(
          pricingUnitPrice *
          quantity *
          itemDiscountPercent /
          100
        );

      itemDiscountSavings =
        itemDiscountAmount;
    }
  }

  const grossAmount =
    roundMairixMoney(
      pricingUnitPrice *
      quantity
    );

  let invoiceDiscountEligible = true;

  if (itemDiscountPercent > 0) {
    invoiceDiscountEligible = false;
  }

  if (
    offerApplied &&
    offerBlocksOtherDiscounts
  ) {
    invoiceDiscountEligible = false;
  }

  const netAmountAfterItemDiscount =
    roundMairixMoney(
      grossAmount -
      itemDiscountAmount
    );

  const itemName =
    mairixCleanText(
      masterItem.name ||
      masterItem["Item Name"] ||
      masterItem.itemName ||
      ""
    );

  return {
    lineNumber: lineNumber,
    itemCode: itemCode,
    itemName: itemName,
    barcode: barcode,
    warehouseCode: context.warehouseCode,
    quantity: quantity,
    availableStock: availableStock,
    originalUnitPrice: originalUnitPrice,
    offerApplied: offerApplied,
    offerPrice: offerPrice,
    offerSavings: offerSavings,
    itemDiscountPercent: itemDiscountPercent,
    itemDiscountAmount: itemDiscountAmount,
    itemDiscountSavings: itemDiscountSavings,
    invoiceDiscountEligible: invoiceDiscountEligible,
    invoiceDiscountPercent: 0,
    invoiceDiscountAmount: 0,
    grossAmount: grossAmount,
    totalDiscountAmount: itemDiscountAmount,
    netAmountBeforeTax: netAmountAfterItemDiscount,
    finalUnitPrice:
      quantity > 0
        ? roundMairixMoney(
            netAmountAfterItemDiscount /
            quantity
          )
        : 0,
    preInvoiceSavings:
      roundMairixMoney(
        offerSavings +
        itemDiscountSavings
      ),
    totalSavings:
      roundMairixMoney(
        offerSavings +
        itemDiscountSavings
      )
  };
}


/************************************************
 * SAFE SHEET NAME RESOLUTION
 ************************************************/

function mairixResolveExistingSheetName(candidates) {

  if (!candidates) {
    return "";
  }


  if (!Array.isArray(candidates)) {

    candidates = [
      candidates
    ];

  }


  if (candidates.length === 0) {
    return "";
  }


  const ss =
    SpreadsheetApp.getActiveSpreadsheet();


  for (
    let i = 0;
    i < candidates.length;
    i++
  ) {

    const sheetName =
      mairixCleanText(
        candidates[i]
      );


    if (!sheetName) {
      continue;
    }


    const sheet =
      ss.getSheetByName(
        sheetName
      );


    if (sheet) {
      return sheetName;
    }

  }


  return "";

}
