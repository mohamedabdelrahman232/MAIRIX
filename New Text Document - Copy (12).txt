function mairixE2ESupplierPaymentCreate(data) {

  if (typeof apiCreateVendorPayment !== "function") {
    throw new Error("apiCreateVendorPayment() is not available.");
  }

  if (!data) {
    throw new Error("Vendor payment data is required.");
  }

  var request = Object.assign({}, data);

  delete request.userId;
  delete request.serverUserId;
  delete request.createdBy;
  delete request.updatedBy;

  var result = apiCreateVendorPayment(request);

  if (!result || result.success !== true) {
    throw new Error(
      "Vendor payment integration failed: " +
      JSON.stringify(result && result.error ? result.error : result)
    );
  }

  return result;
}
