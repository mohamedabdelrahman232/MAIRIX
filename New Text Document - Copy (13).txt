function mairixE2ECustomerCollectionCreate(data) {

  if (typeof apiCreateCollection !== "function") {
    throw new Error("apiCreateCollection() is not available.");
  }

  if (!data) {
    throw new Error("Collection data is required.");
  }

  var request = Object.assign({}, data);

  delete request.userId;
  delete request.serverUserId;
  delete request.createdBy;
  delete request.updatedBy;

  var result = apiCreateCollection(request);

  if (!result || result.success !== true) {
    throw new Error(
      "Collection integration failed: " +
      JSON.stringify(result && result.error ? result.error : result)
    );
  }

  return result;
}
