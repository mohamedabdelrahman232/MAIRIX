/************************************************
 * MAIRIX ERP
 * 9-MAIRIX_FinancialEngine.gs
 *
 * CENTRAL FINANCIAL ENGINE
 *
 * RESPONSIBILITIES
 * ----------------------------------------------
 * 1. Central financial movement register
 * 2. Cash balance tracking
 * 3. Customer credit balance tracking
 * 4. Vendor payable balance tracking
 * 5. Financial movement sign convention
 * 6. Central Reference ID
 * 7. Link financial effects to source transactions
 * 8. Separate cash from credit balances
 * 9. Prepare financial data for reports
 *
 * IMPORTANT
 * ----------------------------------------------
 * This engine is NOT the final financial
 * statement engine.
 *
 * It is the CENTRAL FINANCIAL MOVEMENT LAYER.
 *
 * Later:
 *
 * FINANCIAL MOVEMENTS
 *        |
 *        +--> Cash
 *        |
 *        +--> Customers Receivable
 *        |
 *        +--> Vendors Payable
 *        |
 *        +--> Revenue
 *        |
 *        +--> COGS
 *        |
 *        +--> Expenses
 *        |
 *        +--> Commissions
 *        |
 *        +--> Taxes
 *        |
 *        +--> Financial Statements
 *
 *
 * SIGN PRINCIPLE
 * ----------------------------------------------
 *
 * Each financial movement has explicit
 * calculated effect columns.
 *
 * Reports SUM effects.
 *
 * They do NOT need to reconstruct meaning
 * from Movement Type.
 ************************************************/

/************************************************
 * SETUP FINANCIAL ENGINE
 ************************************************/

function setupMAIRIXFinancialEngine() {

  const ss =
    SpreadsheetApp
      .getActiveSpreadsheet();


  /*
   * CENTRAL FINANCIAL MOVEMENTS
   */

  let movementSheet =
    ss.getSheetByName(
      MAIRIX_FINANCIAL_CONFIG.SHEETS.MOVEMENTS
    );


  if (!movementSheet) {

    movementSheet =
      ss.insertSheet(
        MAIRIX_FINANCIAL_CONFIG.SHEETS.MOVEMENTS
      );

  }


  movementSheet.clear();


  movementSheet
    .getRange(
      1,
      1,
      1,
      MAIRIX_FINANCIAL_MOVEMENT_HEADERS.length
    )
    .setValues([
      MAIRIX_FINANCIAL_MOVEMENT_HEADERS
    ]);


  movementSheet
    .setFrozenRows(1);


  movementSheet
    .getRange(
      1,
      1,
      1,
      MAIRIX_FINANCIAL_MOVEMENT_HEADERS.length
    )
    .setFontWeight("bold");


  movementSheet.autoResizeColumns(
    1,
    MAIRIX_FINANCIAL_MOVEMENT_HEADERS.length
  );


  /*
   * CASH BALANCES
   */

  createMAIRIXFinancialBalanceSheet(
    MAIRIX_FINANCIAL_CONFIG.SHEETS.CASH_BALANCES,
    [
      "Account Code",
      "Account Name",
      "Opening Balance",
      "Movement Effect",
      "Current Balance",
      "Last Reference ID"
    ]
  );


  /*
   * CUSTOMER CREDIT
   */

  createMAIRIXFinancialBalanceSheet(
    MAIRIX_FINANCIAL_CONFIG.SHEETS.CUSTOMER_BALANCES,
    [
      "Customer Code",
      "Customer Name",
      "Opening Balance",
      "Movement Effect",
      "Current Balance",
      "Last Reference ID"
    ]
  );


  /*
   * VENDOR PAYABLES
   */

  createMAIRIXFinancialBalanceSheet(
    MAIRIX_FINANCIAL_CONFIG.SHEETS.VENDOR_BALANCES,
    [
      "Vendor Code",
      "Vendor Name",
      "Opening Balance",
      "Movement Effect",
      "Current Balance",
      "Last Reference ID"
    ]
  );


  return {

    success:
      true,

    movementSheet:
      MAIRIX_FINANCIAL_CONFIG.SHEETS.MOVEMENTS,

    cashSheet:
      MAIRIX_FINANCIAL_CONFIG.SHEETS.CASH_BALANCES,

    customerCreditSheet:
      MAIRIX_FINANCIAL_CONFIG.SHEETS.CUSTOMER_BALANCES,

    vendorPayableSheet:
      MAIRIX_FINANCIAL_CONFIG.SHEETS.VENDOR_BALANCES

  };

}

/************************************************
 * GET FINANCIAL SIGN
 ************************************************/

function getMAIRIXFinancialSign(
  movementType
) {

  const type =
    mairixCleanText(
      movementType
    ).toUpperCase();


  const sign =
    MAIRIX_FINANCIAL_SIGN_MAP[type];


  if (!sign) {

    throw new Error(
      "Unsupported financial movement type: " +
      type
    );

  }


  return sign;

}


/************************************************
 * VALIDATE FINANCIAL MOVEMENT
 ************************************************/

function validateMAIRIXFinancialMovement(
  data
) {

  if (!data) {

    throw new Error(
      "Financial movement data is required."
    );

  }


  const movementType =
    mairixCleanText(
      data.movementType
    ).toUpperCase();


  if (
    !MAIRIX_FINANCIAL_SIGN_MAP[
      movementType
    ]
  ) {

    throw new Error(
      "Invalid financial movement type: " +
      movementType
    );

  }


  const referenceId =
    mairixCleanText(
      data.referenceId
    );


  if (!referenceId) {

    throw new Error(
      "Financial Reference ID is required."
    );

  }


  const amount =
    Number(
      data.netAmount
    );


  if (
    !isFinite(amount) ||
    amount < 0
  ) {

    throw new Error(
      "Financial amount must be zero or greater."
    );

  }


  return {

    movementType:
      movementType,

    referenceId:
      referenceId,

    partyType:
      mairixCleanText(
        data.partyType
      ),

    partyCode:
      mairixCleanText(
        data.partyCode
      ),

    partyName:
      mairixCleanText(
        data.partyName
      ),

    warehouseCode:
      mairixCleanText(
        data.warehouseCode
      ),

    warehouseName:
      mairixCleanText(
        data.warehouseName
      ),

    description:
      mairixCleanText(
        data.description
      ),

    grossAmount:
      Number(
        data.grossAmount || 0
      ),

    discountAmount:
      Number(
        data.discountAmount || 0
      ),

    netAmount:
      amount,

    taxAmount:
      Number(
        data.taxAmount || 0
      ),

    userId:
      mairixCleanText(
        data.userId
      ),

    userName:
      mairixCleanText(
        data.userName
      ),

    currency:
      mairixCleanText(
        data.currency ||
        "EGP"
      ) || "EGP"

  };

}


/************************************************
 * GENERATE FINANCIAL MOVEMENT ID
 ************************************************/

function generateMAIRIXFinancialMovementId() {

  const dateKey =
    Utilities.formatDate(
      new Date(),
      Session.getScriptTimeZone(),
      "yyyyMMdd-HHmmss"
    );


  const suffix =
    Utilities
      .getUuid()
      .substring(
        0,
        8
      )
      .toUpperCase();


  return (
    "FIN-" +
    dateKey +
    "-" +
    suffix
  );

}

/************************************************
 * RECORD FINANCIAL MOVEMENT
 *
 * CENTRAL ENTRY POINT
 ************************************************/

function recordMAIRIXFinancialMovement(
  data
) {

  const movement =
    validateMAIRIXFinancialMovement(
      data
    );


  const sign =
    getMAIRIXFinancialSign(
      movement.movementType
    );


  const sheet =
    getMAIRIXFinancialMovementSheet();


  /*
   * Prevent duplicate financial posting.
   */

  const duplicate =
    findMAIRIXFinancialMovementByReference(
      movement.referenceId,
      movement.movementType
    );


  if (duplicate) {

    throw new Error(
      "Financial movement already exists for Reference ID: " +
      movement.referenceId
    );

  }


  const movementId =
    generateMAIRIXFinancialMovementId();


  /*
   * Explicit calculated effects.
   */

  const cashEffect =
    movement.netAmount *
    sign.cash;


  const customerCreditEffect =
    movement.netAmount *
    sign.customerCredit;


  const vendorPayableEffect =
    movement.netAmount *
    sign.vendorPayable;


  const revenueEffect =
    movement.netAmount *
    sign.revenue;


  /*
   * COGS / Expense / Commission / Tax
   *
   * These are supplied by the originating
   * transaction engine.
   *
   * They are NOT inferred from movement type.
   */

  const cogsEffect =
    Number(
      data.cogsEffect || 0
    );


  const expenseEffect =
    Number(
      data.expenseEffect || 0
    );


  const commissionEffect =
    Number(
      data.commissionEffect || 0
    );


  const taxEffect =
    Number(
      data.taxEffect || 0
    );


  const row = [

    movementId,

    movement.referenceId,

    data.movementDate ||
      new Date(),

    movement.movementType,

    movement.partyType,

    movement.partyCode,

    movement.partyName,

    movement.warehouseCode,

    movement.warehouseName,

    movement.description,

    movement.grossAmount,

    movement.discountAmount,

    movement.netAmount,

    cashEffect,

    customerCreditEffect,

    vendorPayableEffect,

    revenueEffect,

    cogsEffect,

    expenseEffect,

    commissionEffect,

    taxEffect,

    movement.currency,

    movement.userId,

    movement.userName,

    new Date()

  ];


  sheet
    .getRange(
      sheet.getLastRow() + 1,
      1,
      1,
      row.length
    )
    .setValues([
      row
    ]);


  return {

    success:
      true,

    financialMovementId:
      movementId,

    referenceId:
      movement.referenceId,

    movementType:
      movement.movementType,

    netAmount:
      movement.netAmount,

    cashEffect:
      cashEffect,

    customerCreditEffect:
      customerCreditEffect,

    vendorPayableEffect:
      vendorPayableEffect,

    revenueEffect:
      revenueEffect,

    cogsEffect:
      cogsEffect,

    expenseEffect:
      expenseEffect,

    commissionEffect:
      commissionEffect,

    taxEffect:
      taxEffect

  };

}


/************************************************
 * FIND FINANCIAL MOVEMENT
 ************************************************/

function findMAIRIXFinancialMovementByReference(
  referenceId,
  movementType
) {

  const sheet =
    getMAIRIXFinancialMovementSheet();


  const lastRow =
    sheet.getLastRow();


  if (
    lastRow < 2
  ) {

    return null;

  }


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        headers.length
      )
      .getValues();


  const referenceIndex =
    headers.indexOf(
      "Reference ID"
    );


  const typeIndex =
    headers.indexOf(
      "Movement Type"
    );


  for (
    let i = 0;
    i < rows.length;
    i++
  ) {

    if (

      mairixCleanText(
        rows[i][referenceIndex]
      ) ===
      mairixCleanText(
        referenceId
      ) &&

      mairixCleanText(
        rows[i][typeIndex]
      ).toUpperCase() ===
      mairixCleanText(
        movementType
      ).toUpperCase()

    ) {

      const result = {};

      headers.forEach(
        function(header, index) {

          result[
            mairixCleanText(header)
          ] =
            rows[i][index];

        }
      );


      return result;

    }

  }


  return null;

}


/************************************************
 * GET CASH BALANCE
 *
 * CASH IS ALWAYS SEPARATE FROM CREDIT.
 ************************************************/

function getMAIRIXCashBalance() {

  const sheet =
    getMAIRIXFinancialMovementSheet();


  const lastRow =
    sheet.getLastRow();


  if (
    lastRow < 2
  ) {

    return 0;

  }


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const index =
    headers.indexOf(
      "Cash Effect"
    );


  if (
    index === -1
  ) {

    throw new Error(
      "Cash Effect column not found."
    );

  }


  const rows =
    sheet
      .getRange(
        2,
        index + 1,
        lastRow - 1,
        1
      )
      .getValues();


  return rows.reduce(
    function(total, row) {

      return (
        total +
        Number(
          row[0] || 0
        )
      );

    },
    0
  );

}

/************************************************
 * GENERIC PARTY BALANCE
 ************************************************/

function getMAIRIXPartyBalance(
  partyType,
  partyCode,
  effectColumnName
) {

  const sheet =
    getMAIRIXFinancialMovementSheet();


  const lastRow =
    sheet.getLastRow();


  if (
    lastRow < 2
  ) {

    return 0;

  }


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const partyTypeIndex =
    headers.indexOf(
      "Party Type"
    );


  const partyCodeIndex =
    headers.indexOf(
      "Party Code"
    );


  const effectIndex =
    headers.indexOf(
      effectColumnName
    );


  if (
    partyTypeIndex === -1 ||
    partyCodeIndex === -1 ||
    effectIndex === -1
  ) {

    throw new Error(
      "Financial party balance columns are incomplete."
    );

  }


  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        headers.length
      )
      .getValues();


  const normalizedType =
    mairixCleanText(
      partyType
    ).toUpperCase();


  const normalizedCode =
    mairixCleanText(
      partyCode
    );


  return rows.reduce(
    function(total, row) {

      if (
        mairixCleanText(
          row[partyTypeIndex]
        ).toUpperCase() !==
        normalizedType
      ) {

        return total;

      }


      if (
        mairixCleanText(
          row[partyCodeIndex]
        ) !==
        normalizedCode
      ) {

        return total;

      }


      return (
        total +
        Number(
          row[effectIndex] || 0
        )
      );

    },
    0
  );

}

 ************************************************/

  rebuildMAIRIXCashBalanceTable();

  rebuildMAIRIXCustomerBalanceTable();

  return {

    success:
      true

  };

}


/************************************************
 * REBUILD CASH TABLE
 ************************************************/

function rebuildMAIRIXCashBalanceTable() {

  const sheet =
    mairixgetSheet(
      MAIRIX_FINANCIAL_CONFIG.SHEETS.CASH_BALANCES
    );


  const balance =
    getMAIRIXCashBalance();


  if (
    sheet.getLastRow() > 1
  ) {

    sheet
      .getRange(
        2,
        1,
        sheet.getLastRow() - 1,
        6
      )
      .clearContent();

  }


  sheet
    .getRange(
      2,
      1,
      1,
      6
    )
    .setValues([

      [

        "CASH",

        "Cash",

        0,

        balance,

        balance,

        ""

      ]

    ]);


  return balance;

}


/************************************************
 * REBUILD CUSTOMER BALANCES
 ************************************************/

function rebuildMAIRIXCustomerBalanceTable() {

  const sheet =
    mairixgetSheet(
      MAIRIX_FINANCIAL_CONFIG.SHEETS.CUSTOMER_BALANCES
    );


  const movementSheet =
    getMAIRIXFinancialMovementSheet();


  const lastRow =
    movementSheet.getLastRow();


  if (
    lastRow < 2
  ) {

    return {

      rows:
        0

    };

  }


  const headers =
    movementSheet
      .getRange(
        1,
        1,
        1,
        movementSheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const rows =
    movementSheet
      .getRange(
        2,
        1,
        lastRow - 1,
        headers.length
      )
      .getValues();


  const groups = {};


  rows.forEach(
    function(row) {

      const type =
        mairixCleanText(
          row[
            headers.indexOf(
              "Party Type"
            )
          ]
        ).toUpperCase();


      const code =
        mairixCleanText(
          row[
            headers.indexOf(
              "Party Code"
            )
          ]
        );


      if (
        type !== "CUSTOMER" ||
        !code
      ) {

        return;

      }


      if (!groups[code]) {

        groups[code] = {

          code:
            code,

          name:
            mairixCleanText(
              row[
                headers.indexOf(
                  "Party Name"
                )
              ]
            ),

          effect:
            0,

          reference:
            ""

        };

      }


      groups[code].effect +=
        Number(
          row[
            headers.indexOf(
              "Customer Credit Effect"
            )
          ] || 0
        );


      groups[code].reference =
        mairixCleanText(
          row[
            headers.indexOf(
              "Reference ID"
            )
          ]
        );

    }
  );


  if (
    sheet.getLastRow() > 1
  ) {

    sheet
      .getRange(
        2,
        1,
        sheet.getLastRow() - 1,
        6
      )
      .clearContent();

  }


  const output =
    Object.keys(groups)
      .map(
        function(code) {

          const group =
            groups[code];


          return [

            group.code,

            group.name,

            0,

            group.effect,

            group.effect,

            group.reference

          ];

        }
      );


  if (
    output.length
  ) {

    sheet
      .getRange(
        2,
        1,
        output.length,
        6
      )
      .setValues(
        output
      );

  }


  return {

    rows:
      output.length

  };

}

/************************************************
 * REFERENCE DETAILS
 *
 * Used later by reports / UI.
 *
 * Any report reference can open the
 * complete financial movement.
 ************************************************/

function getMAIRIXFinancialReference(
  referenceId
) {

  const sheet =
    getMAIRIXFinancialMovementSheet();


  const lastRow =
    sheet.getLastRow();


  if (
    lastRow < 2
  ) {

    return [];

  }


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const rows =
    sheet
      .getRange(
        2,
        1,
        lastRow - 1,
        headers.length
      )
      .getValues();


  const normalized =
    mairixCleanText(
      referenceId
    );


  return rows
    .filter(
      function(row) {

        return (
          mairixCleanText(
            row[
              headers.indexOf(
                "Reference ID"
              )
            ]
          ) === normalized
        );

      }
    )
    .map(
      function(row) {

        const result = {};

        headers.forEach(
          function(header, index) {

            result[
              mairixCleanText(header)
            ] =
              row[index];

          }
        );


        return result;

      }
    );

}
