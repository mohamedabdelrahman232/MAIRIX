/************************************************
 * GET ACTIVE SPREADSHEET
 ************************************************/

function mairixGetIdentitySpreadsheet() {

  return SpreadsheetApp.getActiveSpreadsheet();

}


/************************************************
 * DELETE SHEET SAFELY
 ************************************************/

function mairixDeleteSheetIfExists(sheetName) {

  const ss =
    mairixGetIdentitySpreadsheet();

  const sheet =
    ss.getSheetByName(sheetName);


  if (!sheet) {

    return false;

  }


  /*
   * Google Sheets cannot delete the last sheet.
   */

  if (ss.getSheets().length <= 1) {

    throw new Error(
      "Cannot delete the last spreadsheet sheet."
    );

  }


  ss.deleteSheet(sheet);

  return true;

}


/************************************************
 * CREATE CLEAN SHEET
 ************************************************/

function mairixCreateCleanIdentitySheet(
  sheetName,
  headers
) {

  const ss =
    mairixGetIdentitySpreadsheet();


  const existing =
    ss.getSheetByName(sheetName);


  if (existing) {

    if (ss.getSheets().length <= 1) {

      existing.clear();

    } else {

      ss.deleteSheet(existing);

    }

  }


  const sheet =
    ss.insertSheet(sheetName);


  sheet
    .getRange(
      1,
      1,
      1,
      headers.length
    )
    .setValues(
      [headers]
    );


  mairixFormatIdentitySheet(
    sheet,
    headers.length
  );


  return sheet;

}


/************************************************
 * FORMAT IDENTITY SHEET
 ************************************************/

function mairixFormatIdentitySheet(
  sheet,
  columnCount
) {

  sheet
    .getRange(
      1,
      1,
      1,
      columnCount
    )
    .setFontWeight(
      "bold"
    );


  sheet.setFrozenRows(1);


  for (
    let column = 1;
    column <= columnCount;
    column++
  ) {

    sheet.setColumnWidth(
      column,
      170
    );

  }

}


/************************************************
 * DELETE LEGACY ACCESS SHEETS
 ************************************************/

function mairixDeleteLegacyIdentitySheets() {

  const results = {};


  MAIRIX_IDENTITY_SCHEMA
    .LEGACY_SHEETS
    .forEach(
      function(sheetName) {

        const ss =
          mairixGetIdentitySpreadsheet();


        const sheet =
          ss.getSheetByName(sheetName);


        if (!sheet) {

          results[sheetName] =
            "NOT_FOUND";

          return;

        }


        if (
          ss.getSheets().length <= 1
        ) {

          results[sheetName] =
            "SKIPPED_LAST_SHEET";

          return;

        }


        ss.deleteSheet(sheet);


        results[sheetName] =
          "DELETED";

      }
    );


  return results;

}


/************************************************
 * FULL CLEAN IDENTITY RESET
 *
 * WARNING:
 *
 * This permanently deletes all data from:
 *
 * JOB_TITLES
 * ROLES
 * EMPLOYEES
 * USERS
 * ROLE_PERMISSION_MATRIX
 * USER_PERMISSION_OVERRIDES
 * USER_WAREHOUSES
 *
 * Legacy permission sheets are also deleted.
 *
 * Business transaction sheets are untouched.
 ************************************************/

function mairixResetIdentitySchemaCompletely() {

  Logger.log(
    "=========================================="
  );

  Logger.log(
    "MAIRIX COMPLETE IDENTITY RESET"
  );

  Logger.log(
    "=========================================="
  );


  const ss =
    mairixGetIdentitySpreadsheet();


  const results = {

    legacySheets: {},

    newSheets: {}

  };


  /**********************************************
   * 1. DELETE LEGACY PERMISSION TABLES
   **********************************************/

  results.legacySheets =
    mairixDeleteLegacyIdentitySheets();


  /**********************************************
   * 2. DELETE/RECREATE OFFICIAL SCHEMA
   **********************************************/

  Object.keys(
    MAIRIX_IDENTITY_SCHEMA.SHEETS
  ).forEach(
    function(key) {

      const sheetName =
        MAIRIX_IDENTITY_SCHEMA
          .SHEETS[key];


      const headers =
        MAIRIX_IDENTITY_SCHEMA
          .HEADERS[key];


      mairixCreateCleanIdentitySheet(
        sheetName,
        headers
      );


      results.newSheets[key] =
        "CREATED";

    }
  );


  SpreadsheetApp.flush();


  Logger.log(
    "=========================================="
  );

  Logger.log(
    "IDENTITY RESET COMPLETED"
  );

  Logger.log(
    JSON.stringify(
      results,
      null,
      2
    )
  );

  Logger.log(
    "=========================================="
  );


  return {

    success: true,

    results: results

  };

}


/************************************************
 * VERIFY CLEAN IDENTITY SCHEMA
 ************************************************/

function mairixVerifyCleanIdentitySchema() {

  const ss =
    mairixGetIdentitySpreadsheet();


  const result = {

    success: true,

    sheets: {}

  };


  Object.keys(
    MAIRIX_IDENTITY_SCHEMA.SHEETS
  ).forEach(
    function(key) {

      const sheetName =
        MAIRIX_IDENTITY_SCHEMA
          .SHEETS[key];


      const expectedHeaders =
        MAIRIX_IDENTITY_SCHEMA
          .HEADERS[key];


      const sheet =
        ss.getSheetByName(sheetName);


      if (!sheet) {

        result.success = false;

        result.sheets[key] = {

          exists: false

        };

        return;

      }


      const actualHeaders =
        sheet
          .getRange(
            1,
            1,
            1,
            expectedHeaders.length
          )
          .getDisplayValues()[0];


      const invalidHeaders = [];


      for (
        let i = 0;
        i < expectedHeaders.length;
        i++
      ) {

        if (
          actualHeaders[i] !==
          expectedHeaders[i]
        ) {

          invalidHeaders.push({

            column: i + 1,

            expected:
              expectedHeaders[i],

            actual:
              actualHeaders[i]

          });

        }

      }


      const records =
        Math.max(
          0,
          sheet.getLastRow() - 1
        );


      result.sheets[key] = {

        exists: true,

        valid:
          invalidHeaders.length === 0,

        records: records,

        invalidHeaders:
          invalidHeaders

      };


      if (
        invalidHeaders.length > 0
      ) {

        result.success = false;

      }

    }
  );


  Logger.log(
    "=========================================="
  );

  Logger.log(
    "MAIRIX CLEAN IDENTITY SCHEMA VERIFICATION"
  );

  Logger.log(
    "=========================================="
  );

  Logger.log(
    JSON.stringify(
      result,
      null,
      2
    )
  );

  Logger.log(
    "=========================================="
  );


  return result;

}


/************************************************
 * FULL RESET + VERIFY TEST
 ************************************************/

function testMAIRIXCleanIdentitySchema() {

  Logger.log(
    "=========================================="
  );

  Logger.log(
    "MAIRIX CLEAN IDENTITY SCHEMA TEST"
  );

  Logger.log(
    "=========================================="
  );


  const reset =
    mairixResetIdentitySchemaCompletely();


  const verification =
    mairixVerifyCleanIdentitySchema();


  if (
    verification.success !== true
  ) {

    throw new Error(
      "MAIRIX Identity Schema verification failed."
    );

  }


  Logger.log(
    "CLEAN IDENTITY SCHEMA: PASSED"
  );

  Logger.log(
    "=========================================="
  );


  return {

    success: true,

    reset: reset,

    verification: verification

  };

}

/************************************************
 * MAIRIX ERP
 * EMPLOYEE PROVISIONING ENGINE
 *
 * RESPONSIBILITY:
 *
 * Create a complete MAIRIX identity automatically:
 *
 * EMPLOYEE
 *    ↓
 * USER
 *    ↓
 * JOB TITLE
 *    ↓
 * ROLE
 *    ↓
 * WAREHOUSE ACCESS
 *
 * ADMIN INPUT ONLY:
 *
 * 1. Employee Name
 * 2. Gmail
 * 3. Phone
 * 4. Job Title
 * 5. Password
 * 6. Confirm Password
 * 7. Allowed Warehouses
 *
 ************************************************/


/************************************************
 * EMPLOYEE / USER CODE SETTINGS
 ************************************************/

const MAIRIX_PROVISIONING = Object.freeze({

  EMPLOYEE_PREFIX: "EMP",

  USER_PREFIX: "USR",

  CODE_WIDTH: 7,

  DEFAULT_ACTIVE: true

});


/************************************************
 * CREATE COMPLETE EMPLOYEE IDENTITY
 *
 * MAIN ENTRY POINT
 ************************************************/

function mairixCreateEmployeeIdentity(payload) {

  if (!payload || typeof payload !== "object") {

    throw new Error(
      "Employee creation payload is required."
    );

  }


  const lock =
    LockService.getScriptLock();


  lock.waitLock(30000);


  try {

    /**********************************************
     * 1. NORMALIZE INPUT
     **********************************************/

    const input =
      mairixNormalizeEmployeePayload(
        payload
      );


    /**********************************************
     * 2. VALIDATE INPUT
     **********************************************/

    mairixValidateEmployeePayload(
      input
    );


    /**********************************************
     * 3. GET JOB TITLE
     **********************************************/

    const jobTitle =
      mairixGetJobTitleByCode(
        input.jobTitleCode
      );


    if (!jobTitle) {

      throw new Error(
        "Invalid Job Title Code: " +
        input.jobTitleCode
      );

    }


    /**********************************************
     * 4. VALIDATE ROLE
     **********************************************/

    const role =
      mairixGetRoleByCode(
        jobTitle.roleCode
      );


    if (!role) {

      throw new Error(
        "Role linked to Job Title does not exist: " +
        jobTitle.roleCode
      );

    }


    if (role.active !== true) {

      throw new Error(
        "Role is inactive: " +
        jobTitle.roleCode
      );

    }


    /**********************************************
     * 5. VALIDATE EMAIL UNIQUENESS
     **********************************************/

    mairixAssertEmailAvailable(
      input.email
    );


    /**********************************************
     * 6. VALIDATE WAREHOUSES
     **********************************************/

    mairixValidateEmployeeWarehouses(
      input.warehouseCodes
    );


    /**********************************************
     * 7. GENERATE CODES
     **********************************************/

    const employeeCode =
      mairixGenerateSequentialCode(
        "EMPLOYEE",
        MAIRIX_PROVISIONING.EMPLOYEE_PREFIX
      );


    const userId =
      mairixGenerateSequentialCode(
        "USER",
        MAIRIX_PROVISIONING.USER_PREFIX
      );


    /**********************************************
     * 8. CREATE PASSWORD RECORD
     **********************************************/

    const passwordData =
      mairixPrepareNewUserPassword(
        input.password,
        input.confirmPassword
      );


    /**********************************************
     * 9. CREATE EMPLOYEE
     **********************************************/

    const employee =
      mairixInsertEmployee({

        employeeCode: employeeCode,

        employeeName: input.employeeName,

        email: input.email,

        phone: input.phone,

        jobTitleCode: jobTitle.jobTitleCode,

        jobTitle: jobTitle.jobTitle,

        department: jobTitle.department,

        roleCode: jobTitle.roleCode,

        active: true

      });


    /**********************************************
     * 10. CREATE USER
     **********************************************/

    const user =
      mairixInsertUser({

        userId: userId,

        employeeCode: employeeCode,

        email: input.email,

        userName: input.employeeName,

        roleCode: jobTitle.roleCode,

        roleName: jobTitle.roleName,

        passwordData: passwordData,

        active: true

      });


    /**********************************************
     * 11. ASSIGN WAREHOUSES
     **********************************************/

    const warehouses =
      mairixAssignEmployeeWarehouses({

        employeeCode: employeeCode,

        userId: userId,

        warehouseCodes:
          input.warehouseCodes

      });


    /**********************************************
     * 12. FINAL RESULT
     **********************************************/

    const result = {

      success: true,

      employee: employee,

      user: user,

      role: {

        roleCode:
          jobTitle.roleCode,

        roleName:
          jobTitle.roleName

      },

      warehouses: warehouses,

      createdAt:
        new Date().toISOString()

    };


    Logger.log(
      "=========================================="
    );

    Logger.log(
      "MAIRIX EMPLOYEE IDENTITY CREATED"
    );

    Logger.log(
      JSON.stringify(
        result,
        null,
        2
      )
    );

    Logger.log(
      "=========================================="
    );


    return result;


  } finally {

    lock.releaseLock();

  }

}


/************************************************
 * NORMALIZE EMPLOYEE PAYLOAD
 ************************************************/

function mairixNormalizeEmployeePayload(payload) {

  return {

    employeeName:

      String(
        payload.employeeName ||
        payload.name ||
        ""
      ).trim(),


    email:

      String(
        payload.email ||
        ""
      )
        .trim()
        .toLowerCase(),


    phone:

      String(
        payload.phone ||
        ""
      ).trim(),


    jobTitleCode:

      String(
        payload.jobTitleCode ||
        ""
      ).trim(),


    password:

      String(
        payload.password ||
        ""
      ),


    confirmPassword:

      String(
        payload.confirmPassword ||
        ""
      ),


    warehouseCodes:

      Array.isArray(
        payload.warehouseCodes
      )
        ? payload.warehouseCodes
            .map(
              function(code) {

                return String(
                  code || ""
                ).trim();

              }
            )
            .filter(
              function(code) {

                return !!code;

              }
            )

        : []

  };

}


/************************************************
 * VALIDATE EMPLOYEE PAYLOAD
 ************************************************/

function mairixValidateEmployeePayload(input) {

  if (!input.employeeName) {

    throw new Error(
      "Employee Name is required."
    );

  }


  if (!input.email) {

    throw new Error(
      "Gmail / Email is required."
    );

  }


  if (!mairixIsValidEmail(
    input.email
  )) {

    throw new Error(
      "Invalid Email format."
    );

  }


  if (!input.phone) {

    throw new Error(
      "Phone number is required."
    );

  }


  if (!input.jobTitleCode) {

    throw new Error(
      "Job Title is required."
    );

  }


  if (!input.password) {

    throw new Error(
      "Password is required."
    );

  }


  if (
    input.password !==
    input.confirmPassword
  ) {

    throw new Error(
      "Password and Confirm Password do not match."
    );

  }

}


/************************************************
 * EMAIL VALIDATION
 ************************************************/

function mairixIsValidEmail(email) {

  const pattern =
    /^[^\s@]+@[^\s@]+\.[^\s@]+$/;


  return pattern.test(
    String(email || "")
      .trim()
  );

}


/************************************************
 * ASSERT EMAIL AVAILABLE
 ************************************************/

function mairixAssertEmailAvailable(email) {

  const usersSheet =
    mairixGetProvisioningSheet(
      MAIRIX_IDENTITY_SCHEMA.SHEETS.USERS
    );


  if (
    usersSheet.getLastRow() < 2
  ) {

    return true;

  }


  const headers =
    usersSheet
      .getRange(
        1,
        1,
        1,
        usersSheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const emailIndex =
    headers.indexOf(
      "Gmail"
    );


  if (emailIndex === -1) {

    throw new Error(
      'USERS sheet is missing "Gmail" column.'
    );

  }


  const values =
    usersSheet
      .getRange(
        2,
        emailIndex + 1,
        usersSheet.getLastRow() - 1,
        1
      )
      .getDisplayValues();


  const normalizedEmail =
    String(email)
      .trim()
      .toLowerCase();


  for (
    let i = 0;
    i < values.length;
    i++
  ) {

    const existingEmail =
      String(
        values[i][0] || ""
      )
        .trim()
        .toLowerCase();


    if (
      existingEmail ===
      normalizedEmail
    ) {

      throw new Error(
        "Email already belongs to another MAIRIX user."
      );

    }

  }


  return true;

}


/************************************************
 * GENERATE SEQUENTIAL CODE
 *
 * Examples:
 *
 * EMP-0000001
 * EMP-0000002
 *
 * USR-0000001
 * USR-0000002
 ************************************************/

function mairixGenerateSequentialCode(
  entity,
  prefix
) {

  const properties =
    PropertiesService
      .getScriptProperties();


  const propertyKey =
    "MAIRIX_SEQUENCE_" +
    String(entity)
      .trim()
      .toUpperCase();


  const currentValue =
    properties.getProperty(
      propertyKey
    );


  let nextNumber =
    currentValue
      ? parseInt(
          currentValue,
          10
        ) + 1
      : 1;


  if (
    isNaN(nextNumber) ||
    nextNumber < 1
  ) {

    nextNumber = 1;

  }


  properties.setProperty(
    propertyKey,
    String(nextNumber)
  );


  return (

    prefix +

    "-" +

    String(nextNumber)
      .padStart(
        MAIRIX_PROVISIONING.CODE_WIDTH,
        "0"
      )

  );

}

/************************************************
 * GET JOB TITLE BY CODE
 ************************************************/

function mairixGetJobTitleByCode(jobTitleCode) {

  const sheet =
    mairixGetProvisioningSheet(
      MAIRIX_IDENTITY_SCHEMA.SHEETS.JOB_TITLES
    );


  if (
    sheet.getLastRow() < 2
  ) {

    return null;

  }


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const codeIndex =
    headers.indexOf(
      "Job Title Code"
    );


  const titleIndex =
    headers.indexOf(
      "Job Title"
    );


  const departmentIndex =
    headers.indexOf(
      "Department"
    );


  const roleCodeIndex =
    headers.indexOf(
      "Role Code"
    );


  const roleNameIndex =
    headers.indexOf(
      "Role Name"
    );


  const activeIndex =
    headers.indexOf(
      "Active"
    );


  const values =
    sheet
      .getRange(
        2,
        1,
        sheet.getLastRow() - 1,
        sheet.getLastColumn()
      )
      .getDisplayValues();


  const normalizedCode =
    String(jobTitleCode)
      .trim();


  for (
    let i = 0;
    i < values.length;
    i++
  ) {

    const row =
      values[i];


    if (
      String(
        row[codeIndex] || ""
      ).trim()
      !==
      normalizedCode
    ) {

      continue;

    }


    const active =
      activeIndex === -1
        ? true
        : isMAIRIXTrueValue(
            row[activeIndex]
          );


    if (!active) {

      throw new Error(
        "Selected Job Title is inactive."
      );

    }


    return {

      jobTitleCode:
        row[codeIndex],

      jobTitle:
        titleIndex !== -1
          ? row[titleIndex]
          : "",

      department:
        departmentIndex !== -1
          ? row[departmentIndex]
          : "",

      roleCode:
        roleCodeIndex !== -1
          ? row[roleCodeIndex]
          : "",

      roleName:
        roleNameIndex !== -1
          ? row[roleNameIndex]
          : "",

      active:
        active

    };

  }


  return null;

}

/************************************************
 * ASSIGN EMPLOYEE WAREHOUSES
 ************************************************/

function mairixAssignEmployeeWarehouses(data) {

  /*
   * If no warehouses are selected,
   * the employee can still exist.
   *
   * Warehouse permissions are added
   * only when selected.
   */

  if (
    !data.warehouseCodes ||
    data.warehouseCodes.length === 0
  ) {

    return [];

  }


  const sheet =
    mairixGetProvisioningSheet(
      MAIRIX_IDENTITY_SCHEMA.SHEETS.USER_WAREHOUSES
    );


  const headers =
    sheet
      .getRange(
        1,
        1,
        1,
        sheet.getLastColumn()
      )
      .getDisplayValues()[0];


  const now =
    new Date();


  const created = [];


  data.warehouseCodes.forEach(
    function(warehouseCode) {

      const row =
        headers.map(
          function(header) {

            switch (
              String(header).trim()
            ) {

              case "User ID":
                return data.userId;

              case "Employee Code":
                return data.employeeCode;

              case "Warehouse Code":
                return warehouseCode;

              case "Active":
                return true;

              case "Created At":
                return now;

              default:
                return "";

            }

          }
        );


      sheet.appendRow(
        row
      );


      created.push(
        warehouseCode
      );

    }
  );


  return created;

}
